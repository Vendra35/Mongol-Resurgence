# =================================================================
# MONGOL RESURGENCE - SCRIPTABLE HINTS
# =================================================================
# Every hint_tag referenced by a situation must be defined here, or
# the reference dangles silently. All six were previously undefined.
# Structure follows vanilla in_game/common/scriptable_hints
# (hint_black_death): priority / hide / sort_priority.
# =================================================================

hint_mongol_resurgence = {
	priority = {
		can_see_situation = situation:mongol_resurgence
	}
	hide = {
		NOT = { is_situation_active = situation:mongol_resurgence }
	}
	sort_priority = 200
}
