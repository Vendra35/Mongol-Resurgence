# =================================================================
# MONGOL RESURGENCE - SCRIPTED GEOGRAPHY
# =================================================================
# Every stretch of ground this mod cares about, named once. Before
# this file the same region lists were written out 259 times across
# seven files, and a goal change meant editing six places correctly:
# the goal trigger, the wargoal's allowed_locations AND
# allowed_subjugation, the CB-grant recipient net, the AI railroad's
# find-target and fallback lists, the completion failsafe's handover,
# and the tooltip/secondary_map_color red-lining. Only ONE of those
# six had an automated check.
#
# HOW TO USE (vanilla scripted_geography.info):
#   country scope   has_presence_in = scripted_geography:X
#   location scope  is_in_scripted_geography = scripted_geography:X
#   iteration       scripted_geography:X = {
#                       every_location_in_scripted_geography = { … }
#                       any_location_in_scripted_geography   = { … }
#                       every_present_country                = { … }
#                   }
#   localisation    [ShowScriptedGeographyName( … )], needs <key> loc
#
# THE ONE RULE FOR THIS FILE: **these are ATOMS. Never define a union
# geography.** Scripted geographies do not nest (verified: zero
# vanilla definitions reference another), so a union would mean
# writing its members a second time — which is the duplication this
# file exists to delete. Callers that need several atoms OR them
# together at the call site:
#
#   OR = {
#       is_in_scripted_geography = scripted_geography:MR_geo_khorasan
#       is_in_scripted_geography = scripted_geography:MR_geo_xinjiang
#       is_in_scripted_geography = scripted_geography:MR_geo_north_china
#   }
#
# Consequence: every region and area name below appears exactly once
# in the entire mod. Adding one to a phase goal is a one-line edit
# here, and every consumer follows automatically.
# =================================================================
# --- PHASE 1 -----------------------------------------------------
# The Mongolian heartland. The Phase 1 goal, and still the Phase 1
# wargoal, CB net, failsafe and red line. Karakorum lives here, which
# is why Phases 2 and 3 keep reaching back into it.
MR_geo_heartland = {
	region = {
		mongolia_region
	}
}

# The Gobi, in TWO atoms. Phase 1's goal is presence in BOTH halves,
# and an atom holding both could only ever express "either half" —
# which is how the goal was accidentally weakened once. One atom per
# thing that must be separately true.
MR_geo_eastern_gobi = {
	area = {
		eastern_gobi_area
	}
}

MR_geo_western_gobi = {
	area = {
		western_gobi_area
	}
}

# --- PHASE 2 -----------------------------------------------------
# Four atoms. Phase 2 scores three GROUPS (20 points each) and
# mr_p2_*_cleared checks them separately; the corridor group is two
# atoms for the reason given below. Their union is the Phase 2 goal.
# The Silk Road proper, deliberately TWO atoms rather than one.
# Phase 2 wants both halves and always ORs them. Phase 3 wants
# khorasan ALONE — its failsafe reaches here only for the Samarkand
# seat (transoxiana_area, inside khorasan_region), and handing over
# Xinjiang as well would gift a claimant 136 locations no Phase 3 end
# condition asks for. A single combined atom made that impossible to
# express, so the atom is split and the callers that want both say so.
MR_geo_khorasan = {
	region = {
		khorasan_region
	}
}

MR_geo_xinjiang = {
	region = {
		xinjiang_region
	}
}

# The Yuan seat. Dadu's own region.
MR_geo_north_china = {
	region = {
		north_china_region
	}
}

# The northern marches, in THREE atoms. They form one scoring group
# (mr_p2_northern_marches_cleared ORs all three), but the Phase 2
# failsafe treats them differently when it hands territory over:
# the Siberian frontier is steppe and is cored on handover, while
# Manchuria and Tibet are settled ground the horde ruled but never
# lived in, and free cores there over-feed the AI claimant — the same
# reasoning khorasan already carried. An atom cannot be half-cored,
# so the geography follows the policy.
MR_geo_manchuria = {
	region = {
		manchuria_region
	}
}

MR_geo_tibet = {
	region = {
		tibet_region
	}
}

MR_geo_siberian_marches = {
	area = {
		bursol_area
		omsk_area
		kulykol_area
	}
}

# --- PHASE 3 -----------------------------------------------------
# Six atoms, one per goal group, for the same reason as Phase 2.

MR_geo_persia = {

	}

	}

# The middle Volga. Kazan the location sits in kazan_area, whose

MR_geo_volga = {

		kazan_area
		bolghar_area
		bashkiria_area
	}
}

# Mesopotamia. Baghdad's own area, sacked in 1258. Taken as an area

	}
}

# Song China, which fell in 1279 and completed what the Jin war began.

	}

# Korea. Goryeo submitted in 1259 and was never annexed, so the goal

	}
}

# --- REACH, NOT GOALS --------------------------------------------
# Ground the westward CB and wargoal must cover so the war is legal,
# but which no end trigger demands be cleared. Kept separate from the
# goal atoms precisely so the two never get confused.
# The Russian lands. Its own atom because russian_region plays TWO
# roles: Phase 3's end trigger wants mere PRESENCE here (a foothold),
# while the westward CB and wargoal need the whole thing in reach.
# Folding it into MR_geo_western_reach would have made the foothold
# goal impossible to express without widening it.
MR_geo_russian_lands = {
	region = {
		russian_region
	}
}

# The Urals. Split out for the same dual-role reason as the Russian
# lands: the westward CB needs the whole corridor between the Volga
# and Siberia, while the Torghut migration wants this region ALONE as
# the waypoint of its trek. The Volga AREAS inside it are a Phase 3
# goal and live in MR_geo_volga.
MR_geo_ural = {
	region = {
		ural_region
	}
}

# The rest of the westward theatre. armenian_highlands is the land
# bridge between Tabriz and Anatolia.
MR_geo_western_reach = {
	region = {
		anatolia_region
	}
	area = {
		armenian_highlands_area
	}
}

# =================================================================
# THE GREAT PARTITION (mr_great_partition) - SUCCESSOR ULUSES
# =================================================================
# Where each successor khanate is seated when the restored empire
# comes apart. These answer a DIFFERENT question from the atoms
# above — those say "what must be cleared to finish a phase", these
# say "which ulus is this ground, and does the Khaghan still hold
# it". Overlap with a goal atom is therefore expected and correct;
# the rule this file follows is one atom per distinct BOUNDARY, and
# a successor's boundary is not a phase goal's boundary.
#
# The set is deliberately the polities that ACTUALLY EXISTED on the
# steppe around 1650-1700, not the 1337 khanates. A Golden Horde
# reappearing in 1650 would be a second piece of alternate history;
# the point of this situation is the opposite — the alternate
# timeline closing back onto the real map. So the western ulus comes
# apart into Kazakh, Nogai, Crimea and Bashkir rather than into GLH.
#
# Successors that need NO new atom, because an existing one already
# draws the right boundary:
#   Khalkha / the Mongol heartland  MR_geo_heartland
#   Oirat (the Dzungars)            MR_geo_dzungaria
#   Chagatai (Yarkent, Chagatayid   MR_geo_tarim
#     until 1705, then the Khoja)
#
# Sibir (SIB) is deliberately NOT a successor. The Sibir Khanate was
# destroyed in 1598, before this situation's window opens, and by
# 1650 western Siberia is Russian. Producing one here would be a
# second piece of alternate history in the situation whose whole
# point is closing the alternate timeline. The Siberian frontier
# still feeds the cohesion score (mr_ulus_sibir_held) — losing it is
# pressure — it just yields no khanate. Russia takes that ground,
# which is what happened.

# The Kazakh Khanate (KAZ), 1465-1847 — the largest surviving
# steppe state and the true heir of the Desht-i Kipchak. Zhetysu
# (Semirechye) is included because it was Kazakh ground in this
# window; the Dzungars only took it later.
MR_geo_kazakh_steppe = {
	area = {
		desht_kipchak_area
		zhetysu_area
	}
}

# The Crimean Khanate (CRI) — the last Chinggisid state standing,
# an Ottoman vassal that outlived every other one, to 1783.
MR_geo_crimean_seat = {
	area = {
		crimea_area
	}
}

# The Nogai Horde (NOG) — the steppe around the northern Caspian,
# between the Volga mouth and the Yaik. There is no nogai_area in
# definitions.txt, so this is the ground it actually held:
# astrakhan at the Volga mouth, majar on the north Caucasus steppe,
# lower_yik reaching east toward the Ural river. Contiguous.
#
# yedisan_area is deliberately NOT here despite being named for the
# Yedisan, the Nogai's western branch: on the map it sits next to
# Moldavia and Podolia, north-west of the Black Sea — roughly
# 1500 km from the other three, with the whole Pontic steppe in
# between. Including it would make this successor a split blob
# reaching from the Ural river to the Dniester. The historical
# Yedisan was a Crimean/Ottoman vassal by this period anyway, so it
# is not the Khaghan's to lose.
MR_geo_nogai_steppe = {
	area = {
		astrakhan_area
		majar_area
		lower_yik_area
	}
}

# The Bashkirs (BSH) — the southern Urals. Split out from
# MR_geo_volga, which stays as it is: that atom is a Phase 3 goal
# and must keep its own boundary. Two atoms sharing an area is fine;
# what this file forbids is an atom REFERENCING another atom.
MR_geo_bashkiria = {
	area = {
		bashkiria_area
	}
}
