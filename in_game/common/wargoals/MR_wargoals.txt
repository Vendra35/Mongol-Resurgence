# =================================================================
# MONGOL RESURGENCE - WAR GOALS
# =================================================================
# These are what the mod's casus belli point at via war_goal_type.
# EU5 splits peace terms into wargoals with allowed_locations /
# allowed_subjugation scoping; EU4-style po_* flags do not exist.
#
# COVERAGE RULE: each wargoal's allowed_locations must cover every
# location its phase's end trigger demands, or the war can be won
# while the goal stays legally untakeable. Phase 3's goal is the
# four khanates (seats + Persia + the two footholds) — split across
# the westward CB (Rus/Volga/Persia/Anatolia theatre) and the
# silk-road CB (China/Transoxiana theatre), both granted at Phase 3
# start.
#
# Costs are cut so the AI can actually complete a railroad phase,
# following PD's PD_war_goal_prussian_ambition.
# =================================================================
# --- Phase 1: unification of the steppe --------------------------
# Deliberately narrow: this is a war for Mongolia, not a blank cheque.
MR_war_goal_steppe_unification = {
	type = superiority
	attacker = {
		conquer_cost = 1
		subjugate_cost = 1
		antagonism = 0.25
		allowed_locations = {
			scope:location.region = region:mongolia_region
		}
		allowed_subjugation = {
			OR = {
				scope:loser.capital.region = region:mongolia_region
				scope:loser.capital.region = region:xinjiang_region
				scope:loser.capital.region = region:north_china_region
				scope:loser.capital.region = region:manchuria_region
				# scope:loser.capital.region = region:steppes_region // Too far away for situaton 1.
			}
		}
	}
	defender = {
	}
	ticking_war_score = 0.5
}

# --- Phase 2/3: the Silk Road and the Ming hegemony war ----------
# Covers samarkand (khorasan), dadu (north_china), xinjiang_region
# (a Phase 2 goal region since the corridor was made contiguous)
# and mongolia_region (so a Karakorum lost after Phase 1 stays
# legally retakable). The three southern China regions stay in
# scope as conquest freedom for the Ming hegemony war, though the
# Phase 3 goal no longer demands the old southern seats.
MR_war_goal_silk_road = {
	type = superiority
	attacker = {
		conquer_cost = 1
		subjugate_cost = 1
		antagonism = 0.25
		allowed_locations = {
			OR = {
				scope:location.region = region:khorasan_region
				scope:location.region = region:xinjiang_region
				scope:location.region = region:mongolia_region
				scope:location.region = region:north_china_region
				scope:location.region = region:persia_region
				scope:location.region = region:east_china_region
				scope:location.region = region:west_china_region
				scope:location.region = region:south_china_region
				scope:location.region = region:tibet_region				## New addition for immersion
			}
		}
		allowed_subjugation = {
			OR = {
				scope:loser.capital.region = region:khorasan_region
				scope:loser.capital.region = region:xinjiang_region
				scope:loser.capital.region = region:mongolia_region
				scope:loser.capital.region = region:north_china_region
				scope:loser.capital.region = region:persia_region
				scope:loser.capital.region = region:east_china_region
				scope:loser.capital.region = region:west_china_region
				scope:loser.capital.region = region:south_china_region
				scope:loser.capital.region = region:tibet_region				## New addition for immersion
			}
		}
	}
	defender = {
	}
	ticking_war_score = 0.5
}

# --- Phase 3: the westward advance -------------------------------
# COVERAGE (must span everything mr_dominance_end_trigger demands
# in the west): sarai_al_jadid (steppes/lower_don), kazan (ural),
# the Russian foothold (russian_region), the whole Ilkhanate —
# tabriz (persia/azerbaijan), baghdad (crescent/iraq_arabi_area,
# taken as an area so the rest of the Levant stays out of scope) —
# and the Anatolian foothold (cappadocia, plus the Armenian
# highlands corridor between Tabriz and Anatolia). The
# location-to-area link is the same one PD's find-target uses
# (brandenburg_rise.txt:400, this.area = area:X).
MR_war_goal_westward_advance = {
	type = superiority
	attacker = {
		conquer_cost = 1
		subjugate_cost = 1
		antagonism = 0.25
		allowed_locations = {
			OR = {
				scope:location.region = region:russian_region
				scope:location.region = region:steppes_region
				scope:location.region = region:ural_region
				scope:location.region = region:persia_region
				scope:location.region = region:anatolia_region
				scope:location.area = area:iraq_arabi_area
				scope:location.area = area:armenian_highlands_area
			}
		}
		allowed_subjugation = {
			OR = {
				scope:loser.capital.region = region:russian_region
				scope:loser.capital.region = region:steppes_region
				scope:loser.capital.region = region:ural_region
				scope:loser.capital.region = region:persia_region
				scope:loser.capital.region = region:anatolia_region
				scope:loser.capital.area = area:iraq_arabi_area
				scope:loser.capital.area = area:armenian_highlands_area
			}
		}
	}
	defender = {
	}
	ticking_war_score = 0.5
}
