# =================================================================
# MONGOL RESURGENCE - WAR GOALS
# =================================================================
# These are what the mod's casus belli point at via war_goal_type.
# EU5 splits peace terms into wargoals with allowed_locations /
# allowed_subjugation scoping; EU4-style po_* flags do not exist.
#
# COVERAGE RULE: each wargoal's allowed_locations must cover every
# location its phase's end trigger demands, or the war can be won
# while the goal stays legally untakeable. Phase 3's goal is the
# four khanates (seats + Persia + the two footholds) — split across
# the westward CB (Rus/Volga/Persia/Anatolia theatre) and the
# silk-road CB (China/Transoxiana theatre), both granted at Phase 3
# start.
#
# Costs are cut so the AI can actually complete a railroad phase,
# following PD's PD_war_goal_prussian_ambition.
# =================================================================
# --- Phase 1: unification of the steppe --------------------------
# Deliberately narrow: this is a war for Mongolia, not a blank cheque.
MR_war_goal_steppe_unification = {
	type = superiority
	attacker = {
		conquer_cost = 0.25
		subjugate_cost = 1
		antagonism = 0.25
		allowed_locations = {
			scope:location ?= {
				OR = {
					is_in_scripted_geography = scripted_geography:MR_geo_heartland
				}
			}
		}
		# DELIBERATELY EXPLICIT, not the geography atoms. This is a
		# capital-region test that names a hand-picked set; expressing it
		# as OR(heartland, khorasan, xinjiang, north_china,
		# northern_marches) would widen it by 949 locations and let
		# Phase 1 vassalise Khorasan and Tibet. The steppes line below
		# was cut for the same reason.
		allowed_subjugation = {
			OR = {
				scope:loser.capital.region = region:mongolia_region
				scope:loser.capital.region = region:xinjiang_region
				scope:loser.capital.region = region:north_china_region
				scope:loser.capital.region = region:manchuria_region
				# scope:loser.capital.region = region:steppes_region // Too far away for situaton 1.
			}
		}
	}
	defender = {
	}
	ticking_war_score = 0.5
}

# --- Phase 2/3: the Silk Road and the Ming hegemony war ----------
# Covers samarkand (khorasan), dadu (north_china), xinjiang_region
# (a Phase 2 goal region since the corridor was made contiguous)
# and mongolia_region (so a Karakorum lost after Phase 1 stays
# legally retakable).
#
# The eastern conquest is split across the two phases the way the
# history ran, so this one wargoal has to span both:
#   Phase 2 goals  manchuria (Jin, 1234), tibet (Yuan protectorate,
#                  1240s), north_china, xinjiang, khorasan, plus the
#                  bursol/omsk/kulykol border areas
#   Phase 3 goals  east/west/south China (Song, 1279) and korea,
#                  which the end trigger accepts as a SUBJECT
#                  (Goryeo submitted in 1259 and was never annexed)
MR_war_goal_silk_road = {
	type = superiority
	attacker = {
		conquer_cost = 0.1
		subjugate_cost = 1
		antagonism = 0.25
		allowed_locations = {
			scope:location ?= {
				OR = {
					is_in_scripted_geography = scripted_geography:MR_geo_heartland
					is_in_scripted_geography = scripted_geography:MR_geo_khorasan
					is_in_scripted_geography = scripted_geography:MR_geo_xinjiang
					is_in_scripted_geography = scripted_geography:MR_geo_north_china
					is_in_scripted_geography = scripted_geography:MR_geo_manchuria
					is_in_scripted_geography = scripted_geography:MR_geo_tibet
					is_in_scripted_geography = scripted_geography:MR_geo_siberian_marches
					is_in_scripted_geography = scripted_geography:MR_geo_song_china
					is_in_scripted_geography = scripted_geography:MR_geo_korea
					is_in_scripted_geography = scripted_geography:MR_geo_persia
				}
			}
		}
		allowed_subjugation = {
			# scope:X.capital = { is_in_scripted_geography = … } is the
			# attested way to ask where a country is seated
			# (hre.txt:899, :981).
			scope:loser.capital ?= {
				OR = {
					is_in_scripted_geography = scripted_geography:MR_geo_heartland
					is_in_scripted_geography = scripted_geography:MR_geo_khorasan
					is_in_scripted_geography = scripted_geography:MR_geo_xinjiang
					is_in_scripted_geography = scripted_geography:MR_geo_north_china
					is_in_scripted_geography = scripted_geography:MR_geo_manchuria
					is_in_scripted_geography = scripted_geography:MR_geo_tibet
					is_in_scripted_geography = scripted_geography:MR_geo_siberian_marches
					is_in_scripted_geography = scripted_geography:MR_geo_song_china
					is_in_scripted_geography = scripted_geography:MR_geo_korea
					is_in_scripted_geography = scripted_geography:MR_geo_persia
				}
			}
		}
	}
	defender = {
	}
	ticking_war_score = 1
}

# --- Phase 3: the westward advance -------------------------------

# the Russian foothold (russian_region), the whole Ilkhanate —
# tabriz (persia/azerbaijan), baghdad (crescent/iraq_arabi_area,
# taken as an area so the rest of the Levant stays out of scope) —
# and the Anatolian foothold (cappadocia, plus the Armenian

# --- The Great Partition: carving the dying empire ---------------
# The vulture's wargoal. Granted to the Khaghan's NEIGHBOURS as
# cohesion falls, not to the claimant — this is the one MR wargoal
# aimed AT the Mongols rather than for them, and it is what gives a
# player who never touched the steppe a reason to care about the
# endgame.
#
# COVERAGE: the successor uluses plus the ground the empire took in
# Phases 2 and 3. Anything the Khaghan holds on a frontier is fair
# game, which is exactly what Russia and the Qing did to the real
# steppe between 1650 and 1757. The heartland is INCLUDED: unlike the
# partition's own release logic, an outside power carving Mongolia
# itself is historically the ending, not a forbidden move.
#
# No subjugation block. A neighbour carving pieces off a collapsing
# empire takes land; it does not inherit the Chinggisid claim, and
# allowing whole-realm subjugation here would let one lucky vulture
# swallow the entire Yeke Mongol Ulus in a single peace.
MR_war_goal_carve_the_ulus = {
	type = superiority
	attacker = {
		conquer_cost = 0.5
		subjugate_cost = 1
		antagonism = 0.25
		allowed_locations = {
			scope:location ?= {
				OR = {
					is_in_scripted_geography = scripted_geography:MR_geo_heartland
					is_in_scripted_geography = scripted_geography:MR_geo_kazakh_steppe
					is_in_scripted_geography = scripted_geography:MR_geo_nogai_steppe
					is_in_scripted_geography = scripted_geography:MR_geo_crimean_seat
					is_in_scripted_geography = scripted_geography:MR_geo_bashkiria
					is_in_scripted_geography = scripted_geography:MR_geo_dzungaria
					is_in_scripted_geography = scripted_geography:MR_geo_tarim
					is_in_scripted_geography = scripted_geography:MR_geo_siberian_marches
					is_in_scripted_geography = scripted_geography:MR_geo_xinjiang
					is_in_scripted_geography = scripted_geography:MR_geo_khorasan
					is_in_scripted_geography = scripted_geography:MR_geo_north_china
					is_in_scripted_geography = scripted_geography:MR_geo_manchuria

		}
	}

	}

}
