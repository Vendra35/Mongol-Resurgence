# =================================================================
# Geography verified against map_data/definitions.txt. Earlier
# guesses corrected:
#   central_asia_region  -> does not exist
#   china_region         -> north_china_region (dadu / beiping_area)
#   russia_region        -> russian_region
#   transoxiana_region   -> transoxiana_area, and its REGION is
#                           khorasan_region (not steppes_region)
#
# IMPORTANT: steppes_region is the PONTIC steppe — crimea, azov,
# lower_don, astrakhan, majar, lower_yik. It is the Golden Horde's
# western heartland, NOT the Silk Road corridor. The Samarkand
# corridor is khorasan_region (transoxiana, khwarazm, zhetysu,
# badakhshan, desht_kipchak).
#
# Where the Phase 3 four-khanate seats actually live (MGE_f's own
# formable list is no longer the goal — MGE is proclaimed at Phase
# 2's close and this phase demands the historical empire instead):
#   karakorum      upper_selenga_area   mongolia_region
#   dadu           beiping_area         north_china_region
#   samarkand      transoxiana_area     khorasan_region
#   sarai_al_jadid lower_don_area       steppes_region
#   kazan          kazan_area           ural_region
#   tabriz         azerbaijan_area      persia_region
#   baghdad        iraq_arabi_area      crescent_region
# plus persia_region cleared, a russian_region foothold and a
# cappadocia_area (anatolia_region) presence.
#
# End triggers wrap their clauses in custom_tooltip so the situation
# panel can render them as readable end requirements, the way PD's
# prussian_ambition_end_trigger does.
# =================================================================
# --- Start conditions --------------------------------------------
# Phase 1 can open once the steppe is in play and nobody has yet
# taken up the Chinggisid claim.
mr_can_start_resurgence = {
	NOT = {
		has_game_rule = mr_railroad_off
	}
	current_year >= 1368
	any_country = {
		culture = {
			has_culture_group = culture_group:mongolian_group
		}
		government_type = government_type:steppe_horde
		has_presence_in = scripted_geography:MR_geo_heartland
	}
	NOT = {
		has_global_variable = mr_railroad_complete
	}
	NOT = {
		has_global_variable = mr_railroad_failed
	}
	NOT = {
		has_global_variable = mr_resurgence_started
	}
}

mr_can_start_imperial = {
	NOT = {
		has_game_rule = mr_railroad_off
	}
	has_global_variable = mr_phase_one_complete
	NOT = {
		has_global_variable = mr_imperial_started
	}
	NOT = {
		has_global_variable = mr_railroad_failed
	}
	country_exists = c:MGO
}

mr_can_start_dominance = {
	NOT = {
		has_game_rule = mr_railroad_off
	}
	has_global_variable = mr_phase_two_complete
	NOT = {
		has_global_variable = mr_dominance_started
	}
	NOT = {
		has_global_variable = mr_railroad_failed
	}
	# MGE is proclaimed in Phase 2's on_ending now, so the Phase 3
	# claimant is normally MGE from day one; MGO is the fallback for
	# the pathological path where the proclamation failed.
	OR = {
		country_exists = c:MGE
		country_exists = c:MGO
	}
}

# --- Visibility ---------------------------------------------------
# Root is the country asking whether it can see the situation.
# The steppe powers, their neighbours, and the settled empires they
# threaten all have a stake; distant countries do not.
mr_resurgence_visible = {
	OR = {
		country_exists = c:MGO
		AND = {
			culture = {
				has_culture_group = culture_group:mongolian_group
			}
			government_type = government_type:steppe_horde
		}
		# has_presence_in, not any_owned_location: same answer, without
		# walking the country's whole holdings list. See CLAUDE.md.
		has_presence_in = sub_continent:east_asia
		has_presence_in = sub_continent:north_asia
		has_presence_in = sub_continent:central_asia
	}
}

# tag = is a plain string comparison and never errors; this = c:TAG
# and is_neighbor_of = c:TAG throw "Invalid right side" script errors
# whenever the tag is not on the map (MGE does not exist for almost
# the whole of Phase 3). Links to c:TAG are therefore always guarded
# by a preceding country_exists in the same AND.
mr_imperial_visible = {
	OR = {
		tag = MGO
		AND = {
			country_exists = c:MGO
			is_neighbor_of = c:MGO
		}
		# has_presence_in, not any_owned_location: same answer, without
		# walking the country's whole holdings list. See CLAUDE.md.
		has_presence_in = sub_continent:east_asia
		has_presence_in = sub_continent:north_asia
		has_presence_in = sub_continent:central_asia
		has_presence_in = sub_continent:middle_east		## Includes Persia, Crescent and Anatolia.
	}
}

mr_dominance_visible = {
	OR = {
		tag = MGO
		tag = MGE
		AND = {
			country_exists = c:MGO
			is_neighbor_of = c:MGO
		}
		AND = {
			country_exists = c:MGE
			is_neighbor_of = c:MGE
		}
		# has_presence_in, not any_owned_location: same answer, without
		# walking the country's whole holdings list. See CLAUDE.md.
		has_presence_in = sub_continent:east_asia
		has_presence_in = sub_continent:north_asia
		has_presence_in = sub_continent:central_asia
		has_presence_in = sub_continent:south_asia
		has_presence_in = sub_continent:south_east_asia
		has_presence_in = sub_continent:eastern_europe		## Includes Russia, Steppes region, Poland etc.
		has_presence_in = sub_continent:middle_east		## Includes Persia, Crescent and Anatolia.
		# ADDED 2026-07-31. Phase 3 is the phase that takes Persia, Iraq and
		# Anatolia, and those are precisely the conquests that made the
		# Mongols everyone else's problem. Mamluk Egypt met them at Ain
		# Jalut in 1260 and was in no audience of this mod at all; western
		# Europe sent Carpini in 1245 and Rubruck in 1253 on far less news
		# than a restored empire standing on the Levant. Both belong here,
		# and only here - neither has any business seeing Phase 1, which is
		# a quarrel between steppe hordes nobody west of Persia noticed.
		has_presence_in = sub_continent:north_africa
		has_presence_in = sub_continent:western_europe
	}
}

# --- Is this ground inside the claimant's realm? -----------------
# LOCATION scope. The seat checks used to ask `c:MGO = { owns =
# location:samarkand }`, which is true only when the claimant holds
# the seat ITSELF. A vassal holding Samarkand or Khanbaliq deadlocked
# the phase — and the completion failsafe could not break the
# deadlock either, because it deliberately never takes land from the
# claimant's own subjects. The goal was unreachable and nothing said
# so.
#
# top_owner walks the ownership chain to the top overlord in one link
# (156 vanilla uses; `conquistadors.txt:64` guards it with
# has_owner = yes exactly like this). country_exists still guards the
# c:TAG link, which errors every tick while the tag is off the map.
mr_in_claimant_realm = {
	has_owner = yes
	OR = {
		AND = {
			country_exists = c:MGO
			top_owner ?= c:MGO
		}
		AND = {
			country_exists = c:MGE
			top_owner ?= c:MGE
		}
	}
}

# --- End conditions ----------------------------------------------
# Each phase ends when its territorial goal is met. Ownership, not
# occupation: a phase should not complete off the back of a siege
# that is handed back at the peace table.
#
# The goals are the situations' red-lined REGIONS, the way PD's
# prussian_ambition_end_trigger demands its areas: the phase is done
# when no country outside the claimant's realm (itself or its
# subjects) holds a single location in the goal regions. This keeps
# the map drawing and the end requirement in agreement.
#
# The scan is region/area-scoped (PD's shape:
# NOT = { area:silesia_area = { any_ownable_location_in_area = {
# owner = { NOR = { this = c:PRU  is_subject_of = c:PRU } } } } }),
# not a world-wide any_country sweep. Two deviations from PD, both
# deliberate:
#   * owner ?= instead of owner = — khorasan/xinjiang/west_china hold
#     unowned locations, which the bare link errors on. The ?= makes
#     an ownerless location simply not match, which is what we want.
#   * two tags to clear instead of one, so FOUR NOR members.
#
# ONE custom_tooltip PER REQUIREMENT, and each text on ONE line. The
# situation panel renders end conditions as a CHECKLIST — one tick per
# custom_tooltip — so a single tooltip wrapped around every clause
# fights the widget: Phase 3 showed its text AND the raw clause-by-
# clause breakdown underneath. Vanilla's eight situation end-condition
# tooltips are all one compact sentence and not one of them contains a
# newline. Each mr_p2_*/mr_p3_* goal trigger therefore carries its own
# tooltip internally, and the end triggers are just a list of them.
#
# The region scans below use owner ?= rather than owner =: khorasan,
# xinjiang and west China contain unowned locations and the bare link
# errors on them, while ?= makes an ownerless location simply not
# match — which is what "nobody outside our realm holds it" means.
#
# WALK THE WHOLE SUBJECT CHAIN. `is_subject_of = c:MGO` is true only
# for a DIRECT vassal, so a vassal's vassal counted as a foreign power
# and its ground blocked the goal — found in Phase 2 testing, where
# the phase simply refused to end over land the claimant effectively
# ruled. `top_overlord_or_this ?= c:MGO` is vanilla's idiom for
# "MGO, or anything anywhere under MGO" (situation_triggers.txt:80,
# hundred_years_war.txt:185) and it also returns the country itself
# when it has no overlord — which is why the separate `tag = MGO`
# member is gone: it is subsumed.
# country_exists still guards the link, because c:TAG on the right
# side errors every tick while the tag is off the map.
# Phase 1: the entire Mongolian heartland is under one banner.
mr_resurgence_end_trigger = {
	custom_tooltip = {
		text = mr_resurgence_seats_tt
		country_exists = c:MGO
		location:karakorum = {
			mr_in_claimant_realm = yes
		}
		scripted_geography:MR_geo_eastern_gobi = {
			any_location_in_scripted_geography = {
				mr_in_claimant_realm = yes
			}
		}
		scripted_geography:MR_geo_western_gobi = {
			any_location_in_scripted_geography = {
				mr_in_claimant_realm = yes
			}
		}
	}
	custom_tooltip = {
		text = mr_p1_heartland_cleared_tt
		NOT = {
			scripted_geography:MR_geo_heartland = {
				any_location_in_scripted_geography = {
					owner ?= {
						NOT = {
							AND = {
								country_exists = c:MGO
								top_overlord_or_this ?= c:MGO
							}
						}
					}
				}
			}
		}
	}
}

# LATE-STEPPE SITUATIONS (1600s)
# =================================================================
# Three moments vanilla has no content for at all — verified absent:
# kalmyk/torghut, dzungar/zunghar and chahar all return zero files
# across events, common and setup. They run in the Phase 3 era and
# are built on tags that do exist (OIR, MGO, MGE), since none of
# the historical successor states have their own tag.
# --- Ligdan Khan and the Chahar (1604-1634) ----------------------
# The last Chinggisid Great Khan's failed reunification: the dark
# mirror of Phase 1. Only meaningful while a Chinggisid claimant
# actually exists to be challenged.
mr_can_start_chahar = {
	NOT = {
		has_game_rule = mr_railroad_off
	}
	current_year >= 1604
	current_year < 1634
	NOT = {
		has_global_variable = mr_chahar_started
	}
	# Ligdan's reunification bid needs a FRAGMENTED steppe. It fires
	# whether or not the railroad failed — that failure is exactly the
	# world where the last Chinggisid tries and fails — but not when a
	# claimant already dominates the heartland.
	NOT = {
		any_country = {

# hold this ground? These are the INVERSE of the mr_p2_*/mr_p3_*
# goal triggers above — those ask "is this clear of outsiders" while

# consumed by both the end trigger and the monthly panel. When an

# `any_location_in_scripted_geography` + `owner ?=` is the attested
# shape (see mr_p3_volga_cleared); the bare `owner =` link errors on
# ownerless locations.
#
# THE OWNERLESS TRAP (found in the first Great Partition game test,
# 2026-07-30). These eight triggers used to ask
# `NOT = { mr_in_claimant_realm = yes }` directly on the location.
# mr_in_claimant_realm opens with `has_owner = yes`, so negating it
# turns "this ground has no owner" into "this ground is lost" — and
# every atom below contains ground that can NEVER have an owner:
# default.map lists 918 lakes, 1868 impassable mountains and 153
# non_ownable locations, and definitions.txt files them inside the
# ordinary area/region tree. mongolia_region alone holds 50 of them
# (buir_nuur_province carries lake_buir, which is what the end-
# condition tooltip was showing). Measured across the eight atoms:
# heartland 50/263, kazakh 28/183, tarim 34/88, siberian marches
# 34/101, dzungaria 12/48, crimea 4/18, nogai 3/98, bashkiria 2/28.
# Not one of them could ever read "held", so mr_partition_end_trigger
# was true on the situation's first tick, cohesion computed to 0, and
# the endgame opened and resolved inside a single month.
#
# The fix is the shape the paragraph above already promised and the
# mr_p2_*/mr_p3_* goal triggers already use: reach the owner through
# `owner ?=`, which makes an ownerless location simply not match.
# That is also why the Phase 1-3 goals were never hit by this — they
# were written this way from the start.

# =================================================================
# THE GREAT PARTITION - LIFECYCLE TRIGGERS
# =================================================================
# The situation opens on a FLAG, not on a calendar date. Phase 3
# ends when its goal is met, which under the Terminator pace can be
# decades before its 1650 deadline; a fixed start date would leave a
# fast campaign idling for seventy years. Phase 3's on_ending fires
# a delayed event (mr_dominance.140, years = 5) that sets
# mr_partition_ready, so the opening tracks when the empire was
# actually finished. The delay was 30 years until the user's balance
# pass shortened it to 5; the `current_date > 1600` clause below is
# what actually sets the floor now, so a fast campaign lands on 1600
# and a slow one opens five years after it wins:
#   Phase 3 done 1560 -> flag 1565 -> opens 1600 (the floor)
#   Phase 3 done 1650 -> flag 1655 -> opens 1655
# The schedule runs 56 years from the opening, so even the latest
# start finishes inside the situation's own 1600-1720 window.
mr_can_start_partition = {
	NOT = {
		has_game_rule = mr_railroad_off
	}
	# THERE MUST BE AN EMPIRE TO COME APART - and that is a different
	# question from "did the railroad win". This used to demand
	# mr_railroad_complete AND forbid mr_railroad_failed, which meant a
	# Phase 3 that failed left the Khaghan holding Mongolia, Manchuria,
	# Tibet, north China, Xinjiang and Khorasan for the rest of the game
	# with nothing able to take any of it back. That is not a rare corner:
	# the completion failsafe never takes land from a HUMAN or their
	# subjects, by design, so any player holding Persia, Anatolia, Ming
	# China or Russia can make Phase 3 fail simply by surviving. The
	# permanently-alternate map that produced is the one thing this
	# situation exists to prevent.
	#
	# mr_phase_two_complete is the honest gate: the north and the east
	# were taken, so there IS an imperial realm. Phase 1 or Phase 2
	# failing still means no partition - a Khaghan who never got past the
	# steppe has nothing to give back, and cohesion would open near 30 and
	# end the situation on its first tick.
	has_global_variable = mr_phase_two_complete
	has_global_variable = mr_partition_ready
	NOT = {
		has_global_variable = mr_partition_started
	}
	# A resolved partition never reopens. Both terminals are checked,
	# not just one: on_ending sets exactly one of them and there is no
	# path that sets neither.
	NOT = {
		has_global_variable = mr_partition_collapsed
	}
	NOT = {
		has_global_variable = mr_partition_survived
	}
	current_date > 1600.1.1
	# The Khaghan is normally MGE; MGO is the failed-proclamation
	# fallback, kept for the same reason Phase 3 keeps it.
	OR = {
		country_exists = c:MGE
		country_exists = c:MGO
	}
}

# THREE ways the empire comes apart, and they are deliberately not
# the same kind of thing. Time expiry is the fourth way out, so
# on_ending's "held" branch stays reachable.
#
# 1. The tag is gone from the map. Nothing left to hold anything.
# 2. KARAKORUM has left the realm. The seat, not the region — the
#    same clause Phase 1 uses to say the heartland is won, read
#    backwards. This used to be `NOT = { mr_ulus_heartland_held }`,
#    i.e. ANY of 263 heartland locations outside the realm, which
#    after the ownerless fix would still end a century-long endgame
#    over one lost border village.
# 3. COHESION HAS RUN OUT (user ruling 2026-07-30). Karakorum alone
#    is not enough and cannot be: it sits deeper in Asia than any
#    European power can reach, so an empire that has lost every
#    western ulus but kept its seat would never collapse — it would
#    idle to the 1720 timeout and collect the survival reward. The
#    cohesion floor is the clause that makes the outer uluses matter,
#    and it is the AND the old comment here promised but the code
#    never had.
#
# mr_ulus_heartland_held keeps its job: it is worth 30 cohesion in
# on_monthly, so losing part of Mongolia is heavy pressure rather
# than an instant ending.
#
# var: at situation scope is how situations read their own variables
# in can_end (great_pestilence.txt:14-18 does exactly this); the
# has_variable guard is attested at the same scope in
# black_death.txt:158 and costs nothing, and an unset variable
# reading back as 0 would be this bug all over again.
mr_partition_end_trigger = {
	OR = {
		custom_tooltip = {
			text = mr_partition_khaghan_gone_tt
			NOR = {
				country_exists = c:MGE
				country_exists = c:MGO
			}
		}
		custom_tooltip = {
			text = mr_partition_heartland_lost_tt
			NOT = {
				location:karakorum = {
					mr_in_claimant_realm = yes
				}
			}
		}
		custom_tooltip = {
			text = mr_partition_cohesion_spent_tt
			has_variable = MR_cohesion_score
			var:MR_cohesion_score < 40
		}
	}
}

	# THE WORLD THAT HISTORICALLY HEARD. Three sub-continents were added
	# 2026-07-31 and each has a reason:
	#   south_east_asia - the Mongols invaded Dai Viet, Champa, Pagan in
	#     1287 and Java in 1293. Phase 3 already showed them the empire;
	#     leaving them out of its collapse was an inconsistency.
	#   north_africa - Mamluk Egypt BEAT the Mongols at Ain Jalut in 1260,
	#     the most famous defeat they ever took. Egypt was in no audience

	#   western_europe - they saw no Mongol army and heard everything.
	#     Frederick II's circular letters, Louis IX, the council of 1241,
	#     Carpini sent in 1245 and Rubruck in 1253, Matthew Paris writing
	#     it down in England, and the Baltic herring market collapsing in
	#     1238 because Gotland's fishermen would not sail. News, envoys
	#     and panic, which is exactly what a spectator beat is.
	# Deliberately absent: sub-Saharan Africa, the Americas, Oceania. No
	# word of the Mongols reached them and none should.
	OR = {
		has_presence_in = sub_continent:east_asia
		has_presence_in = sub_continent:north_asia
		has_presence_in = sub_continent:central_asia
		has_presence_in = sub_continent:south_asia
		has_presence_in = sub_continent:south_east_asia
		has_presence_in = sub_continent:eastern_europe
		has_presence_in = sub_continent:middle_east
		has_presence_in = sub_continent:north_africa
		has_presence_in = sub_continent:western_europe
	}
}

# =================================================================
# THE WORLD BEYOND THE CONQUEST - who hears the news, and when
# =================================================================
# The Great Partition always had three audiences: the Khaghan, his
# neighbours, and distant spectators. The three phases that BUILD the
# empire had only two - the claimant, and whoever held the ground it
# wanted. So a player in western Europe, in Egypt or in India could
# watch a Mongol empire grow from Korea to Anatolia across three
# centuries and never be told a thing.
#
# This is the outer ring, and it is drawn on what the world actually
# heard in the thirteenth century rather than on distance. Everyone
# here has a historical reason to receive word from the east; the
# Americas, Oceania and sub-Saharan Africa have none and are absent.
# The inner ring - the steppe itself, and whoever holds the ground a
# phase is currently taking - already gets its own targeted events, so
# neighbours are excluded here and nobody hears the same news twice.
mr_distant_observer = {
	NOR = {
		tag = MGE
		tag = MGO
	}
	NOT = {
		AND = {
			country_exists = c:MGO
			is_neighbor_of = c:MGO
		}
	}
	NOT = {
		AND = {
			country_exists = c:MGE
			is_neighbor_of = c:MGE
		}
	}
	OR = {
		has_presence_in = sub_continent:western_europe
		has_presence_in = sub_continent:eastern_europe
		has_presence_in = sub_continent:north_africa
		has_presence_in = sub_continent:middle_east
		has_presence_in = sub_continent:south_asia
		has_presence_in = sub_continent:south_east_asia
	}
}
