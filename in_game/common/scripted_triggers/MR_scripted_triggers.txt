# =================================================================
# MONGOL RESURGENCE - SCRIPTED TRIGGERS
# =================================================================
# Region names verified against vanilla. The fabricated ones this
# file used to reference resolve as follows:
#   central_asia_region  -> steppes_region
#   china_region         -> north_china_region
#   russia_region        -> russian_region
#   transoxiana_region   -> no vanilla equivalent; keyed on
#                           location:samarkand instead
# =================================================================


# --- Start conditions --------------------------------------------

# Phase 1 can open once the steppe is in play and nobody has yet
# taken up the Chinggisid claim.
mr_can_start_resurgence = {
	NOT = { has_game_rule = mr_railroad_off }
	current_year >= 1368
	any_country = {
		culture = { has_culture_group = culture_group:mongolian_group }
		government_type = government_type:steppe_horde
		any_owned_location = {
			region = region:mongolia_region
		}
	}
	NOT = { has_global_variable = mr_railroad_complete }
	NOT = { has_global_variable = mr_railroad_failed }
	NOT = { has_global_variable = mr_resurgence_started }
}

mr_can_start_imperial = {
	NOT = { has_game_rule = mr_railroad_off }
	has_global_variable = mr_phase_one_complete
	NOT = { has_global_variable = mr_imperial_started }
	NOT = { has_global_variable = mr_railroad_failed }
	country_exists = c:MGO
}

mr_can_start_dominance = {
	NOT = { has_game_rule = mr_railroad_off }
	has_global_variable = mr_phase_two_complete
	NOT = { has_global_variable = mr_dominance_started }
	NOT = { has_global_variable = mr_railroad_failed }
	country_exists = c:MGO
}


# --- Visibility ---------------------------------------------------
# Root is the country asking whether it can see the situation.
# The steppe powers, their neighbours, and the settled empires they
# threaten all have a stake; distant countries do not.

mr_resurgence_visible = {
	OR = {
		country_exists = c:MGO
		AND = {
			culture = { has_culture_group = culture_group:mongolian_group }
			government_type = government_type:steppe_horde
		}
		any_owned_location = {
			region = region:mongolia_region
		}
		any_owned_location = {
			region = region:steppes_region
		}
		any_owned_location = {
			region = region:north_china_region
		}
	}
}

mr_imperial_visible = {
	OR = {
		this = c:MGO
		is_neighbor_of = c:MGO
		any_owned_location = {
			region = region:steppes_region
		}
		any_owned_location = {
			region = region:north_china_region
		}
		any_owned_location = {
			region = region:persia_region
		}
	}
}

mr_dominance_visible = {
	OR = {
		this = c:MGO
		this = c:MGE
		is_neighbor_of = c:MGO
		is_neighbor_of = c:MGE
		any_owned_location = {
			region = region:russian_region
		}
		any_owned_location = {
			region = region:steppes_region
		}
		any_owned_location = {
			region = region:persia_region
		}
	}
}


# --- End conditions ----------------------------------------------
# Each phase ends when its territorial goal is met. Ownership, not
# occupation: a phase should not complete off the back of a siege
# that is handed back at the peace table.

# Phase 1: the Mongolian heartland is under one banner.
mr_resurgence_end_trigger = {
	country_exists = c:MGO
	c:MGO = {
		owns = location:karakorum
		any_owned_location = {
			region = region:mongolia_region
		}
	}
}

# Phase 2: the Silk Road corridor is held from the steppe to Persia.
mr_imperial_end_trigger = {
	country_exists = c:MGO
	c:MGO = {
		owns = location:samarkand
		any_owned_location = {
			region = region:steppes_region
		}
		any_owned_location = {
			region = region:north_china_region
		}
	}
}

# Phase 3: the four seats of the old khanates are held at once —
# Karakorum, Sarai, Zhongdu and Baghdad.
mr_dominance_end_trigger = {
	OR = {
		country_exists = c:MGE
		country_exists = c:MGO
	}
	any_country = {
		OR = {
			this = c:MGE
			this = c:MGO
		}
		owns = location:karakorum
		owns = location:sarai_al_jadid
		owns = location:zhongdu
		owns = location:baghdad
	}
}
