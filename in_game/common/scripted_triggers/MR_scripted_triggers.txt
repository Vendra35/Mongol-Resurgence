# Mongol Railroad Mod - Scripted Triggers

# =============================================
# On-Action Conditions
# =============================================

mr_can_start_resurgence = {
	has_game_rule = { rule = rule_mongol_railroad setting = historical }
	current_year >= 1337
	any_country = {
		culture = { has_culture_group = culture_group:mongolian_group }
		government_type = government_type:steppe_horde
		any_owned_location = { is_in_region = region:mongolia_region }
	}
	NOT = { has_global_variable = mongol_railroad_complete }
	NOT = { has_global_variable = mongol_railroad_failed }
	NOT = { has_global_variable = mongol_resurgence_emerged }
}

mr_can_start_imperial = {
	has_global_variable = mongol_resurgence_complete
	NOT = { has_global_variable = mongol_imperial_active }
	NOT = { has_global_variable = mongol_imperial_complete }
	NOT = { has_global_variable = mongol_railroad_failed }
	country_exists = c:MGO
}

mr_can_start_dominance = {
	has_global_variable = mongol_imperial_complete
	NOT = { has_global_variable = mongol_dominance_active }
	NOT = { has_global_variable = mongol_railroad_complete }
	NOT = { has_global_variable = mongol_railroad_failed }
	country_exists = c:MGE
}

# =============================================
# Situation Visibility & End Triggers
# =============================================

mr_resurgence_visible = {
	has_global_variable = mongol_resurgence_active
	OR = {
		country_exists = c:MGO
		any_country = {
			culture = { has_culture_group = culture_group:mongolian_group }
			government_type = government_type:steppe_horde
			any_owned_location = { is_in_region = region:mongolia_region }
		}
	}
}

mr_resurgence_end_trigger = {
	country_exists = c:MGO
	c:MGO = {
		culture = { has_culture_group = culture_group:mongolian_group }
		government_type = government_type:steppe_horde
		any_owned_location = { is_in_region = region:mongolia_region }
		any_owned_location = { is_in_region = region:central_asia_region }
		total_development = 150
	}
}

mr_imperial_visible = {
	has_global_variable = mongol_imperial_active
	country_exists = c:MGE
}

mr_imperial_end_trigger = {
	country_exists = c:MGE
	c:MGE = {
		total_development = 500
		any_owned_location = { is_in_region = region:persia_region }
		any_owned_location = { is_in_region = region:china_region }
		any_owned_location = { is_in_region = region:russia_region }
	}
}

mr_dominance_visible = {
	has_global_variable = mongol_dominance_active
	country_exists = c:MGE
}

mr_dominance_end_trigger = {
	c:MGE = {
		controls = location:sarai_al_jadid
		controls = location:zhongdu
		controls = location:baghdad
		controls = location:samarkand
	}
}
