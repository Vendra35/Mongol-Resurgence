# =================================================================
# MONGOL RESURGENCE - SCRIPTED TRIGGERS
# =================================================================
# Region names verified against vanilla. The fabricated ones this
# file used to reference resolve as follows:
#   central_asia_region  -> steppes_region
#   china_region         -> north_china_region
#   russia_region        -> russian_region
#   transoxiana_region   -> no vanilla equivalent; keyed on
#                           location:samarkand instead
# =================================================================


# --- Start conditions --------------------------------------------

# Phase 1 can open once the steppe is in play and nobody has yet
# taken up the Chinggisid claim.
mr_can_start_resurgence = {
	NOT = { has_game_rule = mr_railroad_off }
	current_year >= 1368
	any_country = {
		culture = { has_culture_group = culture_group:mongolian_group }
		government_type = government_type:steppe_horde
		any_owned_location = {
			region = region:mongolia_region
		}
	}
	NOT = { has_global_variable = mr_railroad_complete }
	NOT = { has_global_variable = mr_railroad_failed }
	NOT = { has_global_variable = mr_resurgence_started }
}

mr_can_start_imperial = {
	NOT = { has_game_rule = mr_railroad_off }
	has_global_variable = mr_phase_one_complete
	NOT = { has_global_variable = mr_imperial_started }
	NOT = { has_global_variable = mr_railroad_failed }
	country_exists = c:MGO
}

mr_can_start_dominance = {
	NOT = { has_game_rule = mr_railroad_off }
	has_global_variable = mr_phase_two_complete
	NOT = { has_global_variable = mr_dominance_started }
	NOT = { has_global_variable = mr_railroad_failed }
	country_exists = c:MGO
}


# --- Visibility ---------------------------------------------------
# Root is the country asking whether it can see the situation.
# The steppe powers, their neighbours, and the settled empires they
# threaten all have a stake; distant countries do not.

mr_resurgence_visible = {
	OR = {
		country_exists = c:MGO
		AND = {
			culture = { has_culture_group = culture_group:mongolian_group }
			government_type = government_type:steppe_horde
		}
		any_owned_location = {
			region = region:mongolia_region
		}
		any_owned_location = {
			region = region:steppes_region
		}
		any_owned_location = {
			region = region:north_china_region
		}
	}
}

mr_imperial_visible = {
	OR = {
		this = c:MGO
		is_neighbor_of = c:MGO
		any_owned_location = {
			region = region:steppes_region
		}
		any_owned_location = {
			region = region:north_china_region
		}
		any_owned_location = {
			region = region:persia_region
		}
	}
}

mr_dominance_visible = {
	OR = {
		this = c:MGO
		this = c:MGE
		is_neighbor_of = c:MGO
		is_neighbor_of = c:MGE
		any_owned_location = {
			region = region:russian_region
		}
		any_owned_location = {
			region = region:steppes_region
		}
		any_owned_location = {
			region = region:persia_region
		}
	}
}


# --- End conditions ----------------------------------------------
# Each phase ends when its territorial goal is met. Ownership, not
# occupation: a phase should not complete off the back of a siege
# that is handed back at the peace table.

# Phase 1: the Mongolian heartland is under one banner.
mr_resurgence_end_trigger = {
	country_exists = c:MGO
	c:MGO = {
		owns = location:karakorum
		any_owned_location = {
			region = region:mongolia_region
		}
	}
}

# Phase 2: the Silk Road corridor is held from the steppe to Persia.
mr_imperial_end_trigger = {
	country_exists = c:MGO
	c:MGO = {
		owns = location:samarkand
		any_owned_location = {
			region = region:steppes_region
		}
		any_owned_location = {
			region = region:north_china_region
		}
	}
}

# Phase 3: the four seats of the old khanates are held at once —
# Karakorum, Sarai, Dadu and Baghdad.
mr_dominance_end_trigger = {
	OR = {
		country_exists = c:MGE
		country_exists = c:MGO
	}
	any_country = {
		OR = {
			this = c:MGE
			this = c:MGO
		}
		owns = location:karakorum
		owns = location:sarai_al_jadid
		owns = location:dadu
		owns = location:baghdad
	}
}


# =================================================================
# LATE-STEPPE SITUATIONS (1600s)
# =================================================================
# Three moments vanilla has no content for at all — verified absent:
# kalmyk/torghut, dzungar/zunghar and chahar all return zero files
# across events, common and setup. They run in the Phase 3 era and
# are built on tags that do exist (OIR, MGO, MGE), since none of
# the historical successor states have their own tag.

# --- Ligdan Khan and the Chahar (1604-1634) ----------------------
# The last Chinggisid Great Khan's failed reunification: the dark
# mirror of Phase 1. Only meaningful while a Chinggisid claimant
# actually exists to be challenged.
mr_can_start_chahar = {
	NOT = { has_game_rule = mr_railroad_off }
	current_year >= 1604
	current_year < 1634
	NOT = { has_global_variable = mr_chahar_started }
	NOT = { has_global_variable = mr_railroad_failed }
	any_country = {
		culture = { has_culture_group = culture_group:mongolian_group }
		government_type = government_type:steppe_horde
		any_owned_location = { region = region:mongolia_region }
	}
}

mr_chahar_visible = {
	OR = {
		this = c:MGO
		this = c:MGE
		any_owned_location = { region = region:mongolia_region }
		any_owned_location = { region = region:north_china_region }
	}
}

# Ends when one power holds the Mongolian heartland outright.
mr_chahar_end_trigger = {
	any_country = {
		culture = { has_culture_group = culture_group:mongolian_group }
		owns = location:karakorum
		any_owned_location = { region = region:north_china_region }
	}
}

# =================================================================
# READ-ONLY VANILLA HOOKS
# =================================================================
# We never write to vanilla state — these only read it, so there is
# no compatibility risk and no file is overwritten.
#
# Tumu: flavor_chi_mon sets lost_emperor on the Middle Kingdom's
# leader country while the emperor is held. It persists until
# clear_tumu_crisis_variables runs, so it is safe to read.
#
# NOTE: no equivalent hook exists for the Treaty of Ugra. Vanilla's
# treaty_of_ugra / treaty_of_ugra_declined variables are removed as
# soon as the event resolves (flavor_MOS.txt:2473, :2541), so there
# is nothing durable left to read. Do not add one without first
# confirming a persistent signal exists.
mr_vanilla_tumu_crisis_active = {
	any_country = {
		has_variable = lost_emperor
	}
}
