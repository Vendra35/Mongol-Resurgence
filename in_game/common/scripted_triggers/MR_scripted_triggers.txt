# =================================================================
# MONGOL RESURGENCE - SCRIPTED TRIGGERS
# =================================================================
# Geography verified against map_data/definitions.txt. Earlier
# guesses corrected:
#   central_asia_region  -> does not exist
#   china_region         -> north_china_region (dadu / beiping_area)
#   russia_region        -> russian_region
#   transoxiana_region   -> transoxiana_area, and its REGION is
#                           khorasan_region (not steppes_region)
#
# IMPORTANT: steppes_region is the PONTIC steppe — crimea, azov,
# lower_don, astrakhan, majar, lower_yik. It is the Golden Horde's
# western heartland, NOT the Silk Road corridor. The Samarkand
# corridor is khorasan_region (transoxiana, khwarazm, zhetysu,
# badakhshan, desht_kipchak).
#
# Where the Phase 3 four-khanate seats actually live (MGE_f's own
# formable list is no longer the goal — MGE is proclaimed at Phase
# 2's close and this phase demands the historical empire instead):
#   karakorum      upper_selenga_area   mongolia_region
#   dadu           beiping_area         north_china_region
#   samarkand      transoxiana_area     khorasan_region
#   sarai_al_jadid lower_don_area       steppes_region
#   kazan          kazan_area           ural_region
#   tabriz         azerbaijan_area      persia_region
#   baghdad        iraq_arabi_area      crescent_region
# plus persia_region cleared, a russian_region foothold and a
# cappadocia_area (anatolia_region) presence.
#
# End triggers wrap their clauses in custom_tooltip so the situation
# panel can render them as readable end requirements, the way PD's
# prussian_ambition_end_trigger does.
# =================================================================
# --- Start conditions --------------------------------------------
# Phase 1 can open once the steppe is in play and nobody has yet
# taken up the Chinggisid claim.
mr_can_start_resurgence = {
	NOT = {
		has_game_rule = mr_railroad_off
	}
	current_year >= 1368
	any_country = {
		culture = {
			has_culture_group = culture_group:mongolian_group
		}
		government_type = government_type:steppe_horde
		any_owned_location = {
			region = region:mongolia_region
		}
	}
	NOT = {
		has_global_variable = mr_railroad_complete
	}
	NOT = {
		has_global_variable = mr_railroad_failed
	}
	NOT = {
		has_global_variable = mr_resurgence_started
	}
}

mr_can_start_imperial = {
	NOT = {
		has_game_rule = mr_railroad_off
	}
	has_global_variable = mr_phase_one_complete
	NOT = {
		has_global_variable = mr_imperial_started
	}
	NOT = {
		has_global_variable = mr_railroad_failed
	}
	country_exists = c:MGO
}

mr_can_start_dominance = {
	NOT = {
		has_game_rule = mr_railroad_off
	}
	has_global_variable = mr_phase_two_complete
	NOT = {
		has_global_variable = mr_dominance_started
	}
	NOT = {
		has_global_variable = mr_railroad_failed
	}
	# MGE is proclaimed in Phase 2's on_ending now, so the Phase 3
	# claimant is normally MGE from day one; MGO is the fallback for
	# the pathological path where the proclamation failed.
	OR = {
		country_exists = c:MGE
		country_exists = c:MGO
	}
}

# --- Visibility ---------------------------------------------------
# Root is the country asking whether it can see the situation.
# The steppe powers, their neighbours, and the settled empires they
# threaten all have a stake; distant countries do not.
mr_resurgence_visible = {
	OR = {
		country_exists = c:MGO
		AND = {
			culture = {
				has_culture_group = culture_group:mongolian_group
			}
			government_type = government_type:steppe_horde
		}
		any_owned_location = {
			sub_continent = sub_continent:east_asia
		}
		any_owned_location = {
			sub_continent = sub_continent:north_asia
		}
		any_owned_location = {
			sub_continent = sub_continent:central_asia
		}
	}
}

# tag = is a plain string comparison and never errors; this = c:TAG
# and is_neighbor_of = c:TAG throw "Invalid right side" script errors
# whenever the tag is not on the map (MGE does not exist for almost
# the whole of Phase 3). Links to c:TAG are therefore always guarded
# by a preceding country_exists in the same AND.
mr_imperial_visible = {
	OR = {
		tag = MGO
		AND = {
			country_exists = c:MGO
			is_neighbor_of = c:MGO
		}
		any_owned_location = {
			sub_continent = sub_continent:east_asia
		}
		any_owned_location = {
			sub_continent = sub_continent:north_asia
		}
		any_owned_location = {
			sub_continent = sub_continent:central_asia
		}
		## Includes Persia, Crescent and Anatolia.
		any_owned_location = {
			sub_continent = sub_continent:middle_east
		}
	}
}

mr_dominance_visible = {
	OR = {
		tag = MGO
		tag = MGE
		AND = {
			country_exists = c:MGO
			is_neighbor_of = c:MGO
		}
		AND = {
			country_exists = c:MGE
			is_neighbor_of = c:MGE
		}
		any_owned_location = {
			sub_continent = sub_continent:east_asia
		}
		any_owned_location = {
			sub_continent = sub_continent:north_asia
		}
		any_owned_location = {
			sub_continent = sub_continent:central_asia
		}
		any_owned_location = {
			sub_continent = sub_continent:south_asia
		}
		any_owned_location = {
			sub_continent = sub_continent:south_east_asia
		}
		## Includes Russia, Steppes region, Poland etc.
		any_owned_location = {
			sub_continent = sub_continent:eastern_europe
		}
		any_owned_location = {
			sub_continent = sub_continent:middle_east
		}
	}
}

# --- End conditions ----------------------------------------------
# Each phase ends when its territorial goal is met. Ownership, not
# occupation: a phase should not complete off the back of a siege
# that is handed back at the peace table.
#
# The goals are the situations' red-lined REGIONS, the way PD's
# prussian_ambition_end_trigger demands its areas: the phase is done
# when no country outside the claimant's realm (itself or its
# subjects) holds a single location in the goal regions. This keeps
# the map drawing and the end requirement in agreement.
# Phase 1: the entire Mongolian heartland is under one banner.
mr_resurgence_end_trigger = {
	custom_tooltip = {
		text = mr_resurgence_end_tt
		country_exists = c:MGO
		c:MGO = {
			owns = location:karakorum
			has_presence_in = area:eastern_gobi_area
			has_presence_in = area:western_gobi_area
		}
		NOT = {
			any_country = {
				NOR = {
					tag = MGO
					is_subject_of = c:MGO
				}
				any_owned_location = {
					region = region:mongolia_region
				}
			}
		}
	}
}

# Phase 2: the whole Silk Road corridor — Khorasan, Xinjiang and
# North China — answers to the claimant or its subjects. DUAL-TAG
# like Phase 3's trigger: under the Vanilla buff rule nothing
# blocks country formation, so a HUMAN claimant can legally form
# MGE mid-phase via the formable — a c:MGO-only trigger then made
# the phase unwinnable for the very player who mastered the
# corridor.
mr_imperial_end_trigger = {
	custom_tooltip = {
		text = mr_imperial_end_tt
		any_country = {
			OR = {
				tag = MGO
				tag = MGE
			}
			owns = location:samarkand
			owns = location:dadu
		}
		NOT = {
			any_country = {
				NOR = {
					tag = MGO
					tag = MGE
					AND = {
						country_exists = c:MGO
						is_subject_of = c:MGO
					}
					AND = {
						country_exists = c:MGE
						is_subject_of = c:MGE
					}
				}
				OR = {
					any_owned_location = {
						region = region:khorasan_region
					}
					any_owned_location = {
						region = region:xinjiang_region
					}
					any_owned_location = {
						region = region:north_china_region
					}
				}
			}
		}
	}
}

# Phase 3: the four khanates restored. MGE itself is proclaimed at
# Phase 2's close (form_country ignores the formable's allow block,
# the same way PD forms PRU_f in pd_brandenburg_dhe.4), so this
# phase no longer chases MGE_f's formable requirements. Instead it
# demands the historical empire at its height:
#   Yuan          karakorum + dadu           (heartland + China seat)
#   Chagatai      samarkand                  (Transoxiana seat)
#   Golden Horde  sarai_al_jadid + kazan     (Volga seats)
#   Ilkhanate     tabriz + baghdad + ALL of persia_region cleared
#   plus the westward advance made real: a foothold in the Russian
#   lands and one in Cappadocia (Kose Dag country — the deepest the
#   historical Mongols ever rode into Anatolia).
# Geography verified against definitions.txt: tabriz is
# azerbaijan_area/persia_region:2315, baghdad is iraq_arabi_area/
# crescent_region:2136, cappadocia_area is anatolia_region:2054.
mr_dominance_end_trigger = {
	custom_tooltip = {
		text = mr_dominance_end_tt
		any_country = {
			OR = {
				tag = MGE
				tag = MGO
			}
			owns = location:karakorum
			owns = location:dadu
			owns = location:samarkand
			owns = location:sarai_al_jadid
			owns = location:kazan
			owns = location:tabriz
			owns = location:baghdad
			any_owned_location = {
				region = region:russian_region
			}
			has_presence_in = area:cappadocia_area
		}
		# The Ilkhanate clause: no power outside the claimant's realm
		# holds a single location in the Persia region. c:TAG links sit
		# behind country_exists guards in the same AND, as everywhere.
		NOT = {
			any_country = {
				NOR = {
					tag = MGE
					tag = MGO
					AND = {
						country_exists = c:MGE
						is_subject_of = c:MGE
					}
					AND = {
						country_exists = c:MGO
						is_subject_of = c:MGO
					}
				}
				any_owned_location = {
					region = region:persia_region
				}
			}
		}
	}
}

# =================================================================
# LATE-STEPPE SITUATIONS (1600s)
# =================================================================
# Three moments vanilla has no content for at all — verified absent:
# kalmyk/torghut, dzungar/zunghar and chahar all return zero files
# across events, common and setup. They run in the Phase 3 era and
# are built on tags that do exist (OIR, MGO, MGE), since none of
# the historical successor states have their own tag.
# --- Ligdan Khan and the Chahar (1604-1634) ----------------------
# The last Chinggisid Great Khan's failed reunification: the dark
# mirror of Phase 1. Only meaningful while a Chinggisid claimant
# actually exists to be challenged.
mr_can_start_chahar = {
	NOT = {
		has_game_rule = mr_railroad_off
	}
	current_year >= 1604
	current_year < 1634
	NOT = {
		has_global_variable = mr_chahar_started
	}
	# Ligdan's reunification bid needs a FRAGMENTED steppe. It fires
	# whether or not the railroad failed — that failure is exactly the
	# world where the last Chinggisid tries and fails — but not when a
	# claimant already dominates the heartland.
	NOT = {
		any_country = {
			OR = {
				tag = MGO
				tag = MGE
			}
			any_owned_location = {
				region = region:mongolia_region
			}
		}
	}
	any_country = {
		culture = {
			has_culture_group = culture_group:mongolian_group
		}
		government_type = government_type:steppe_horde
		any_owned_location = {
			region = region:mongolia_region
		}
	}
}

mr_chahar_visible = {
	OR = {
		tag = MGO
		tag = MGE
		any_owned_location = {
			region = region:mongolia_region
		}
		any_owned_location = {
			region = region:north_china_region
		}
	}
}

# Ends when one power holds the Mongolian heartland outright.
mr_chahar_end_trigger = {
	any_country = {
		culture = {
			has_culture_group = culture_group:mongolian_group
		}
		owns = location:karakorum
		any_owned_location = {
			region = region:north_china_region
		}
	}
}

# --- The Torghut Migration (1616-1630) ---------------------------
# The Kalmyk trek west to the Volga: an actual Mongol westward wave
# into Russia, which is the Phase 3 premise in miniature.
mr_can_start_torghut = {
	NOT = {
		has_game_rule = mr_railroad_off
	}
	current_year >= 1616
	current_year < 1630
	NOT = {
		has_global_variable = mr_torghut_started
	}
	# Departs from Dzungaria (xinjiang_region), heads for the Volga.
	# If a Mongol horde already sits on the Volga, the migration story
	# is moot — the destination is taken.
	NOT = {
		any_country = {
			culture = {
				has_culture_group = culture_group:mongolian_group
			}
			government_type = government_type:steppe_horde
			owns = location:sarai_al_jadid
		}
	}
	any_country = {
		culture = {
			has_culture_group = culture_group:mongolian_group
		}
		government_type = government_type:steppe_horde
		OR = {
			any_owned_location = {
				region = region:xinjiang_region
			}
			any_owned_location = {
				region = region:khorasan_region
			}
		}
	}
}

mr_torghut_visible = {
	OR = {
		tag = OIR
		tag = MGO
		tag = MGE
		any_owned_location = {
			region = region:steppes_region
		}
		any_owned_location = {
			region = region:russian_region
		}
		any_owned_location = {
			region = region:ural_region
		}
	}
}

# Ends once a steppe power actually reaches the Volga.
mr_torghut_end_trigger = {
	# The trek must actually have happened: the Urals beat gates the
	# end, so a horde that already owned the Volga cannot end this the
	# month it starts.
	has_global_variable = mr_torghut_urals
	any_country = {
		culture = {
			has_culture_group = culture_group:mongolian_group
		}
		government_type = government_type:steppe_horde
		owns = location:sarai_al_jadid
	}
}

# =================================================================
# READ-ONLY VANILLA HOOKS
# =================================================================
# We never write to vanilla state — these only read it, so there is
# no compatibility risk and no file is overwritten.
#
# Tumu: flavor_chi_mon sets lost_emperor on the Middle Kingdom's
# leader country while the emperor is held. It persists until
# clear_tumu_crisis_variables runs, so it is safe to read.
#
# NOTE: no equivalent hook exists for the Treaty of Ugra. Vanilla's
# treaty_of_ugra / treaty_of_ugra_declined variables are removed as
# soon as the event resolves (flavor_MOS.txt:2473, :2541), so there
# is nothing durable left to read. Do not add one without first
# confirming a persistent signal exists.
mr_vanilla_tumu_crisis_active = {
	any_country = {
		has_variable = lost_emperor
	}
}

### Army Balance For AI support: keep the horde in the saddle
MR_percent_of_army_balance = {
	OR = {
		AND = {
			expected_army_size <= 12
			regular_army_size < {
				value = root.expected_army_size
				multiply = 0.9
			}
		}
		AND = {
			expected_army_size > 12
			expected_army_size <= 30
			regular_army_size < {
				value = root.expected_army_size
				multiply = 0.75
			}
		}
		AND = {
			expected_army_size > 30
			expected_army_size <= 50
			regular_army_size < {
				value = root.expected_army_size
				multiply = 0.6
			}
		}
		AND = {
			expected_army_size > 50
			expected_army_size <= 120
			regular_army_size < {
				value = root.expected_army_size
				multiply = 0.45
			}
		}
		AND = {
			expected_army_size > 120
			regular_army_size < {
				value = root.expected_army_size
				multiply = 0.3
			}
		}
	}
}
