# =================================================================
# MONGOL RESURGENCE - ON ACTIONS
# =================================================================
# The situations own their own lifecycle now (can_start opens them,
# on_monthly drives the beats and the AI railroad, can_end closes
# them), so this file no longer hand-fires phase events — that
# duplication is what produced the dangling trigger_event calls in
# the original version.
#
# What it does instead is the job PD uses its on_action for:
# sustaining the AI claimant. This matters because a situation's
# on_monthly only ticks while that situation is ACTIVE. Between
# phases — after Phase 1 closes in 1420 and before Phase 2's
# can_start passes — nothing else is supporting the AI at all.
#
# Corrections carried over from the original version:
#   on_yearly_pulse   -> yearly_country_pulse   (the real hook)
#   on_country_formed -> does not exist; dropped
#   trigger_event     -> trigger_event_silently / _non_silently
# =================================================================

yearly_country_pulse = {
	on_actions = {
		mr_claimant_manpower_pulse
		mr_claimant_army_pulse
	}
}


# --- Keep the horde's military base growing ----------------------
# Mirrors PD's on_brandenburg_manpower_pulse: the AI claimant builds
# the military infrastructure it needs, which it otherwise neglects.
mr_claimant_manpower_pulse = {
	trigger = {
		NOT = { has_game_rule = mr_railroad_off }
		is_ai = yes
		OR = {
			this = c:MGO
			this = c:MGE
		}
	}
	events = {
		mr_dominance.995
	}
}


# --- Keep the horde's armies raised, in peacetime ----------------
# Mirrors PD's on_brandenburg_construct_army_pulse. Peace-only, so
# it tops the horde up between wars rather than during them.
mr_claimant_army_pulse = {
	trigger = {
		NOT = { has_game_rule = mr_railroad_off }
		is_ai = yes
		at_war = no
		monthly_manpower >= 0.1
		OR = {
			this = c:MGO
			this = c:MGE
		}
	}
	events = {
		mr_dominance.996
	}
}
