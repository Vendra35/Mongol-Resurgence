# =================================================================
# THE GREAT PARTITION (opens 1600-1720, gated on Phase 3 ending)
# =================================================================
# The endgame of the railroad, and its mirror. Phases 1-3 gathered
# the four khanates; this one runs the same seven seats in reverse.
#
# The empire is never conquered from outside — that is not how any
# Mongol partition ever happened. Moengke died in 1259 and the Toluid
# civil war split the realm into uluses; the Chinggisid world came
# apart at the succession, every time. So the pressure here is
# internal and the outside powers are vultures, not conquerors.
#
# HISTORICAL CONVERGENCE. The successors are deliberately the
# polities that ACTUALLY EXISTED on the steppe around 1650-1700 —
# Kazakh, Oirat, Crimea, Nogai, Bashkir, Chagatai in the Tarim — and
# NOT the 1337 khanates. A Golden Horde reappearing in 1650 would be
# a second piece of alternate history; the point of this situation is
# the opposite, the alternate timeline closing back onto the real
# map. By 1710 the Kazakhs are on the Kazakh steppe, the Dzungars in
# Dzungaria, Crimea is an Ottoman vassal, and Russia and the Manchu
# are eating the edges — which is what happened.
#
# Successor ground is named ONCE, in scripted_geography (MR_geo_*),
# and asked about ONCE, through the mr_ulus_*_held triggers. Both
# the cohesion score and on_ending's outcome branch read the same
# triggers, the way mr_p2_*_cleared is read by both the end trigger
# and the panel. Change an ulus boundary in the atom and every
# consumer follows.
#
# WINDOW. can_start is gated on the mr_partition_ready flag, not on
# a date — see the note on mr_can_start_partition. Phase 3 can finish
# any time between 1550 and 1650, and a fixed start would leave a
# fast campaign idling. The 1600-1720 dates below are the outer
# bound only.
# =================================================================
mr_great_partition = {
	monthly_spawn_chance = monthly_spawn_chance_unique
	can_start = {
		mr_can_start_partition = yes
	}
	visible = {
		mr_partition_visible = yes
	}
	can_end = {
		OR = {
			mr_partition_end_trigger = yes
			# Time expiry, same reason as every other phase: without it
			# the situation can only ever end in collapse, which leaves
			# on_ending's "the ulus holds" branch unreachable.
			custom_tooltip = {
				text = mr_partition_timeout_tt
				current_date > 1720.1.1
			}
		}
	}
	on_start = {
		set_global_variable = mr_partition_started
		set_variable = {
			name = mr_partition_momentum
			value = 0
		}
		# Cohesion is a 0-100 score like MR_dominance_score, but it
		# counts DOWN: 100 is an empire that still holds every ulus.
		# Seeded full so the opening months read honestly before the
		# first monthly recompute lands.
		set_variable = {
			name = MR_cohesion_score
			value = 100
		}
		# What the Khaghan has bought back at the kurultai. Cohesion is
		# recomputed from scratch every month, so a decision that simply
		# added to the score would be wiped on the next tick; concessions
		# are a standing term the recompute adds back in.
		set_variable = {
			name = mr_partition_concessions
			value = 0
		}
		set_variable = {
			name = mr_partition_kurultai_cooldown
			value = 0
		}
		# Panel header. The Khaghan is normally MGE; MGO is the
		# failed-proclamation fallback that Phase 3 also carries.
		# Guarded, because c:TAG on the right of a comparison errors
		# every tick while the tag is off the map.
		if = {
			limit = {
				country_exists = c:MGE
			}
			set_variable = {
				name = mr_leading_country
				value = c:MGE
			}
		}
		else_if = {
			limit = {
				country_exists = c:MGO
			}
			set_variable = {
				name = mr_leading_country
				value = c:MGO
			}
		}
	}
	on_monthly = {
		# --- THE CLOCK ---
		# Months since the situation opened. This counter existed from the
		# start and was read by NOTHING; it is now what drives the schedule
		# below, which is the job it was always shaped for. Incremented
		# FIRST, so a theatre that comes due this month secedes this month
		# and not next.
		change_variable = {
			name = mr_partition_momentum
			add = 1
		}
		# =============================================================
		# THE SCHEDULE - HISTORY RESUMES
		# =============================================================
		# MEASURED IN GAME 2026-07-30, and the reason this block exists:
		# with cohesion as the only driver the endgame is a 120-year no-op.
		# Cohesion falls only when the Khaghan actually LOSES ground, and
		# nothing in this mod makes a restored empire lose any - the phase
		# rewards are permanent buffs and the Phase 3 failsafe hands it
		# whatever it missed. Worse, it was circular: the one lever pointed
		# AT the Mongols, cb_MR_carve_the_ulus, is granted at cohesion 55,
		# which you can only reach by having already lost five uluses. The
		# tool for breaking the empire was handed out after the empire had
		# broken.
		#
		# So the clock drives the secessions and the secessions drive
		# cohesion, which fires the beats and finally grants the CB. From a
		# full empire that ladder runs:
		#   100 -> 92 (+4y) -> 80 (+8y, beat 85) -> 72 -> 62
		#       -> 54 (+28y, beat 55 and the carve CB) -> 42
		#       -> 30 (+56y, beat 40 and the end trigger)
		#
		# ORDER IS HISTORY, PACE IS THE CAMPAIGN. On the user's ruling of
		# "year OR elapsed time, whichever comes first": the year clause
		# CANNOT be the real date. The situation opens at 1600 at the
		# earliest and almost every real date - Crimea 1441, Kazakh 1465,
		# Safavid 1501, Kazan 1552, Nogai 1557, Joseon 1392 - is already
		# in the past, so every theatre would fire in month one and the
		# whole arc would collapse into a single tick. The ORDER therefore
		# carries the history, the elapsed-months clause sets the pace, and
		# the year clause is a BACKSTOP ladder (1650 -> 1702, four years
		# apart) that stops a late-opening campaign running past 1720. The
		# real dates live in the comments and in the heirs themselves.
		#
		# Each theatre is one scripted effect in
		# scripted_effects/MR_partition_effects.txt, written once and also
		# called by on_ending's final sweep. Each checks its own global
		# flag, so nothing fires twice and a reconquest is not snapped back
		# out of the Khaghan's hands every month.
		if = {
			limit = {
				NOT = {
					has_game_rule = MR_partition_schedule_no
				}
			}
			# --- 1. THE CRIMEAN KHANATE (1441) - Haci Giray's ulus, furthest
			# from Karakorum and the Chinggisid state that outlived every
			# other one, to 1783.
			if = {
				limit = {
					OR = {
						var:mr_partition_momentum > 48
						current_date > 1650.1.1
					}
				}
				mr_return_crimea = yes
			}
			# --- 2. THE KAZAKH KHANATE (1465) - Kerey and Janibek break away
			# onto the Desht-i Kipchak.
			if = {
				limit = {
					OR = {
						var:mr_partition_momentum > 96
						current_date > 1654.1.1
					}
				}
				mr_return_kazakh = yes
			}
			# --- 3. SAFAVID PERSIA (1501) - Ismail I takes Tabriz. The Caucasus
			# comes with it, Safavid ground to 1722.
			if = {
				limit = {
					OR = {
						var:mr_partition_momentum > 144
						current_date > 1658.1.1
					}
				}
				mr_return_persia = yes
			}
			# --- 4. THE BASHKIRS (1662) - under Moscow from the 1550s and in
			# revolt for a century after. BEFORE both Russian sweeps:
			# MR_geo_ural contains bashkiria_area.
			if = {
				limit = {
					OR = {
						var:mr_partition_momentum > 192
						current_date > 1662.1.1
					}
				}
				mr_return_bashkiria = yes
			}
			# --- 5. THE MIDDLE VOLGA (1552, 1556) - Kazan and Astrakhan fall to
			# Ivan IV. Also before the northern sweep, same reason.
			if = {
				limit = {
					OR = {
						var:mr_partition_momentum > 240
						current_date > 1666.1.1
					}
				}
				mr_return_middle_volga = yes
			}
			# --- 6. THE NOGAI HORDE (1557) - the Greater/Lesser split, then
			# Kalmyk pressure off the Volga in the 1630s.
			if = {
				limit = {
					OR = {
						var:mr_partition_momentum > 288
						current_date > 1670.1.1
					}
				}
				mr_return_nogai = yes
			}
			# --- 7. RUSSIA IN THE NORTH (1582-1598) - Yermak crosses the Urals,
			# the Sibir Khanate is finished by 1598.
			if = {
				limit = {
					OR = {
						var:mr_partition_momentum > 336
						current_date > 1674.1.1
					}
				}
				mr_return_north = yes
			}
			# --- 8. THE OTTOMANS (1638) - Murad IV takes Baghdad back; Anatolia
			# was never anything else after 1453.
			if = {
				limit = {
					OR = {
						var:mr_partition_momentum > 384
						current_date > 1678.1.1
					}
				}
				mr_return_ottoman = yes
			}
			# --- 9. THE MANCHUS AT HOME (1616) - Nurhaci proclaims the Later
			# Jin. Their homeland first, China later, as it happened.
			if = {
				limit = {
					OR = {
						var:mr_partition_momentum > 432
						current_date > 1682.1.1
					}
				}
				mr_return_manchuria = yes
			}
			# --- 10. THE DZUNGAR KHANATE (1634) - Erdeni Batur founds the last
			# steppe empire.
			if = {
				limit = {
					OR = {
						var:mr_partition_momentum > 480
						current_date > 1686.1.1
					}
				}
				mr_return_dzungaria = yes
			}
			# --- 11. THE GANDEN PHODRANG (1642) - the Fifth Dalai Lama's state,
			# installed by Gushi Khan, himself a Mongol.
			if = {
				limit = {
					OR = {
						var:mr_partition_momentum > 528
						current_date > 1690.1.1
					}
				}
				mr_return_tibet = yes
			}
			# --- 12. THE QING TAKE CHINA (1644) - Beijing falls to the Manchus
			# and the Ming are finished.
			if = {
				limit = {
					OR = {
						var:mr_partition_momentum > 576
						current_date > 1694.1.1
					}
				}
				mr_return_china = yes
			}
			# --- 13. KOREA (1392) - Joseon. Goryeo submitted in 1259 and was
			# never annexed, so here Korea usually leaves as a subject
			# released rather than as ground lost.
			if = {
				limit = {
					OR = {
						var:mr_partition_momentum > 624
						current_date > 1698.1.1
					}
				}
				mr_return_korea = yes
			}
			# --- 14. CHAGATAI IN THE TARIM (1680) - Yarkent stayed Chagatayid to
			# 1705; Galdan took Kashgar in 1680. Last, because it is the
			# nearest thing to the heartland that still becomes somebody
			# else.
			if = {
				limit = {
					OR = {
						var:mr_partition_momentum > 672
						current_date > 1702.1.1
					}
				}
				mr_return_tarim = yes
			}
		}
		# --- COHESION ---
		# Computed AFTER the schedule and before the beats, so a theatre
		# that seceded this month is already inside the number the beats
		# react to. Everything
		# that can change what the panel should show runs inside the
		# same tick; computed first, the panel lags a month behind its
		# own situation. All three earlier phases were fixed for
		# exactly this, and the bug surfaced as a progress bar reading
		# zero while every requirement was already green.
		#
		# Drivers are things an AI Khaghan actually experiences —
		# ground lost, ulus by ulus. A driver only a human would trip
		# would leave the situation frozen in an all-AI game, which is
		# how this mod is usually watched.
		set_variable = {
			name = MR_cohesion_score
			value = 100
		}
		if = {
			limit = {
				NOT = {
					mr_ulus_heartland_held = yes
				}
			}
			change_variable = {
				name = MR_cohesion_score
				subtract = 30
			}
		}
		if = {
			limit = {
				NOT = {
					mr_ulus_kazakh_held = yes
				}
			}
			change_variable = {
				name = MR_cohesion_score
				subtract = 12
			}
		}
		if = {
			limit = {
				NOT = {
					mr_ulus_dzungaria_held = yes
				}
			}
			change_variable = {
				name = MR_cohesion_score
				subtract = 12
			}
		}
		if = {
			limit = {
				NOT = {
					mr_ulus_tarim_held = yes
				}
			}
			change_variable = {
				name = MR_cohesion_score
				subtract = 12
			}
		}
		if = {
			limit = {
				NOT = {
					mr_ulus_nogai_held = yes
				}
			}
			change_variable = {
				name = MR_cohesion_score
				subtract = 10
			}
		}
		if = {
			limit = {
				NOT = {
					mr_ulus_crimea_held = yes
				}
			}
			change_variable = {
				name = MR_cohesion_score
				subtract = 8
			}
		}
		if = {
			limit = {
				NOT = {
					mr_ulus_bashkiria_held = yes
				}
			}
			change_variable = {
				name = MR_cohesion_score
				subtract = 8
			}
		}
		# The Siberian frontier yields no successor khanate — the Sibir
		# Khanate fell in 1598 and that ground is Russian by now — but
		# losing it is still pressure.
		if = {
			limit = {
				NOT = {
					mr_ulus_sibir_held = yes
				}
			}
			change_variable = {
				name = MR_cohesion_score
				subtract = 8
			}
		}
		# Everything the kurultai has conceded is added back. This is why
		# concessions is a variable and not a one-off change to the score:
		# the recompute above starts from 100 each month and would erase
		# any direct adjustment.
		change_variable = {
			name = MR_cohesion_score
			add = var:mr_partition_concessions
		}
		# Floor the score so the panel bar cannot render negative, and cap
		# it so heavy concessions cannot overfill the bar past its max.
		if = {
			limit = {
				var:MR_cohesion_score < 0
			}
			set_variable = {
				name = MR_cohesion_score
				value = 0
			}
		}
		if = {
			limit = {
				var:MR_cohesion_score > 100
			}
			set_variable = {
				name = MR_cohesion_score
				value = 100
			}
		}
		# --- THRESHOLD BEATS ---
		# Fired as cohesion falls PAST each mark, once each, via a global
		# flag — the same shape the earlier phases use for mr_beat_*_fired.
		# Placed after the score is recomputed and before the panel header,
		# so a beat and the number it is reacting to always agree.
		#
		# Three audiences, mirroring the mr_dominance.22 / .23 split that
		# the earlier phases already send at their own openings:
		#   the KHAGHAN gets the strain from the inside
		#   NEIGHBOURS get the opportunity — they border the dying empire
		#   SPECTATORS get atmosphere; they have a stake in the theatre but
		#     no shared border, so for them this is news, not a chance
		# Neighbour and spectator sets are disjoint by construction, so no
		# country ever receives two versions of the same beat.
		if = {
			limit = {
				var:MR_cohesion_score < 85
				NOT = {
					has_global_variable = mr_partition_beat_85_fired
				}
			}
			every_country = {
				limit = {
					OR = {
						tag = MGE
						tag = MGO
					}
				}
				trigger_event_non_silently = mr_partition.1
			}
			every_country = {
				limit = {
					mr_partition_is_neighbour = yes
				}
				trigger_event_non_silently = mr_partition.11
			}
			set_global_variable = mr_partition_beat_85_fired
		}
		if = {
			limit = {
				var:MR_cohesion_score < 55
				NOT = {
					has_global_variable = mr_partition_beat_55_fired
				}
			}
			every_country = {
				limit = {
					OR = {
						tag = MGE
						tag = MGO
					}
				}
				trigger_event_non_silently = mr_partition.2
			}
			# The vultures gather. Every neighbour gets the carve CB at
			# this mark and keeps it for the rest of the window — this is
			# the lever that lets a player who never touched the steppe
			# take part in the endgame, and the reason an all-AI game
			# still ends with Russia and the Manchu on Mongol ground.
			# Granted against whichever of MGE/MGO is the Khaghan; the
			# country_exists guard sits in the same AND, per house rule.
			every_country = {
				limit = {
					mr_partition_is_neighbour = yes
				}
				trigger_event_non_silently = mr_partition.12
				if = {
					limit = {
						country_exists = c:MGE
					}
					add_casus_belli = {
						target = c:MGE
						type = casus_belli:cb_MR_carve_the_ulus
						years = 80
					}
				}
				if = {
					limit = {
						country_exists = c:MGO
					}
					add_casus_belli = {
						target = c:MGO
						type = casus_belli:cb_MR_carve_the_ulus
						years = 80
					}
				}
			}
			every_country = {
				limit = {
					mr_partition_is_spectator = yes
				}
				trigger_event_non_silently = mr_partition.22
			}
			set_global_variable = mr_partition_beat_55_fired
		}
		if = {
			limit = {
				var:MR_cohesion_score < 40
				NOT = {
					has_global_variable = mr_partition_beat_40_fired
				}
			}
			every_country = {
				limit = {
					OR = {
						tag = MGE
						tag = MGO
					}
				}
				trigger_event_non_silently = mr_partition.3
			}
			every_country = {
				limit = {
					mr_partition_is_neighbour = yes
				}
				trigger_event_non_silently = mr_partition.13
			}
			every_country = {
				limit = {
					mr_partition_is_spectator = yes
				}
				trigger_event_non_silently = mr_partition.23
			}
			set_global_variable = mr_partition_beat_40_fired
		}
		# --- THE KURULTAI ---
		# Once the first crack has shown, the assembly puts the question
		# to the Khaghan every five years or so. No option is free: every
		# answer costs land, order or treasure, which is the whole point —
		# the historical Khaghans had exactly these three moves and none
		# of them was good. Fired to the claimant, human or AI alike, with
		# ai_chance on every option so an observed game still plays it.
		change_variable = {
			name = mr_partition_kurultai_cooldown
			add = 1
		}
		if = {
			limit = {
				var:mr_partition_kurultai_cooldown > 60
				var:MR_cohesion_score < 85
			}
			every_country = {
				limit = {
					OR = {
						tag = MGE
						tag = MGO
					}
				}
				trigger_event_non_silently = mr_partition.100
			}
			set_variable = {
				name = mr_partition_kurultai_cooldown
				value = 0
			}
		}
		# Panel header last, for the same reason as the score.
		if = {
			limit = {
				country_exists = c:MGE
			}
			set_variable = {
				name = mr_leading_country
				value = c:MGE
			}
		}
		else_if = {
			limit = {
				country_exists = c:MGO
			}
			set_variable = {
				name = mr_leading_country
				value = c:MGO
			}
		}
	}
	on_ending = {
		# Branch on the goal trigger, never on a side-signal - the same rule
		# every other phase follows.
		#
		# TWO ways into the collapse branch, and the second is new
		# (2026-07-30). First, the end trigger: the tag is gone, Karakorum
		# has left the realm, or cohesion has run out. Second, the 1720
		# timeout reached by an empire still nominally alive but already
		# hollowed out - without that clause a Khaghan sitting on Karakorum
		# with a cohesion of 45 would time out into the SURVIVAL branch and
		# collect a permanent reward for an empire that came apart anyway.
		if = {
			limit = {
				OR = {
					mr_partition_end_trigger = yes
					AND = {
						has_variable = MR_cohesion_score
						var:MR_cohesion_score < 70
					}
				}
			}
			set_global_variable = mr_partition_collapsed
			# THE FINAL SWEEP. Whatever the schedule has not already handed
			# back goes home now, all at once.
			#
			# Deliberately NOT "only ground the Khaghan already lost". No
			# Mongol partition ever worked that way: Moengke died in 1259
			# with the empire completely intact and it split anyway, at the
			# succession, along the ulus lines the conquest itself drew.
			# Berke, Hulagu, Kublai and Ariq Boeke each held their ulus
			# firmly and separated regardless. Modelling the split as a
			# consequence of territorial loss would invert the real
			# causation - and it would mean a well-played AI Khaghan
			# produces no successors at all, leaving the map a blob and
			# breaking the one thing this situation is for.
			#
			# Every effect checks its own flag, so theatres that already
			# seceded on schedule do nothing here. ORDER STILL MATTERS for
			# the three that overlap: Bashkiria and the middle Volga before
			# the northern sweep, because MR_geo_ural contains both.
			mr_return_crimea = yes
			mr_return_kazakh = yes
			mr_return_persia = yes
			mr_return_bashkiria = yes
			mr_return_middle_volga = yes
			mr_return_nogai = yes
			mr_return_north = yes
			mr_return_ottoman = yes
			mr_return_manchuria = yes
			mr_return_dzungaria = yes
			mr_return_tibet = yes
			mr_return_china = yes
			mr_return_korea = yes
			mr_return_tarim = yes
		}
		else = {
			set_global_variable = mr_partition_survived
			# The window closed with the Khaghan on his throne, Karakorum his
			# and cohesion still above 70 - the uluses never went. The reward
			# is PERMANENT, like every other phase reward here, and it is not
			# a cavalry buff: an empire that survives this century survives
			# by ceasing to be a horde in the old sense. Granted to whichever
			# of MGE/MGO held on.
			every_country = {
				limit = {
					OR = {
						tag = MGE
						tag = MGO
					}
				}
				add_country_modifier = {
					modifier = MR_the_ulus_endures
					years = -1
					mode = replace
				}
				trigger_event_non_silently = mr_partition.200
			}
		}
	}
	on_ended = {
		# The wiki recommends clearing situation variables in on_ended
		# to keep them out of the savefile.
		remove_variable = mr_leading_country
		remove_variable = MR_cohesion_score
		remove_variable = mr_partition_momentum
