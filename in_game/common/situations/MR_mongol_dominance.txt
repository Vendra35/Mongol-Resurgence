# =================================================================
# PHASE 3 - THE FOUR KHANATES (1550-1650)
# =================================================================
# The westward finale. The Yeke Mongol Ulus (MGE) is proclaimed at
# Phase 2's close now, so this phase opens WITH the empire on the
# map and asks it to restore the old ulus in full: the Yuan seats
# (Karakorum, Dadu), the Chagatai seat (Samarkand), the Golden
# Horde seats (Sarai, Kazan), and the whole Ilkhanate — Tabriz,
# Baghdad, Persia cleared of every outside power — plus a foothold
# in the Russian lands and one in Cappadocia (Kose Dag country).
#
# NOTE ON c:MGE: MGE exists from on_start on the normal path, which
# retires the error-spam problem this file used to fight — but the
# MGO fallback (a failed proclamation) is still tolerated, so
# identity checks keep using tag = (string comparison, never
# errors) and every c:MGE / c:MGO link still sits behind a
# country_exists guard in the same AND.
#
# AI railroad: same machine as Phases 1-2 (PD Ascension pattern).
# The claimant is scoped dynamically (random_country over tag MGE /
# MGO); wars are declared by mr_dominance.992, which picks the
# westward or silk-road CB by where the target's land lies. The AI
# claimant carries MR_mongol_preparing_for_conquest all phase.
# =================================================================
mongol_dominance = {
	monthly_spawn_chance = monthly_spawn_chance_unique
	can_start = {
		mr_can_start_dominance = yes
	}
	visible = {
		mr_dominance_visible = yes
	}
	can_end = {
		OR = {
			mr_dominance_end_trigger = yes
			# Time expiry. Without this the situation can only ever end in
			# success, which leaves on_ending's failure branch unreachable
			# and lets a failed phase run to the end of the campaign.
			custom_tooltip = {
				text = mr_dominance_timeout_tt
				current_date > 1650.1.1
			}
		}
	}
	hint_tag = hint_mongol_dominance
	on_start = {
		set_global_variable = mr_dominance_started
		set_variable = {
			name = mr_dominance_momentum
			value = 0
		}
		# --- AI RAILROAD BOOKKEEPING (PD ascension shape) ---
		# The claimant is normally MGE now (proclaimed at Phase 2's
		# close), with MGO as the failed-proclamation fallback —
		# can_start guarantees one of the two exists. The seed target
		# is the claimant itself, PD's "no target" sentinel; the
		# event's after block re-seeds with root.
		set_variable = {
			name = mr_dom_conquest_cooldown
			value = 48
		}
		set_variable = {
			name = mr_dom_conquest_target
			value = 0
		}
		if = {
			limit = {
				country_exists = c:MGE
			}
			set_variable = {
				name = mr_dom_conquest_target_country
				value = c:MGE
			}
			# Values the .gui panel reads off the situation scope.
			set_variable = {
				name = mr_leading_country
				value = c:MGE
			}
		}
		else_if = {
			limit = {
				country_exists = c:MGO
			}
			set_variable = {
				name = mr_dom_conquest_target_country
				value = c:MGO
			}
			set_variable = {
				name = mr_leading_country
				value = c:MGO
			}
		}
		random_country = {
			limit = {
				OR = {
					any_owned_location = {
						region = region:russian_region
					}
					any_owned_location = {
						region = region:steppes_region
					}
					any_owned_location = {
						region = region:ural_region
					}
				}
				NOR = {
					tag = MGO
					tag = MGE
				}
			}
			save_scope_as = mr_theatre_rival
		}
		if = {
			limit = {
				country_exists = scope:mr_theatre_rival
			}
			set_variable = {
				name = mr_rival_country
				value = scope:mr_theatre_rival
			}
		}
		every_country = {
			limit = {
				OR = {
					tag = MGO
					tag = MGE
				}
			}
			# The Phase 2 reward expires here, as its tooltip promises
			# ("removed upon the start of the 3rd Situation").
			remove_country_modifier = MR_empire_fulfilled			## !!!!!!!!!!!!!Might be problem because it should ne permanent modifier no? LOOK LATER!!!!!!!!!!!!!!!
			# The war-preparation lock, PD Ascension style: the AI
			# claimant must not freelance-declare — the railroad event
			# declares for it. Never applied to a human player.
			if = {
				limit = {
					is_ai = yes
				}
				add_country_modifier = {
					modifier = MR_mongol_preparing_for_conquest
					years = -1
					mode = replace
				}
			}
			trigger_event_non_silently = mr_dominance.130
			# Phase buff per the buff rule, PD-style: Terminator gets
			# the Pax Mongolica engine, Historical the balanced
			# variant, Vanilla nothing.
			if = {
				limit = {
					has_game_rule = MR_buff_enabled
				}
				add_country_modifier = {
					modifier = MR_pax_mongolica
					years = -1
					mode = replace
				}
			}
			else_if = {
				limit = {
					has_game_rule = MR_buff_historical
				}
				add_country_modifier = {
					modifier = MR_dominance_historical_modifier
					years = -1
					mode = replace
				}
			}
		}
		# The claimant gets the westward CB against every power holding
		# the western goal theatres: the Russian lands, the Pontic and
		# Ural steppes, Persia, Mesopotamia and Anatolia — the Golden
		# Horde and Ilkhanate domains this phase demands.
		every_country = {
			limit = {
				OR = {
					any_owned_location = {
						region = region:russian_region
					}
					any_owned_location = {
						region = region:steppes_region
					}
					any_owned_location = {
						region = region:ural_region
					}
					any_owned_location = {
						region = region:persia_region
					}
					any_owned_location = {
						region = region:anatolia_region
					}
					has_presence_in = area:iraq_arabi_area
					has_presence_in = area:armenian_highlands_area
				}
				NOR = {
					tag = MGO
					tag = MGE
				}
			}
			save_scope_as = mr_westward_target
			every_country = {
				limit = {
					OR = {
						tag = MGO
						tag = MGE
					}
				}
				add_casus_belli = {
					target = scope:mr_westward_target
					type = casus_belli:cb_MR_westward_advance
					# Covers the whole 1550-1650 phase window.
					years = 100
				}
			}
		}
		# The eastern seats (Karakorum, Dadu, Samarkand) stay covered by
		# re-granting the Silk Road CB — the Phase 2 grant expired
		# decades before this phase begins, and a seat lost after Phase
		# 2 must stay legally retakable.
		every_country = {
			limit = {
				OR = {
					any_owned_location = {
						region = region:north_china_region
					}
					any_owned_location = {
						region = region:mongolia_region
					}
					any_owned_location = {
						region = region:xinjiang_region
					}
					any_owned_location = {
						region = region:khorasan_region
					}
				}
				NOR = {
					tag = MGO
					tag = MGE
				}
			}
			save_scope_as = mr_seat_target
			every_country = {
				limit = {
					OR = {
						tag = MGO
						tag = MGE
					}
				}
				add_casus_belli = {
					target = scope:mr_seat_target
					type = casus_belli:cb_MR_silk_road
					# Covers the whole 1550-1650 phase window.
					years = 100
				}
			}
		}
	}
	on_monthly = {
		change_variable = {
			name = mr_dominance_momentum
			add = 1
		}
		# Keep the panel header on the current claimant: once MGE forms,
		# it replaces MGO as the leading country.
		if = {
			limit = {
				country_exists = c:MGE
				NOT = {
					has_global_variable = mr_mge_seen
				}
			}
			set_variable = {
				name = mr_leading_country
				value = c:MGE
			}
			set_global_variable = mr_mge_seen
		}
		# ========== PANEL PROGRESS (0-100) ==========
		# The .gui reads MR_dominance_score: ten points per goal met.
		# The ten milestones mirror mr_dominance_end_trigger exactly —
		# seven khanate seats, Persia cleared, the Russian foothold and
		# the Cappadocian foothold. 100 = the end trigger is satisfied.
		set_variable = {
			name = MR_dominance_score
			value = 0
		}
		if = {
			limit = {
				any_country = {
					OR = {
						tag = MGO
						tag = MGE
					}
					owns = location:karakorum
				}
			}
			change_variable = {
				name = MR_dominance_score
				add = 10
			}
		}
		if = {
			limit = {
				any_country = {
					OR = {
						tag = MGO
						tag = MGE
					}
					owns = location:dadu
				}
			}
			change_variable = {
				name = MR_dominance_score
				add = 10
			}
		}
		if = {
			limit = {
				any_country = {
					OR = {
						tag = MGO
						tag = MGE
					}
					owns = location:samarkand
				}
			}
			change_variable = {
				name = MR_dominance_score
				add = 10
			}
		}
		if = {
			limit = {
				any_country = {
					OR = {
						tag = MGO
						tag = MGE
					}
					owns = location:sarai_al_jadid
				}
			}
			change_variable = {
				name = MR_dominance_score
				add = 10
			}
		}
		if = {
			limit = {
				any_country = {
					OR = {
						tag = MGO
						tag = MGE
					}
					owns = location:kazan
				}
			}
			change_variable = {
				name = MR_dominance_score
				add = 10
			}
		}
		if = {
			limit = {
				any_country = {
					OR = {
						tag = MGO
						tag = MGE
					}
					owns = location:tabriz
				}
			}
			change_variable = {
				name = MR_dominance_score
				add = 10
			}
		}
		if = {
			limit = {
				any_country = {
					OR = {
						tag = MGO
						tag = MGE
					}
					owns = location:baghdad
				}
			}
			change_variable = {
				name = MR_dominance_score
				add = 10
			}
		}
		# The Ilkhanate clause: Persia cleared of every outside power.
		if = {
			limit = {
				NOT = {
					any_country = {
						NOR = {
							tag = MGE
							tag = MGO
							AND = {
								country_exists = c:MGE
								is_subject_of = c:MGE
							}
							AND = {
								country_exists = c:MGO
								is_subject_of = c:MGO
							}
						}
						any_owned_location = {
							region = region:persia_region
						}
					}
				}
			}
			change_variable = {
				name = MR_dominance_score
				add = 10
			}
		}
		if = {
			limit = {
				any_country = {
					OR = {
						tag = MGO
						tag = MGE
					}
					any_owned_location = {
						region = region:russian_region
					}
				}
			}
			change_variable = {
				name = MR_dominance_score
				add = 10
			}
		}
		if = {
			limit = {
				any_country = {
					OR = {
						tag = MGO
						tag = MGE
					}
					has_presence_in = area:cappadocia_area
				}
			}
			change_variable = {
				name = MR_dominance_score
				add = 10
			}
		}
		# ========== AI RAILROAD: FIND A TARGET ==========
		# The claimant is whichever of MGO/MGE exists (a player can form
		# MGE mid-phase), scoped via random_country. Targets: the
		# weakest neighbour holding land in either theatre — the
		# westward goal regions (russian / steppes / ural) or the seat
		# regions (mongolia / khorasan / xinjiang / the four Chinas).
		if = {
			limit = {
				var:mr_dom_conquest_target = 0
				any_country = {
					OR = {
						tag = MGO
						tag = MGE
					}
					at_war = no
				}
			}
			random_country = {
				limit = {
					OR = {
						tag = MGO
						tag = MGE
					}
					at_war = no
				}
				save_scope_as = mr_dom_claimant
				ordered_neighbor_country = {
					limit = {
						is_rebel_country = no
						NOT = {
							has_truce_with = prev
						}
						NOT = {
							is_subject_of = prev
						}
						defensive_alliance_strength < scope:mr_dom_claimant.offensive_alliance_strength
						OR = {
							any_owned_location = {
								region = region:russian_region
							}
							any_owned_location = {
								region = region:steppes_region
							}
							any_owned_location = {
								region = region:ural_region
							}
							any_owned_location = {
								region = region:persia_region
							}
							any_owned_location = {
								region = region:anatolia_region
							}
							any_owned_location = {
								region = region:mongolia_region
							}
							any_owned_location = {
								region = region:khorasan_region
							}
							any_owned_location = {
								region = region:xinjiang_region
							}
							any_owned_location = {
								region = region:north_china_region
							}
							has_presence_in = area:iraq_arabi_area
							has_presence_in = area:armenian_highlands_area
						}
					}
					max = 1
					order_by = {
						value = num_locations
						multiply = -1
					}
					situation:mongol_dominance = {
						set_variable = {
							name = mr_dom_conquest_target_country
							value = prev
						}
						set_variable = {
							name = mr_dom_conquest_target
							value = 1
						}
					}
				}
			}
		}
		change_variable = {
			name = mr_dom_conquest_cooldown
			add = 1
		}
		# --- FALLBACK: the chosen target stopped being valid ---
		# (dead, became the claimant itself, or lost every holding in
		# both theatres). Subject/truce staleness is re-checked by the
		# declare gate below, PD-style.
		if = {
			limit = {
				var:mr_dom_conquest_target = 1
				OR = {
					NOT = {
						country_exists = var:mr_dom_conquest_target_country
					}
					var:mr_dom_conquest_target_country = {
						OR = {
							tag = MGO
							tag = MGE
							NOT = {
								OR = {
									any_owned_location = {
										region = region:russian_region
									}
									any_owned_location = {
										region = region:steppes_region
									}
									any_owned_location = {
										region = region:ural_region
									}
									any_owned_location = {
										region = region:persia_region
									}
									any_owned_location = {
										region = region:anatolia_region
									}
									any_owned_location = {
										region = region:mongolia_region
									}
									any_owned_location = {
										region = region:khorasan_region
									}
									any_owned_location = {
										region = region:xinjiang_region
									}
									any_owned_location = {
										region = region:north_china_region
									}
									has_presence_in = area:iraq_arabi_area
									has_presence_in = area:armenian_highlands_area
								}
							}
						}
					}
				}
			}
			set_variable = {
				name = mr_dom_conquest_target
				value = 0
			}
		}
		# ========== AI RAILROAD: DECLARE ==========
		# PD Ascension pacing: Terminator roughly yearly, otherwise
		# every ~5 years, with the same months_since_war gates.
		if = {
			limit = {
				var:mr_dom_conquest_target = 1
				OR = {
					AND = {
						has_game_rule = MR_buff_enabled
						var:mr_dom_conquest_cooldown > 12
					}
					AND = {
						NOT = {
							has_game_rule = MR_buff_enabled
						}
						var:mr_dom_conquest_cooldown > 60
					}
				}
				any_country = {
					OR = {
						tag = MGO
						tag = MGE
					}
				}
			}
			random_country = {
				limit = {
					OR = {
						tag = MGO
						tag = MGE
					}
				}
				if = {
					limit = {
						OR = {
							AND = {
								has_game_rule = MR_buff_enabled
								months_since_war >= 12
							}
							AND = {
								NOT = {
									has_game_rule = MR_buff_enabled
								}
								months_since_war >= 24
							}
						}
						situation:mongol_dominance = {
							var:mr_dom_conquest_target_country = {
								NOT = {
									is_subject_of = prev
								}
								NOT = {
									has_truce_with = prev
								}
							}
						}
					}
					trigger_event_silently = mr_dominance.992
				}
			}
		}
		# --- The New World Order (~10 years) ---
		if = {
			limit = {
				var:mr_dominance_momentum >= 120
				NOT = {
					has_global_variable = mr_beat_131_fired
				}
			}
			every_country = {
				limit = {
					OR = {
						tag = MGO
						tag = MGE
					}
				}
				trigger_event_non_silently = mr_dominance.131
			}
			set_global_variable = mr_beat_131_fired
		}
		# --- The Four Corners: all four khanate capitals held ---
		# Not momentum-timed — this one fires the month it becomes
		# TRUE: Karakorum, Dadu, Samarkand and Sarai under one banner.
		if = {
			limit = {
				NOT = {
					has_global_variable = mr_beat_135_fired
				}
				any_country = {
					OR = {
						tag = MGO
						tag = MGE
					}
					owns = location:karakorum
					owns = location:dadu
					owns = location:samarkand
					owns = location:sarai_al_jadid
				}
			}
			every_country = {
				limit = {
					OR = {
						tag = MGO
						tag = MGE
					}
				}
				trigger_event_non_silently = mr_dominance.135
			}
			set_global_variable = mr_beat_135_fired
		}
		# --- The westward campaign against Russia (~18 years) ---
		if = {
			limit = {
				var:mr_dominance_momentum >= 216
				NOT = {
					has_global_variable = mr_beat_201i_fired
				}
				any_country = {
					OR = {
						tag = MGO
						tag = MGE
					}
					any_owned_location = {
						region = region:russian_region
					}
				}
			}
			every_country = {
				limit = {
					OR = {
						tag = MGO
						tag = MGE
					}
				}
				trigger_event_non_silently = mr_imperial.201
			}
			set_global_variable = mr_beat_201i_fired
		}
		# --- The Grand Review (~30 years) ---
		if = {
			limit = {
				var:mr_dominance_momentum >= 360
				NOT = {
					has_global_variable = mr_beat_202i_fired
				}
			}
			every_country = {
				limit = {
					OR = {
						tag = MGO
						tag = MGE
					}
				}
				trigger_event_non_silently = mr_imperial.202
			}
			set_global_variable = mr_beat_202i_fired
		}
		# --- The retreat, if the west holds (~40 years, still short) ---
		if = {
			limit = {
				var:mr_dominance_momentum >= 480
				NOT = {
					has_global_variable = mr_beat_211i_fired
				}
				NOT = {
					mr_dominance_end_trigger = yes
				}
			}
			every_country = {
				limit = {
					OR = {
						tag = MGO
						tag = MGE
					}
				}
				trigger_event_non_silently = mr_imperial.211
			}
			set_global_variable = mr_beat_211i_fired
		}
		# --- COMPLETION FAILSAFE: five years before 1650 ---
		# Grants MGE_f's own nine required locations outright — plus a
		# Russian foothold (Ryazan, historically the first principality
		# the Mongols reached in 1237) so the end trigger's westward
		# clause is genuinely reachable. Gated on the imperial
		# conquest-automation rule and on is_ai; never takes a location
		# from the player.
		if = {
			limit = {
				has_game_rule = MR_imperial_auto_conquest_yes
				current_date > 1645.1.1
				NOT = {
					mr_dominance_end_trigger = yes
				}
				NOT = {
					has_global_variable = mr_failsafe_p3_fired
				}
				any_country = {
					OR = {
						tag = MGO
						tag = MGE
					}
					is_ai = yes
					at_war = no
				}
			}
			set_global_variable = mr_failsafe_p3_fired
			random_country = {
				limit = {
					OR = {
						tag = MGO
						tag = MGE
					}
					is_ai = yes
					at_war = no
				}
				save_scope_as = mr_claimant
			}
			# The areas holding every seat the end trigger demands, the
			# two footholds, and the whole Persia region for the
			# Ilkhanate clause. Iterating areas/regions rather than
			# naming locations one by one keeps this readable and hands
			# over the surrounding land too — the same shape as PD's
			# every_ownable_location_in_area blocks. Tabriz rides in
			# with the persia_region loop (azerbaijan_area is inside
			# persia_region).
			area:upper_selenga_area = {
				every_ownable_location_in_area = {
					limit = {
						owner ?= {
							is_ai = yes
							NOR = {
								this = scope:mr_claimant
								is_subject_of = scope:mr_claimant
							}
						}
					}
					change_location_owner = scope:mr_claimant
					add_core = scope:mr_claimant
				}
			}
			area:lower_don_area = {

	on_ending = {
		# 1. Strip the phase modifiers, PD-style.
		every_country = {
			limit = {
				OR = {
					tag = MGO
					tag = MGE
				}
			}
			remove_country_modifier = MR_pax_mongolica
			remove_country_modifier = MR_dominance_historical_modifier
			remove_country_modifier = MR_mongol_preparing_for_conquest
		}
		# 2. The railroad either completes, or the empire crumbles.
		if = {
			limit = {
				mr_dominance_end_trigger = yes
			}
			# Belt-and-braces: MGE is normally proclaimed at Phase 2's
			# close now. This fallback only fires on the pathological
			# path where the claimant somehow reached the finale still
			# wearing the MGO tag.
			every_country = {
				limit = {
					tag = MGO
					NOT = {
						country_exists = c:MGE
					}
				}
				form_country = formable_country:MGE_f
			}
			every_country = {
				limit = {
					OR = {
						tag = MGO
						tag = MGE
					}
				}
				trigger_event_non_silently = mr_dominance.132
			}
			set_global_variable = mr_railroad_complete
		}
		else = {
			every_country = {
				limit = {
					OR = {
						tag = MGO
						tag = MGE
					}
				}
				trigger_event_non_silently = mr_dominance.133
			}
			set_global_variable = mr_railroad_failed
		}
	}
