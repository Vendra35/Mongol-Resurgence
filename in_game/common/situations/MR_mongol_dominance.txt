# =================================================================
# PHASE 3 - THE SILK EMPIRE (1550-1650)
# =================================================================
# The westward advance against Russia and the final formable. The
# phase closes when one power holds all four seats of the old
# khanates at once: Karakorum, Sarai, Dadu and Baghdad.
# =================================================================

mongol_dominance = {
	monthly_spawn_chance = monthly_spawn_chance_unique

	can_start = {
		mr_can_start_dominance = yes
	}

	visible = {
		mr_dominance_visible = yes
	}

	can_end = {
		OR = {
			mr_dominance_end_trigger = yes
			# Time expiry. Without this the situation can only ever end in
			# success, which leaves on_ending's failure branch unreachable
			# and lets a failed phase run to the end of the campaign.
			custom_tooltip = {
				text = mr_dominance_timeout_tt
				current_date > 1650.1.1
			}
		}
	}

	hint_tag = hint_mongol_dominance

	on_start = {
		set_global_variable = mr_dominance_started

		set_variable = {
			name = mr_dominance_momentum
			value = 0
		}

		# Values the .gui panel reads off the situation scope.
		if = {
			limit = { country_exists = c:MGO }
			set_variable = {
				name = mr_leading_country
				value = c:MGO
			}
		}
		random_country = {
			limit = {
				any_owned_location = { region = region:russian_region }
				NOR = {
					this = c:MGO
					this = c:MGE
				}
			}
			save_scope_as = mr_theatre_rival
		}
		if = {
			limit = { exists = scope:mr_theatre_rival }
			set_variable = {
				name = mr_rival_country
				value = scope:mr_theatre_rival
			}
		}

		every_country = {
			limit = {
				OR = {
					this = c:MGO
					this = c:MGE
				}
			}
			trigger_event_non_silently = mr_dominance.130

			# Phase buff, plus the historical-mode variant when the
			# railroad is running on the historical game rule.
			add_country_modifier = {
				modifier = MR_pax_mongolica
				years = -1
				mode = replace
			}
			if = {
				limit = { has_game_rule = mr_railroad_historical }
				add_country_modifier = {
					modifier = MR_dominance_historical_modifier
					years = -1
					mode = replace
				}
			}
		}

		# The claimant gets a CB against the Russian principalities.
		every_country = {
			limit = {
				any_owned_location = { region = region:russian_region }
				NOR = {
					this = c:MGO
					this = c:MGE
				}
			}
			save_scope_as = mr_westward_target
			every_country = {
				limit = {
					OR = {
						this = c:MGO
						this = c:MGE
					}
				}
				add_casus_belli = {
					target = scope:mr_westward_target
					type = casus_belli:cb_MR_westward_advance
					years = 50
				}
			}
		}
	}

	on_monthly = {
		change_variable = {
			name = mr_dominance_momentum
			add = 1
		}

		# --- The New World Order (~10 years) ---
		if = {
			limit = {
				var:mr_dominance_momentum >= 120
				NOT = { has_global_variable = mr_beat_131_fired }
			}
			every_country = {
				limit = {
					OR = {
						this = c:MGO
						this = c:MGE
					}
				}
				trigger_event_non_silently = mr_dominance.131
			}
			set_global_variable = mr_beat_131_fired
		}

		# --- The westward campaign against Russia (~18 years) ---
		if = {
			limit = {
				var:mr_dominance_momentum >= 216
				NOT = { has_global_variable = mr_beat_201i_fired }
				any_country = {
					OR = {
						this = c:MGO
						this = c:MGE
					}
					any_owned_location = { region = region:russian_region }
				}
			}
			every_country = {
				limit = {
					OR = {
						this = c:MGO
						this = c:MGE
					}
				}
				trigger_event_non_silently = mr_imperial.201
			}
			set_global_variable = mr_beat_201i_fired
		}

		# --- The Grand Review (~30 years) ---
		if = {
			limit = {
				var:mr_dominance_momentum >= 360
				NOT = { has_global_variable = mr_beat_202i_fired }
			}
			every_country = {
				limit = {
					OR = {
						this = c:MGO
						this = c:MGE
					}
				}
				trigger_event_non_silently = mr_imperial.202
			}
			set_global_variable = mr_beat_202i_fired
		}

		# --- The retreat, if the west holds (~40 years, still short) ---
		if = {
			limit = {
				var:mr_dominance_momentum >= 480
				NOT = { has_global_variable = mr_beat_211i_fired }
				NOT = { mr_dominance_end_trigger = yes }
			}
			every_country = {
				limit = {
					OR = {
						this = c:MGO
						this = c:MGE
					}
				}
				trigger_event_non_silently = mr_imperial.211
			}
			set_global_variable = mr_beat_211i_fired
		}

		# --- COMPLETION FAILSAFE: five years before 1650 ---
		# Grants MGE_f's own nine required locations outright, so the
		# formable is genuinely reachable at the end of the railroad.
		# AI-only, and never takes a location from the player.
		if = {
			limit = {
				current_date > 1645.1.1
				NOT = { mr_dominance_end_trigger = yes }
				NOT = { has_global_variable = mr_completion_failsafe_fired }
				any_country = {
					OR = {
						this = c:MGO
						this = c:MGE
					}
					is_ai = yes
				}
			}
			set_global_variable = mr_completion_failsafe_fired

			random_country = {
				limit = {
					OR = {
						this = c:MGO
						this = c:MGE
					}
					is_ai = yes
				}
				save_scope_as = mr_claimant
			}

			# The areas holding MGE_f's nine required locations. Iterating
			# areas rather than naming locations one by one keeps this
			# readable and hands over the surrounding land too — the same
			# shape as PD's every_ownable_location_in_area blocks.
			area:upper_selenga_area = {
				every_ownable_location_in_area = {
					limit = {
						owner = {
							is_ai = yes
							NOR = {
								this = scope:mr_claimant
								is_subject_of = scope:mr_claimant
							}
						}
					}
					change_location_owner = scope:mr_claimant
					add_core = scope:mr_claimant
				}
			}
			area:lower_don_area = {
				every_ownable_location_in_area = {
					limit = {
						owner = {
							is_ai = yes
							NOR = {
								this = scope:mr_claimant
								is_subject_of = scope:mr_claimant
							}
						}
					}
					change_location_owner = scope:mr_claimant
					add_core = scope:mr_claimant
				}
			}
			area:kazan_area = {
				every_ownable_location_in_area = {
					limit = {
						owner = {
							is_ai = yes
							NOR = {
								this = scope:mr_claimant
								is_subject_of = scope:mr_claimant
							}
						}
					}
					change_location_owner = scope:mr_claimant
					add_core = scope:mr_claimant
				}
			}
			area:transoxiana_area = {
				every_ownable_location_in_area = {
					limit = {
						owner = {
							is_ai = yes
							NOR = {
								this = scope:mr_claimant
								is_subject_of = scope:mr_claimant
							}
						}
					}
					change_location_owner = scope:mr_claimant
					add_core = scope:mr_claimant
				}
			}
			area:crimea_area = {
				every_ownable_location_in_area = {
					limit = {
						owner = {
							is_ai = yes
							NOR = {
								this = scope:mr_claimant
								is_subject_of = scope:mr_claimant
							}
						}
					}
					change_location_owner = scope:mr_claimant
					add_core = scope:mr_claimant
				}
			}
			area:beiping_area = {
				every_ownable_location_in_area = {
					limit = {
						owner = {
							is_ai = yes
							NOR = {
								this = scope:mr_claimant
								is_subject_of = scope:mr_claimant
							}
						}
					}
					change_location_owner = scope:mr_claimant
					add_core = scope:mr_claimant
				}
			}
			area:jiangnan_area = {
				every_ownable_location_in_area = {
					limit = {
						owner = {
							is_ai = yes
							NOR = {
								this = scope:mr_claimant
								is_subject_of = scope:mr_claimant
							}
						}
					}
					change_location_owner = scope:mr_claimant
					add_core = scope:mr_claimant
				}
			}
			area:chuandong_area = {
				every_ownable_location_in_area = {
					limit = {
						owner = {
							is_ai = yes
							NOR = {
								this = scope:mr_claimant
								is_subject_of = scope:mr_claimant
							}
						}
					}
					change_location_owner = scope:mr_claimant
					add_core = scope:mr_claimant
				}
			}
			area:guangdong_area = {
				every_ownable_location_in_area = {
					limit = {
						owner = {
							is_ai = yes
							NOR = {
								this = scope:mr_claimant
								is_subject_of = scope:mr_claimant
							}
						}
					}
					change_location_owner = scope:mr_claimant
					add_core = scope:mr_claimant
				}
			}

			scope:mr_claimant = {
				trigger_event_silently = mr_dominance.999
			}
		}
	}

	on_ending = {
		# The railroad either completes, or the empire crumbles.
		if = {
			limit = {
				mr_dominance_end_trigger = yes
			}
			# Form the Mongol Empire. The phase's end trigger is MGE_f's
			# own nine required locations, so by this point the formable
			# is genuinely available rather than merely thematic.
			every_country = {
				limit = {
					this = c:MGO
					NOT = { country_exists = c:MGE }
				}
				form_country = formable_country:MGE_f
			}
			every_country = {
				limit = {
					OR = {
						this = c:MGO
						this = c:MGE
					}
				}
				trigger_event_non_silently = mr_dominance.132
			}
		}
		else = {
			every_country = {
				limit = {
					OR = {
						this = c:MGO
						this = c:MGE
					}
				}
				trigger_event_non_silently = mr_dominance.133
			}
		}
	}

	on_ended = {
		remove_variable = mr_dominance_momentum
		remove_global_variable = mr_beat_131_fired
		remove_global_variable = mr_beat_201i_fired
		remove_global_variable = mr_beat_202i_fired
		remove_global_variable = mr_beat_211i_fired

		# The wiki recommends clearing situation variables in on_ended
		# to keep them out of the savefile.
		remove_variable = mr_leading_country
		remove_variable = mr_rival_country
	}

	tooltip = {
		if = {
			limit = {
				OR = {
					region = region:russian_region
					region = region:steppes_region
					region = region:persia_region
				}
			}
			custom_tooltip = mr_dominance_map_tt
		}
	}

	map_color = {
		if = {
			limit = {
				OR = {
					owner ?= c:MGO
					owner ?= c:MGE
				}
			}
			value = this.owner.country_color
		}
		else = {
			value = define:NMapColors|DEFAULT_COLOR
		}
	}

	secondary_map_color = {
		if = {
			limit = {
				owner ?= {
					OR = {
						is_subject_of = c:MGO
						is_subject_of = c:MGE
					}
				}
			}
			value = dark_grey
		}
		else_if = {
			limit = {
				region = region:russian_region
				NOR = {
					owner ?= c:MGO
					owner ?= c:MGE
				}
			}
			value = red
		}
	}
}
