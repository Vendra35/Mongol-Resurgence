# =================================================================
# PHASE 3 - THE SILK EMPIRE (1550-1650)
# =================================================================
# The westward advance against Russia and the final formable. The
# phase closes when one claimant holds all nine of MGE_f's required
# locations AND has planted its banner in the Russian lands.
#
# NOTE ON c:MGE: MGE does not exist for almost this entire phase —
# it is formed in on_ending. Identity checks therefore use tag =
# (string comparison, never errors) and every c:MGE link sits
# behind a country_exists guard; this = c:MGE comparisons were the
# mod's biggest source of "Invalid right side" error-log spam.
#
# AI railroad: same machine as Phases 1-2 (PD Ascension pattern).
# The claimant is scoped dynamically (random_country over tag MGO /
# MGE) because a player can form MGE mid-phase; wars are declared
# by mr_dominance.992, which picks the westward or silk-road CB by
# where the target's land lies. The AI claimant carries
# MR_mongol_preparing_for_conquest for the whole phase.
# =================================================================
mongol_dominance = {
	monthly_spawn_chance = monthly_spawn_chance_unique
	can_start = {
		mr_can_start_dominance = yes
	}
	visible = {
		mr_dominance_visible = yes
	}
	can_end = {
		OR = {
			mr_dominance_end_trigger = yes
			# Time expiry. Without this the situation can only ever end in
			# success, which leaves on_ending's failure branch unreachable
			# and lets a failed phase run to the end of the campaign.
			custom_tooltip = {
				text = mr_dominance_timeout_tt
				current_date > 1650.1.1
			}
		}
	}
	hint_tag = hint_mongol_dominance
	on_start = {
		set_global_variable = mr_dominance_started
		set_variable = {
			name = mr_dominance_momentum
			value = 0
		}
		# --- AI RAILROAD BOOKKEEPING (PD ascension shape) ---
		# MGO is guaranteed to exist here (can_start requires it), so
		# seeding the target variable with c:MGO is safe. The event's
		# after block re-seeds with root, which handles the MGE case.
		set_variable = {
			name = mr_dom_conquest_cooldown
			value = 48
		}
		set_variable = {
			name = mr_dom_conquest_target
			value = 0
		}
		set_variable = {
			name = mr_dom_conquest_target_country
			value = c:MGO
		}
		# Values the .gui panel reads off the situation scope.
		if = {
			limit = {
				country_exists = c:MGO
			}
			set_variable = {
				name = mr_leading_country
				value = c:MGO
			}
		}
		random_country = {
			limit = {
				OR = {
					any_owned_location = {
						region = region:russian_region
					}
					any_owned_location = {
						region = region:steppes_region
					}
					any_owned_location = {
						region = region:ural_region
					}
				}
				NOR = {
					tag = MGO
					tag = MGE
				}
			}
			save_scope_as = mr_theatre_rival
		}
		if = {
			limit = {
				exists = scope:mr_theatre_rival
			}
			set_variable = {
				name = mr_rival_country
				value = scope:mr_theatre_rival
			}
		}
		every_country = {
			limit = {
				OR = {
					tag = MGO
					tag = MGE
				}
			}
			# The Phase 2 reward expires here, as its tooltip promises
			# ("removed upon the start of the 3rd Situation").
			remove_country_modifier = MR_empire_fulfilled
			# The war-preparation lock, PD Ascension style: the AI
			# claimant must not freelance-declare — the railroad event
			# declares for it. Never applied to a human player.
			if = {
				limit = {
					is_ai = yes
				}
				add_country_modifier = {
					modifier = MR_mongol_preparing_for_conquest
					years = -1
					mode = replace
				}
			}
			trigger_event_non_silently = mr_dominance.130
			# Phase buff per the buff rule, PD-style: Terminator gets
			# the Pax Mongolica engine, Historical the balanced
			# variant, Vanilla nothing.
			if = {
				limit = {
					has_game_rule = MR_buff_enabled
				}
				add_country_modifier = {
					modifier = MR_pax_mongolica
					years = -1
					mode = replace
				}
			}
			else_if = {
				limit = {
					has_game_rule = MR_buff_historical
				}
				add_country_modifier = {
					modifier = MR_dominance_historical_modifier
					years = -1
					mode = replace
				}
			}
		}
		# The claimant gets a CB against the Russian principalities.
		every_country = {
			limit = {
				any_owned_location = {
					region = region:russian_region
				}
				NOR = {
					tag = MGO
					tag = MGE
				}
			}
			save_scope_as = mr_westward_target
			every_country = {
				limit = {
					OR = {
						tag = MGO
						tag = MGE
					}
				}
				add_casus_belli = {
					target = scope:mr_westward_target
					type = casus_belli:cb_MR_westward_advance
					# Covers the whole 1550-1650 phase window.
					years = 100
				}
			}
		}
		# The China and Transoxiana seats MGE_f still demands are covered
		# by re-granting the Silk Road CB — the Phase 2 grant expired
		# decades before this phase begins.
		every_country = {
			limit = {
				OR = {
					any_owned_location = {
						region = region:north_china_region
					}
					any_owned_location = {
						region = region:east_china_region
					}
					any_owned_location = {
						region = region:west_china_region
					}
					any_owned_location = {
						region = region:south_china_region
					}

						region = region:khorasan_region
					}

					tag = MGO
					tag = MGE
				}
			}
			save_scope_as = mr_seat_target
			every_country = {
				limit = {
					OR = {
						tag = MGO
						tag = MGE
					}
				}
				add_casus_belli = {
					target = scope:mr_seat_target
					type = casus_belli:cb_MR_silk_road
					# Covers the whole 1550-1650 phase window.
					years = 100
				}
			}
		}
	}
	on_monthly = {
		change_variable = {
			name = mr_dominance_momentum
			add = 1
		}
		# Keep the panel header on the current claimant: once MGE forms,
		# it replaces MGO as the leading country.
		if = {
			limit = {
				country_exists = c:MGE
				NOT = {
					has_global_variable = mr_mge_seen
				}

				name = mr_leading_country
				value = c:MGE
			}

		}
		# ========== PANEL PROGRESS (0-100) ==========
		# The .gui progressbar reads MR_dominance_score: ten points per
		# imperial seat held by the claimant, ten for the Russian
		# foothold. 100 = the end trigger is satisfied.
		set_variable = {
			name = MR_dominance_score
			value = 0
		}
		if = {
			limit = {
				any_country = {
					OR = {
						tag = MGO
						tag = MGE
					}
					owns = location:karakorum
				}
			}
			change_variable = {
				name = MR_dominance_score
				add = 10
			}
		}
		if = {
			limit = {
				any_country = {
					OR = {
						tag = MGO
						tag = MGE
					}
					owns = location:sarai_al_jadid
				}
			}
			change_variable = {
				name = MR_dominance_score
				add = 10
			}
		}
		if = {
			limit = {
				any_country = {
					OR = {

		# The claimant is whichever of MGO/MGE exists (a player can form

		# weakest neighbour holding land in either theatre — the
		# westward goal regions (russian / steppes / ural) or the seat

		if = {
			limit = {
				var:mr_dom_conquest_target = 0

					OR = {
						tag = MGO

					OR = {
						tag = MGO
						tag = MGE

					at_war = no
				}

					limit = {
						is_rebel_country = no
						NOT = {
							has_truce_with = prev
						}
						NOT = {
							is_subject_of = prev
						}

						OR = {
							any_owned_location = {
								region = region:russian_region
							}
							any_owned_location = {
								region = region:steppes_region
							}
							any_owned_location = {
								region = region:ural_region
							}
							any_owned_location = {
								region = region:mongolia_region
							}
							any_owned_location = {
								region = region:khorasan_region
							}
							any_owned_location = {
								region = region:xinjiang_region
							}
							any_owned_location = {
								region = region:north_china_region
							}
							any_owned_location = {
								region = region:east_china_region
							}
							any_owned_location = {
								region = region:west_china_region
							}
							any_owned_location = {
								region = region:south_china_region
							}
						}
					}
					max = 1
					order_by = {
						value = num_locations
						multiply = -1
					}
					situation:mongol_dominance = {
						set_variable = {
							name = mr_dom_conquest_target_country
							value = prev
						}
						set_variable = {
							name = mr_dom_conquest_target
							value = 1
						}
					}
				}
			}
		}
		change_variable = {
			name = mr_dom_conquest_cooldown
			add = 1
		}
		# --- FALLBACK: the chosen target stopped being valid ---
		# (dead, became the claimant itself, or lost every holding in
		# both theatres). Subject/truce staleness is re-checked by the
		# declare gate below, PD-style.
		if = {
			limit = {
				var:mr_dom_conquest_target = 1
				OR = {
					NOT = {
						exists = var:mr_dom_conquest_target_country
					}
					var:mr_dom_conquest_target_country = {
						OR = {
							tag = MGO
							tag = MGE
							NOT = {
								OR = {
									any_owned_location = {
										region = region:russian_region
									}
									any_owned_location = {
										region = region:steppes_region
									}
									any_owned_location = {
										region = region:ural_region
									}
									any_owned_location = {
										region = region:mongolia_region
									}
									any_owned_location = {
										region = region:khorasan_region
									}
									any_owned_location = {
										region = region:xinjiang_region
									}
									any_owned_location = {
										region = region:north_china_region
									}
									any_owned_location = {
										region = region:east_china_region
									}
									any_owned_location = {
										region = region:west_china_region
									}
									any_owned_location = {
										region = region:south_china_region
									}
								}
							}
						}
					}
				}
			}
			set_variable = {
				name = mr_dom_conquest_target
				value = 0
			}
		}
		# ========== AI RAILROAD: DECLARE ==========
		# PD Ascension pacing: Terminator roughly yearly, otherwise
		# every ~5 years, with the same months_since_war gates.
		if = {
			limit = {
				var:mr_dom_conquest_target = 1
				OR = {
					AND = {
						has_game_rule = MR_buff_enabled
						var:mr_dom_conquest_cooldown > 12
					}
					AND = {
						NOT = {
							has_game_rule = MR_buff_enabled
						}
						var:mr_dom_conquest_cooldown > 60
					}
				}
				any_country = {
					OR = {
						tag = MGO
						tag = MGE
					}
				}
			}
			random_country = {
				limit = {
					OR = {
						tag = MGO
						tag = MGE
					}
				}
				if = {
					limit = {
						OR = {
							AND = {
								has_game_rule = MR_buff_enabled
								months_since_war >= 12
							}
							AND = {
								NOT = {
									has_game_rule = MR_buff_enabled
								}
								months_since_war >= 24
							}
						}
						situation:mongol_dominance = {
							var:mr_dom_conquest_target_country = {
								NOT = {
									is_subject_of = prev
								}
								NOT = {
									has_truce_with = prev
								}
							}
						}
					}
					trigger_event_silently = mr_dominance.992
				}
			}
		}
		# --- The New World Order (~10 years) ---
		if = {
			limit = {
				var:mr_dominance_momentum >= 120
				NOT = {
					has_global_variable = mr_beat_131_fired
				}
			}
			every_country = {
				limit = {
					OR = {
						tag = MGO
						tag = MGE
					}
				}
				trigger_event_non_silently = mr_dominance.131
			}
			set_global_variable = mr_beat_131_fired
		}
		# --- The westward campaign against Russia (~18 years) ---
		if = {
			limit = {
				var:mr_dominance_momentum >= 216
				NOT = {
					has_global_variable = mr_beat_201i_fired
				}
				any_country = {
					OR = {
						tag = MGO
						tag = MGE
					}
					any_owned_location = {
						region = region:russian_region
					}
				}
			}
			every_country = {
				limit = {
					OR = {
						tag = MGO
						tag = MGE
					}
				}
				trigger_event_non_silently = mr_imperial.201
			}
			set_global_variable = mr_beat_201i_fired
		}
		# --- The Grand Review (~30 years) ---
		if = {
			limit = {
				var:mr_dominance_momentum >= 360
				NOT = {
					has_global_variable = mr_beat_202i_fired
				}
			}
			every_country = {
				limit = {
					OR = {
						tag = MGO
						tag = MGE
					}
				}
				trigger_event_non_silently = mr_imperial.202
			}
			set_global_variable = mr_beat_202i_fired
		}
		# --- The retreat, if the west holds (~40 years, still short) ---
		if = {
			limit = {
				var:mr_dominance_momentum >= 480
				NOT = {
					has_global_variable = mr_beat_211i_fired
				}
				NOT = {
					mr_dominance_end_trigger = yes
				}
			}
			every_country = {
				limit = {
					OR = {
						tag = MGO
						tag = MGE
					}
				}
				trigger_event_non_silently = mr_imperial.211
			}
			set_global_variable = mr_beat_211i_fired
		}
		# --- COMPLETION FAILSAFE: five years before 1650 ---
		# Grants MGE_f's own nine required locations outright — plus a
		# Russian foothold (Ryazan, historically the first principality
		# the Mongols reached in 1237) so the end trigger's westward
		# clause is genuinely reachable. Gated on the imperial
		# conquest-automation rule and on is_ai; never takes a location
		# from the player.

			limit = {
				has_game_rule = MR_imperial_auto_conquest_yes
				current_date > 1645.1.1
				NOT = {
					mr_dominance_end_trigger = yes
				}
				NOT = {
					has_global_variable = mr_failsafe_p3_fired
				}
				any_country = {
					OR = {
						tag = MGO
						tag = MGE
					}
					is_ai = yes
					at_war = no
				}
			}
			set_global_variable = mr_failsafe_p3_fired
			random_country = {
				limit = {
					OR = {
						tag = MGO
						tag = MGE
					}
					is_ai = yes
					at_war = no
				}

	on_ending = {
		# 1. Strip the phase modifiers, PD-style.
		every_country = {
			limit = {
				OR = {
					tag = MGO
					tag = MGE
				}
			}
			remove_country_modifier = MR_pax_mongolica
			remove_country_modifier = MR_dominance_historical_modifier
			remove_country_modifier = MR_mongol_preparing_for_conquest
		}
		# 2. The railroad either completes, or the empire crumbles.
		if = {
			limit = {
				mr_dominance_end_trigger = yes
			}
			# Form the Mongol Empire. The phase's end trigger includes
			# MGE_f's own nine required locations, so by this point the
			# formable is genuinely available rather than merely thematic.
			every_country = {
				limit = {
					tag = MGO
					NOT = {
						country_exists = c:MGE
					}
				}
				form_country = formable_country:MGE_f
			}
			every_country = {
				limit = {
					OR = {
						tag = MGO
						tag = MGE
					}
				}
				trigger_event_non_silently = mr_dominance.132
			}
			set_global_variable = mr_railroad_complete
		}
		else = {
			every_country = {
				limit = {
					OR = {
						tag = MGO
						tag = MGE
					}
				}
				trigger_event_non_silently = mr_dominance.133
			}
			set_global_variable = mr_railroad_failed
		}
	}
