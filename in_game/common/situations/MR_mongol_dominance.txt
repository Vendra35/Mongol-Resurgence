# =================================================================
# PHASE 3 - THE FOUR KHANATES (1550-1650)
# =================================================================
# The westward finale. The Yeke Mongol Ulus (MGE) is proclaimed at
# Phase 2's close now, so this phase opens WITH the empire on the
# map and asks it to restore the old ulus in full: the Yuan seats
# (Karakorum, Dadu), the Chagatai seat (Samarkand), the Golden
# Horde seats (Sarai, Kazan), and the whole Ilkhanate — Tabriz,
# Baghdad, Persia cleared of every outside power — plus a foothold
# in the Russian lands and one in Cappadocia (Kose Dag country).
#
# NOTE ON c:MGE: MGE exists from on_start on the normal path, which
# retires the error-spam problem this file used to fight — but the
# MGO fallback (a failed proclamation) is still tolerated, so
# identity checks keep using tag = (string comparison, never
# errors) and every c:MGE / c:MGO link still sits behind a
# country_exists guard in the same AND.
#
# AI railroad: same machine as Phases 1-2 (PD Ascension pattern).
# The claimant is scoped dynamically (random_country over tag MGE /
# MGO); wars are declared by mr_dominance.992, which picks the
# westward or silk-road CB by where the target's land lies. The AI
# claimant carries MR_mongol_preparing_for_conquest all phase.
# =================================================================
mongol_dominance = {
	monthly_spawn_chance = monthly_spawn_chance_unique
	can_start = {
		mr_can_start_dominance = yes
	}
	visible = {
		mr_dominance_visible = yes
	}
	can_end = {
		OR = {
			mr_dominance_end_trigger = yes
			# Time expiry. Without this the situation can only ever end in
			# success, which leaves on_ending's failure branch unreachable
			# and lets a failed phase run to the end of the campaign.
			custom_tooltip = {
				text = mr_dominance_timeout_tt
				current_date > 1650.1.1
			}
		}
	}
	hint_tag = hint_mongol_dominance
	on_start = {
		set_global_variable = mr_dominance_started
		set_variable = {
			name = mr_dominance_momentum
			value = 0
		}
		# --- AI RAILROAD BOOKKEEPING (PD ascension shape) ---
		# The claimant is normally MGE now (proclaimed at Phase 2's
		# close), with MGO as the failed-proclamation fallback —
		# can_start guarantees one of the two exists. The seed target
		# is the claimant itself, PD's "no target" sentinel; the
		# event's after block re-seeds with root.
		set_variable = {
			name = mr_dom_conquest_cooldown
			value = 48
		}
		set_variable = {
			name = mr_dom_conquest_target
			value = 0
		}
		if = {
			limit = {
				country_exists = c:MGE
			}
			set_variable = {
				name = mr_dom_conquest_target_country
				value = c:MGE
			}
			# Values the .gui panel reads off the situation scope.
			set_variable = {
				name = mr_leading_country
				value = c:MGE
			}
		}
		else_if = {
			limit = {
				country_exists = c:MGO
			}
			set_variable = {
				name = mr_dom_conquest_target_country
				value = c:MGO
			}
			set_variable = {
				name = mr_leading_country
				value = c:MGO
			}
		}
		random_country = {
			limit = {
				OR = {
					has_presence_in = region:russian_region
					has_presence_in = region:steppes_region
					has_presence_in = region:ural_region
				}
				NOR = {
					tag = MGO
					tag = MGE
				}
			}
			save_scope_as = mr_theatre_rival
		}
		if = {
			limit = {
				country_exists = scope:mr_theatre_rival
			}
			set_variable = {
				name = mr_rival_country
				value = scope:mr_theatre_rival
			}
		}
		every_country = {
			limit = {
				OR = {
					tag = MGO
					tag = MGE
				}
			}
			# The Phase 2 reward (MR_empire_fulfilled) is deliberately
			# NOT stripped here — it is permanent, marking the
			# proclamation of the Yeke Mongol Ulus, and its tooltip says
			# so. Only phase BUFFS are temporary, and Phase 2's own
			# on_ending removes those.
			# The war-preparation lock, PD Ascension style: the AI
			# claimant must not freelance-declare — the railroad event
			# declares for it. Never applied to a human player.
			if = {
				limit = {
					is_ai = yes
				}
				add_country_modifier = {
					modifier = MR_mongol_preparing_for_conquest
					years = -1
					mode = replace
				}
			}
			# --- THE SEAT MOVES TO KHANBALIQ ---
			# Kublai moved the capital from Karakorum to Dadu in 1272,
			# and the empire stopped being a steppe confederation the
			# day he did. Phase 3 opens on the same decision. Guarded
			# three ways: the claimant must actually OWN Dadu (the
			# Phase 2 goal guarantees it on the success path, but the
			# fallback path does not), it must not already be sitting
			# there, and a HUMAN is asked rather than moved — the mod
			# never relocates a player's capital behind their back.
			if = {
				limit = {
					owns = location:dadu
					NOT = {
						capital = location:dadu
					}
				}
				if = {
					limit = {
						is_ai = yes
					}
					set_capital = location:dadu
					trigger_event_non_silently = mr_dominance.136
				}
				else = {
					trigger_event_non_silently = mr_dominance.137
				}
			}
			trigger_event_non_silently = mr_dominance.130
			# Phase buff per the buff rule, PD-style: Terminator gets
			# the Pax Mongolica engine, Historical the balanced
			# variant, Vanilla nothing.
			if = {
				limit = {
					has_game_rule = MR_buff_enabled
				}
				add_country_modifier = {
					modifier = MR_pax_mongolica
					years = -1
					mode = replace
				}
			}
			else_if = {
				limit = {
					has_game_rule = MR_buff_historical
				}
				add_country_modifier = {
					modifier = MR_dominance_historical_modifier
					years = -1
					mode = replace
				}
			}
		}
		# The claimant gets the westward CB against every power holding
		# the western goal theatres: the Russian lands, the Pontic and
		# Ural steppes, Persia, Mesopotamia and Anatolia — the Golden
		# Horde and Ilkhanate domains this phase demands.
		every_country = {
			limit = {
				OR = {
					has_presence_in = region:russian_region
					has_presence_in = region:steppes_region
					has_presence_in = region:ural_region
					has_presence_in = region:persia_region
					has_presence_in = region:anatolia_region
					has_presence_in = region:caucasus_region
					has_presence_in = area:iraq_arabi_area
					has_presence_in = area:armenian_highlands_area
				}
				NOR = {
					tag = MGO
					tag = MGE
				}
			}
			save_scope_as = mr_westward_target
			every_country = {
				limit = {
					OR = {
						tag = MGO
						tag = MGE
					}
				}
				add_casus_belli = {
					target = scope:mr_westward_target
					type = casus_belli:cb_MR_westward_advance
					# Covers the whole 1550-1650 phase window.
					years = 100
				}
			}
		}
		# The eastern seats (Karakorum, Dadu, Samarkand) stay covered by
		# re-granting the Silk Road CB — the Phase 2 grant expired
		# decades before this phase begins, and a seat lost after Phase
		# 2 must stay legally retakable.
		every_country = {
			limit = {
				OR = {
					has_presence_in = region:north_china_region
					has_presence_in = region:mongolia_region
					has_presence_in = region:xinjiang_region
					has_presence_in = region:khorasan_region
					has_presence_in = region:east_china_region
					has_presence_in = region:west_china_region
					has_presence_in = region:south_china_region
					has_presence_in = region:korea_region
				}
				NOR = {
					tag = MGO
					tag = MGE
				}
			}
			save_scope_as = mr_seat_target
			every_country = {
				limit = {
					OR = {
						tag = MGO
						tag = MGE
					}
				}
				add_casus_belli = {
					target = scope:mr_seat_target
					type = casus_belli:cb_MR_silk_road
					# Covers the whole 1550-1650 phase window.
					years = 100
				}
			}
		}
		# --- THE WORLD REACTS (PD pd_brandenburg.201/202/200 shape) ---
		# SPECTATORS: a stake in the theatre, no land the phase demands.
		every_country = {
			limit = {
				OR = {
					has_presence_in = sub_continent:east_asia
					has_presence_in = sub_continent:north_asia
					has_presence_in = sub_continent:central_asia
					has_presence_in = sub_continent:middle_east
					has_presence_in = sub_continent:south_asia
					has_presence_in = sub_continent:eastern_europe
				}
				NOT = {
					OR = {
						has_presence_in = region:russian_region
						has_presence_in = region:steppes_region
						has_presence_in = region:ural_region
						has_presence_in = region:persia_region
						has_presence_in = region:anatolia_region
						has_presence_in = region:caucasus_region
						has_presence_in = region:east_china_region
						has_presence_in = region:west_china_region
						has_presence_in = region:south_china_region
					}
				}
				NOR = {
					tag = MGO
					tag = MGE
				}
			}
			trigger_event_non_silently = mr_dominance.25
		}
		# VICTIMS: they hold ground inside the phase's goal regions.
		every_country = {
			limit = {
				OR = {
					has_presence_in = region:russian_region
					has_presence_in = region:steppes_region
					has_presence_in = region:ural_region
					has_presence_in = region:persia_region
					has_presence_in = region:anatolia_region
					has_presence_in = region:caucasus_region
					has_presence_in = region:east_china_region
					has_presence_in = region:west_china_region
					has_presence_in = region:south_china_region
				}
				NOR = {
					tag = MGO
					tag = MGE
				}
			}
			trigger_event_non_silently = mr_dominance.26
		}
		# The claimant sheds alliances with those same powers, so the
		# railroad's declare event is not blocked by its own treaties.
		every_country = {
			limit = {
				OR = {
					tag = MGO
					tag = MGE
				}
				# AI ONLY. PD fires its equivalent at c:PRU unconditionally
				# (the_prussian_ascension.txt:138), but PD has no standing
				# rule about it and this mod does: never force-convert, lock
				# or rob a human player. Dissolving a player's treaties with
				# no way to refuse is exactly that. It is also unnecessary
				# for them — the reason this exists is that the AI's declare
				# event stalls on its own alliances, and a human gets a
				# visible, postponable war event and can break alliances
				# themselves.
				is_ai = yes
			}
			trigger_event_non_silently = mr_dominance.27
		}
	}
	on_monthly = {
		change_variable = {
			name = mr_dominance_momentum
			add = 1
		}
		# Keep the panel header on the current claimant: once MGE forms,
		# it replaces MGO as the leading country.
		if = {
			limit = {
				country_exists = c:MGE
				NOT = {
					has_global_variable = mr_mge_seen
				}
			}
			set_variable = {
				name = mr_leading_country
				value = c:MGE
			}
			set_global_variable = mr_mge_seen
		}
		# ========== THE KHAN'S OWN PASTURE ==========
		# The ongoing half of the MR_select_core_region action. The
		# action itself cannot be used in this phase (its situation
		# select_trigger lists Phases 1 and 2 only), but a pasture named
		# back then is still the claimant's, so it keeps working here —
		# without this the mechanic silently stopped at 1550. Vanilla
		# does this in on_location_changed_owner (_hardcoded.txt:4489),
		# but that on_action is defined with an effect block and
		# redefining the name in a mod would replace the vanilla block
		# outright — so the situation does it monthly instead. Anything
		# the claimant has taken inside its chosen region since last
		# month becomes core, and the Khan's death ends the privilege
		# exactly as it ends Timur's (_hardcoded.txt:3517).
		every_country = {
			limit = {
				OR = {
					tag = MGO
					tag = MGE
				}
				has_variable = mr_core_region
			}
			# Saved rather than reached with prev.prev: that construct has
			# ZERO uses anywhere in vanilla, and this file does not invent
			# syntax. save_scope_as + scope: is the attested way down.
			save_scope_as = mr_pasture_holder
			if = {
				limit = {
					has_ruler = yes
					ruler ?= {
						has_character_modifier = MR_great_khan
					}
				}
				var:mr_core_region = {
					# TWO passes, deliberately not one. Coring is a
					# one-way door, so it is limited to what is not core
					# yet; the modifier is NOT, so it is applied on its
					# own terms to every owned location that lacks it.
					# Folding them together (one limit, both effects) was
					# a real bug: the else branch below strips the
					# modifier from every location, and a single-pass
					# version could only re-add it to locations that were
					# still un-cored — which, after the first pass, is
					# none of them. The pasture modifier was lost for
					# good after any interregnum tick.
					every_location_in_region = {
						limit = {
							owner ?= scope:mr_pasture_holder
							NOT = {
								integration_level = core
							}
						}
						change_integration_level = core
					}
					every_location_in_region = {
						limit = {
							owner ?= scope:mr_pasture_holder
							NOT = {
								has_location_modifier = MR_khans_own_pasture
							}
						}
						add_location_modifier = {
							modifier = MR_khans_own_pasture
							years = -1
							mode = add_and_extend
						}
					}
				}
			}
			# No Great Khan on the throne: the pasture is just land again.
			else = {
				var:mr_core_region = {
					every_location_in_region = {
						limit = {
							has_location_modifier = MR_khans_own_pasture
						}
						remove_location_modifier = MR_khans_own_pasture
					}
				}
			}
		}
		# ========== AI RAILROAD: FIND A TARGET ==========
		# The claimant is whichever of MGO/MGE exists (a player can form
		# MGE mid-phase), scoped via random_country. Targets: the
		# weakest neighbour holding land in either theatre — the
		# westward goal regions (russian / steppes / ural) or the seat
		# regions (mongolia / khorasan / xinjiang / the four Chinas).
		if = {
			limit = {
				var:mr_dom_conquest_target = 0
				any_country = {
					OR = {
						tag = MGO
						tag = MGE
					}
					at_war = no
				}
			}
			random_country = {
				limit = {
					OR = {
						tag = MGO
						tag = MGE
					}
					at_war = no
				}
				save_scope_as = mr_dom_claimant
				ordered_neighbor_country = {
					limit = {
						is_rebel_country = no
						NOT = {
							has_truce_with = prev
						}
						NOT = {
							is_subject_of = prev
						}
						defensive_alliance_strength < scope:mr_dom_claimant.offensive_alliance_strength
						OR = {
							has_presence_in = region:russian_region
							has_presence_in = region:steppes_region
							has_presence_in = region:ural_region
							has_presence_in = region:persia_region
							has_presence_in = region:anatolia_region
							has_presence_in = region:caucasus_region
							has_presence_in = region:mongolia_region
							has_presence_in = region:khorasan_region
							has_presence_in = region:xinjiang_region
							has_presence_in = region:north_china_region
							has_presence_in = region:east_china_region
							has_presence_in = region:west_china_region
							has_presence_in = region:south_china_region
							has_presence_in = region:korea_region
							has_presence_in = area:iraq_arabi_area
							has_presence_in = area:armenian_highlands_area
						}
					}
					max = 1
					order_by = {
						value = num_locations
						multiply = -1
					}
					situation:mongol_dominance = {
						set_variable = {
							name = mr_dom_conquest_target_country
							value = prev
						}
						set_variable = {
							name = mr_dom_conquest_target
							value = 1
						}
					}
				}
			}
		}
		change_variable = {
			name = mr_dom_conquest_cooldown
			add = 1
		}
		# --- FALLBACK: the chosen target stopped being valid ---
		# (dead, became the claimant itself, or lost every holding in
		# both theatres). Subject/truce staleness is re-checked by the
		# declare gate below, PD-style.
		if = {
			limit = {
				var:mr_dom_conquest_target = 1
				OR = {
					NOT = {
						country_exists = var:mr_dom_conquest_target_country
					}
					var:mr_dom_conquest_target_country = {
						OR = {
							tag = MGO
							tag = MGE
							NOT = {
								OR = {
									has_presence_in = region:russian_region
									has_presence_in = region:steppes_region
									has_presence_in = region:ural_region
									has_presence_in = region:persia_region
									has_presence_in = region:anatolia_region
									has_presence_in = region:caucasus_region
									has_presence_in = region:mongolia_region
									has_presence_in = region:khorasan_region
									has_presence_in = region:xinjiang_region
									has_presence_in = region:north_china_region
									has_presence_in = region:east_china_region
									has_presence_in = region:west_china_region
									has_presence_in = region:south_china_region
									has_presence_in = region:korea_region
									has_presence_in = area:iraq_arabi_area
									has_presence_in = area:armenian_highlands_area
								}
							}
						}
					}
				}
			}
			set_variable = {
				name = mr_dom_conquest_target
				value = 0
			}
		}
		# ========== AI RAILROAD: DECLARE ==========
		# PD Ascension pacing: Terminator roughly yearly, otherwise
		# every ~5 years, with the same months_since_war gates.
		if = {
			limit = {
				var:mr_dom_conquest_target = 1
				OR = {
					AND = {
						has_game_rule = MR_buff_enabled
						var:mr_dom_conquest_cooldown > 12
					}
					AND = {
						NOT = {
							has_game_rule = MR_buff_enabled
						}
						var:mr_dom_conquest_cooldown > 60
					}
				}
				any_country = {
					OR = {
						tag = MGO
						tag = MGE
					}
				}
			}
			random_country = {
				limit = {
					OR = {
						tag = MGO
						tag = MGE
					}
				}
				if = {
					limit = {
						OR = {
							AND = {
								has_game_rule = MR_buff_enabled
								months_since_war >= 12
							}
							AND = {
								NOT = {
									has_game_rule = MR_buff_enabled
								}
								months_since_war >= 24
							}
						}
						situation:mongol_dominance = {
							var:mr_dom_conquest_target_country = {
								NOT = {
									is_subject_of = prev
								}
								NOT = {
									has_truce_with = prev
								}
							}
						}
					}
					trigger_event_silently = mr_dominance.992
				}
			}
		}
		# --- The New World Order (~10 years) ---
		if = {
			limit = {
				var:mr_dominance_momentum >= 120
				NOT = {
					has_global_variable = mr_beat_131_fired
				}
			}
			every_country = {
				limit = {
					OR = {
						tag = MGO
						tag = MGE
					}
				}
				trigger_event_non_silently = mr_dominance.131
			}
			set_global_variable = mr_beat_131_fired
		}
		# --- The Four Corners: all four khanate capitals held ---
		# Not momentum-timed — this one fires the month it becomes
		# TRUE: Karakorum, Dadu, Samarkand and Sarai under one banner.
		if = {
			limit = {
				NOT = {
					has_global_variable = mr_beat_135_fired
				}
				any_country = {
					OR = {
						tag = MGO
						tag = MGE
					}
					owns = location:karakorum
					owns = location:dadu
					owns = location:samarkand
					owns = location:sarai_al_jadid
				}
			}
			every_country = {
				limit = {
					OR = {
						tag = MGO
						tag = MGE
					}
				}
				trigger_event_non_silently = mr_dominance.135
			}
			set_global_variable = mr_beat_135_fired
		}
		# --- The westward campaign against Russia (~18 years) ---
		if = {
			limit = {
				var:mr_dominance_momentum >= 216
				NOT = {
					has_global_variable = mr_beat_201i_fired
				}
				any_country = {
					OR = {
						tag = MGO
						tag = MGE
					}
					has_presence_in = region:russian_region
				}
			}
			every_country = {
				limit = {
					OR = {
						tag = MGO
						tag = MGE
					}
				}
				trigger_event_non_silently = mr_imperial.201
			}
			set_global_variable = mr_beat_201i_fired
		}
		# --- The Grand Review (~30 years) ---
		if = {
			limit = {
				var:mr_dominance_momentum >= 360
				NOT = {
					has_global_variable = mr_beat_202i_fired
				}
			}
			every_country = {
				limit = {
					OR = {
						tag = MGO
						tag = MGE
					}
				}
				trigger_event_non_silently = mr_imperial.202
			}
			set_global_variable = mr_beat_202i_fired
		}
		# --- The retreat, if the west holds (~40 years, still short) ---
		if = {
			limit = {
				var:mr_dominance_momentum >= 480
				NOT = {
					has_global_variable = mr_beat_211i_fired
				}
				NOT = {
					mr_dominance_end_trigger = yes
				}
