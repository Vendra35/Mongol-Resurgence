# Mongol Railroad Mod - Situation: Mongol Resurgence (Phase 1)

mongol_resurgence_situation = {
	title = "situation_mongol_resurgence_title"
	description = "situation_mongol_resurgence_desc"

	trigger = {
		has_global_variable = mongol_resurgence_active
		NOT = { has_global_variable = mongol_resurgence_complete }
		NOT = { has_global_variable = mongol_railroad_failed }
	}

	visible = {
		mr_resurgence_visible = yes
	}

	icon = gfx_interface_situation_mongol_resurgence

	targets = {
		MGO = {
			country_score = 1
		}
	}

	sort_order = 10

	progress = {
		target = 100
	}

	completion = {
		trigger = {
			mr_resurgence_end_trigger = yes
		}

		effect = {
			set_global_variable = { name = mongol_resurgence_complete }
		}
	}

	abort = {
		trigger = {
			has_global_variable = mongol_railroad_failed
		}
	}

	actions = {
		# Re-emergence: Force MGO formation
		mongol_resurgence_emerge_action = {
			trigger = {
			NOT = { country_exists = c:MGO }
				any_country = {
					culture = { has_culture_group = culture_group:mongolian_group }
					government_type = government_type:steppe_horde
					any_owned_location = { is_in_region = region:mongolia_region }
				}
			}
			effect = {
				random_country = {
					limit = {
						culture = { has_culture_group = culture_group:mongolian_group }
						government_type = government_type:steppe_horde
						any_owned_location = { is_in_region = region:mongolia_region }
					}
					form_country = formable_country:MGO_f
				}
			}
			ai_weight = 1000
		}
	}
}
