# =================================================================
# PHASE 1 - THE NORTHERN YUAN RESURGENCE (1368-1420)
# =================================================================
# The unification of Mongolia. The actor is not a fixed tag: whoever
# holds the Mongolian heartland with a steppe horde government and
# Mongol culture takes up the Chinggisid claim, following vanilla's
# Timur emergence (flavor_tim.8) rather than scripting a country.
#
# Fields here are the documented set from vanilla
# common/situations/readme.txt. EU5 has no title, desc, trigger,
# icon, targets, sort_order, progress, completion, abort or actions
# field on a situation — all of those were fabricated in the
# previous version of this file.
#
# NOTE ON c:MGO: MGO does not exist between 1368 and its emergence
# (~1375 at the latest). Any comparison with c:MGO on the right side
# throws "Invalid right side" script errors until then, so this file
# only ever uses c:MGO behind a country_exists guard in the same
# AND, uses tag = MGO for identity checks, and never compares
# owner ?= c:MGO directly in the map modes.
# =================================================================
mongol_resurgence = {
	monthly_spawn_chance = monthly_spawn_chance_unique
	can_start = {
		mr_can_start_resurgence = yes
	}
	visible = {
		mr_resurgence_visible = yes
	}
	can_end = {
		OR = {
			mr_resurgence_end_trigger = yes
			# Time expiry. Without this the situation can only ever end in
			# success, which leaves on_ending's failure branch unreachable
			# and lets a failed phase run to the end of the campaign.
			custom_tooltip = {
				text = mr_resurgence_timeout_tt
				current_date > 1420.1.1
			}
		}
	}
	hint_tag = hint_mongol_resurgence
	on_start = {
		set_global_variable = mr_resurgence_started
		# Unification pressure. Beats fire as this climbs.
		set_variable = {
			name = mr_resurgence_momentum
			value = 0
		}
		# --- AI RAILROAD BOOKKEEPING ---
		# Same shape as PD's PD_conquest_cooldown / PD_conquest_target /
		# PD_conquest_target_country. The cooldown ticks monthly; when it
		# matures and a target has been picked, the claimant is pushed
		# into a war it would not otherwise start. PD also seeds
		# PD_conquest_target_country with c:BRA here — we cannot, because
		# MGO does not exist yet; the variable is first seeded the month
		# MGO emerges (Beat 104) and re-seeded in mr_dominance.997's
		# after block.
		set_variable = {
			name = mr_conquest_cooldown
			value = 0
		}
		set_variable = {
			name = mr_conquest_target
			value = 0
		}
		# The .gui panel reads these two off the situation scope to
		# draw its dual-portrait header. Without them it renders blank.
		# The claimant must actually hold Mongolian ground — the claim
		# is to the heartland, not to steppe-horde government in general.
		random_country = {
			limit = {
				culture = {
					has_culture_group = culture_group:mongolian_group
				}
				government_type = government_type:steppe_horde
				has_presence_in = scripted_geography:MR_geo_heartland
			}
			save_scope_as = mr_first_claimant
		}
		if = {
			limit = {
				country_exists = scope:mr_first_claimant
			}
			set_variable = {
				name = mr_leading_country
				value = scope:mr_first_claimant
			}
		}
		# The rival is the STRONGEST other steppe power, not a random
		# one — a Persian-fringe horde rivalling the Mongolian claimant
		# looked absurd in testing. order_by = military_strength with
		# max = 1 is the vanilla selection idiom
		# (war_of_religions.txt:54-59); the sub-continent cap keeps the
		# theatre to the eastern steppe.
		ordered_country = {
			limit = {
				culture = {
					has_culture_group = culture_group:mongolian_group
				}
				government_type = government_type:steppe_horde
				NOT = {
					this = scope:mr_first_claimant
				}
				capital ?= {
					OR = {
						sub_continent = sub_continent:east_asia
						sub_continent = sub_continent:north_asia
					}
				}
			}
			order_by = military_strength
			max = 1
			check_range_bounds = no
			save_scope_as = mr_first_rival
		}
		if = {
			limit = {
				country_exists = scope:mr_first_rival
			}
			set_variable = {
				name = mr_rival_country
				value = scope:mr_first_rival
			}
		}
		# The opening event goes to every steppe power with a stake.
		every_country = {
			limit = {
				culture = {
					has_culture_group = culture_group:mongolian_group
				}
				government_type = government_type:steppe_horde
				capital ?= {
					OR = {
						sub_continent = sub_continent:east_asia
						sub_continent = sub_continent:north_asia
					}
				}
			}
			trigger_event_non_silently = mr_dominance.10
			# Phase buff per the buff rule, PD-style (brandenburg_rise
			# grants by BRA_buff_enabled / _historical / _disabled):
			# Terminator gets the full warrior spirit, Historical the
			# balanced variant, Vanilla nothing.
			if = {
				limit = {
					has_game_rule = MR_buff_enabled
				}
				add_country_modifier = {
					modifier = MR_mongol_warrior_spirit
					years = -1
					mode = replace
				}
			}
			else_if = {
				limit = {
					has_game_rule = MR_buff_historical
				}
				add_country_modifier = {
					modifier = MR_mongol_historical_modifier
					years = -1
					mode = replace
				}
			}
			# The CB is granted by the situation rather than picked from
			# the diplomacy screen — cb_MR_* are create_enabled = no.
			every_country = {
				limit = {
					culture = {
						has_culture_group = culture_group:mongolian_group
					}
					government_type = government_type:steppe_horde
					NOT = {
						this = prev
					}
				}
				add_casus_belli = {
					target = prev
					type = casus_belli:cb_MR_steppe_unification
					years = 25
				}
			}
		}
		# --- THE WORLD REACTS (PD pd_brandenburg.101/102 shape) ---
		# SPECTATORS: a stake in the eastern theatre, but no Mongolian
		# ground and not one of the steppe powers that just got .10.
		every_country = {
			limit = {
				OR = {
					has_presence_in = sub_continent:east_asia
					has_presence_in = sub_continent:north_asia
					has_presence_in = sub_continent:central_asia
				}
				NOT = {
					has_presence_in = scripted_geography:MR_geo_heartland
				}
				# Mirrors the mr_dominance.10 recipient set ABOVE exactly,
				# capital filter included. Without the capital clause a
				# Mongol horde seated outside the eastern steppe was
				# excluded here while never having received .10 — it heard
				# nothing at all.
				NOT = {
					AND = {
						culture = {
							has_culture_group = culture_group:mongolian_group
						}
						government_type = government_type:steppe_horde
						capital ?= {
							OR = {
								sub_continent = sub_continent:east_asia
								sub_continent = sub_continent:north_asia
							}
						}
					}
				}
			}
			trigger_event_non_silently = mr_dominance.20
		}
		# VICTIMS: they are standing on the heartland the claim is for.
		# The steppe powers that received .10 are excluded, so nobody
		# gets told twice about the same news.
		every_country = {
			limit = {
				has_presence_in = scripted_geography:MR_geo_heartland
				# Mirrors the mr_dominance.10 recipient set ABOVE exactly,
				# capital filter included. Without the capital clause a
				# Mongol horde seated outside the eastern steppe was
				# excluded here while never having received .10 — it heard
				# nothing at all.
				NOT = {
					AND = {
						culture = {
							has_culture_group = culture_group:mongolian_group
						}
						government_type = government_type:steppe_horde
						capital ?= {
							OR = {
								sub_continent = sub_continent:east_asia
								sub_continent = sub_continent:north_asia
							}
						}
					}
				}
			}
			trigger_event_non_silently = mr_dominance.21
		}
	}
	on_monthly = {
		change_variable = {
			name = mr_resurgence_momentum
			add = 1
		}
		change_variable = {
			name = mr_conquest_cooldown
			add = 1
		}
		# ========== THE KHAN'S OWN PASTURE ==========
		# The ongoing half of the MR_select_core_region action. Vanilla
		# does this in on_location_changed_owner (_hardcoded.txt:4489),
		# but that on_action is defined with an effect block and
		# redefining the name in a mod would replace the vanilla block
		# outright — so the situation does it monthly instead. Anything
		# the claimant has taken inside its chosen region since last
		# month becomes core, and the Khan's death ends the privilege
		# exactly as it ends Timur's (_hardcoded.txt:3517).
		every_country = {
			limit = {
				OR = {
					tag = MGO
					tag = MGE
				}
				has_variable = mr_core_region
			}
			# Saved rather than reached with prev.prev: that construct has
			# ZERO uses anywhere in vanilla, and this file does not invent
			# syntax. save_scope_as + scope: is the attested way down.
			save_scope_as = mr_pasture_holder
			if = {
				limit = {
					has_ruler = yes
					ruler ?= {
						has_character_modifier = MR_great_khan
					}
				}
				var:mr_core_region = {
					# TWO passes, deliberately not one. Coring is a
					# one-way door, so it is limited to what is not core
					# yet; the modifier is NOT, so it is applied on its
					# own terms to every owned location that lacks it.
					# Folding them together (one limit, both effects) was
					# a real bug: the else branch below strips the
					# modifier from every location, and a single-pass
					# version could only re-add it to locations that were
					# still un-cored — which, after the first pass, is
					# none of them. The pasture modifier was lost for
					# good after any interregnum tick.
					every_location_in_region = {
						limit = {
							owner ?= scope:mr_pasture_holder
							NOT = {
								integration_level = core
							}
						}
						change_integration_level = core
					}
					every_location_in_region = {
						limit = {
							owner ?= scope:mr_pasture_holder
							NOT = {
								has_location_modifier = MR_khans_own_pasture
							}
						}
						add_location_modifier = {
							modifier = MR_khans_own_pasture
							years = -1
							mode = add_and_extend
						}
					}
				}
			}
			# No Great Khan on the throne: the pasture is just land again.
			else = {
				var:mr_core_region = {
					every_location_in_region = {
						limit = {
							has_location_modifier = MR_khans_own_pasture
						}
						remove_location_modifier = MR_khans_own_pasture
					}
				}
			}
		}
		# ========== AI RAILROAD: FIND A TARGET ==========
		# Pick the weakest neighbour holding Mongolian ground whose
		# defensive alliance web is weaker than ours. Ordered by
		# num_locations ascending, so the softest target comes first.
		# No is_ai gate, PD-style: a human claimant also gets targets
		# picked and the war event offered — with the option to postpone.
		if = {
			limit = {
				var:mr_conquest_target = 0
				country_exists = c:MGO
				c:MGO = {
					at_war = no
				}
			}
			c:MGO = {
				ordered_neighbor_country = {
					limit = {
						is_rebel_country = no
						NOT = {
							has_truce_with = prev
						}
						NOT = {
							top_overlord_or_this ?= prev
						}
						defensive_alliance_strength < c:MGO.offensive_alliance_strength
						has_presence_in = scripted_geography:MR_geo_heartland
					}
					max = 1
					order_by = {
						value = num_locations
						multiply = -1
					}
					situation:mongol_resurgence = {
						set_variable = {
							name = mr_conquest_target_country
							value = prev
						}
						set_variable = {
							name = mr_conquest_target
							value = 1
						}
					}
				}
			}
		}
		# ========== AI RAILROAD: DECLARE ==========
		# Pacing follows the buff rule alone, PD-style (brandenburg_rise
		# ties pace to BRA_buff_enabled). The old timeline-pacing rule is
		# gone: Terminator declares roughly every 6 months, Historical
		# roughly yearly, Vanilla roughly every two years. Every tier has
		# a branch — with none, a buff-disabled campaign would simply
		# never declare its unification wars.
		if = {
			limit = {
				var:mr_conquest_target = 1
				OR = {
					AND = {
						has_game_rule = MR_buff_enabled
						var:mr_conquest_cooldown > 6
					}
					AND = {
						has_game_rule = MR_buff_historical
						var:mr_conquest_cooldown > 12
					}
					AND = {
						has_game_rule = MR_buff_disabled
						var:mr_conquest_cooldown > 24
					}
				}
				country_exists = c:MGO
			}
			c:MGO = {
				if = {
					limit = {
						OR = {
							AND = {
								has_game_rule = MR_buff_enabled
								months_since_war >= 6
							}
							AND = {
								has_game_rule = MR_buff_historical
								months_since_war >= 12
							}
							AND = {
								has_game_rule = MR_buff_disabled
								months_since_war >= 24
							}
						}
						situation:mongol_resurgence = {
							var:mr_conquest_target_country = {
								NOT = {
									top_overlord_or_this ?= prev
								}
								NOT = {
									has_truce_with = prev
								}
							}
						}
					}
					trigger_event_silently = mr_dominance.997
				}
			}
		}
		# --- FALLBACK: the chosen target stopped being valid ---
		# (annexed, became our subject, truced, or lost its Mongolian
		# holdings). Clear the slot so a new target can be picked.
		if = {
			limit = {
				var:mr_conquest_target = 1
				country_exists = c:MGO
				OR = {
					NOT = {
						country_exists = var:mr_conquest_target_country
					}
					var:mr_conquest_target_country = {
						OR = {
							AND = {
								country_exists = c:MGO
								top_overlord_or_this ?= c:MGO
							}
							NOT = {
								has_presence_in = scripted_geography:MR_geo_heartland
							}
						}
					}
				}
			}
			set_variable = {
				name = mr_conquest_target
				value = 0
			}
		}
		# --- Beat 1: Gathering the Horde (~2 years in) ---
		if = {
			limit = {
				var:mr_resurgence_momentum >= 24
				NOT = {
					has_global_variable = mr_beat_100_fired
				}
			}
			every_country = {
				limit = {
					government_type = government_type:steppe_horde
					culture = {
						has_culture_group = culture_group:mongolian_group
					}
				}
				trigger_event_non_silently = mr_dominance.100
			}
			set_global_variable = mr_beat_100_fired
		}
		# --- Beat 2: The Steppe Stirs (~5 years) ---
		if = {
			limit = {
				var:mr_resurgence_momentum >= 60
				NOT = {
					has_global_variable = mr_beat_101_fired
				}
			}
			every_country = {
				limit = {
					has_presence_in = scripted_geography:MR_geo_heartland
				}
				trigger_event_non_silently = mr_dominance.101
			}
			set_global_variable = mr_beat_101_fired
		}
		# --- Beat 3: The Shadow of the Khan (~9 years) ---
		if = {
			limit = {
				var:mr_resurgence_momentum >= 108
				NOT = {
					has_global_variable = mr_beat_102_fired
				}
			}
			every_country = {
				limit = {
					has_presence_in = scripted_geography:MR_geo_heartland
				}
				trigger_event_non_silently = mr_dominance.102
			}
			set_global_variable = mr_beat_102_fired
		}
		# --- Beat 4: The Thunder of Hooves, the branch (~13 years) ---
		if = {
			limit = {
				var:mr_resurgence_momentum >= 156
				NOT = {
					has_global_variable = mr_beat_103_fired
				}
			}
			every_country = {
				limit = {
					government_type = government_type:steppe_horde
					culture = {
						has_culture_group = culture_group:mongolian_group
					}
				}
				trigger_event_non_silently = mr_dominance.103
			}
			set_global_variable = mr_beat_103_fired
		}
		# --- Beat 5: A New Khan Rises, once MGO actually exists ---
		# mr_dominance.104's immediate also creates the Borjigin Great
		# Khan and puts him on the throne (Timur treatment).
		if = {
			limit = {
				country_exists = c:MGO
				NOT = {
					has_global_variable = mr_beat_104_fired
				}
			}
			c:MGO = {
				trigger_event_non_silently = mr_dominance.104
				# The claimant sheds any alliance with a power standing on
				# the heartland, so the railroad's declare event is not
				# blocked by its own treaties (PD_events.txt:1126 shape).
				# Fired here, not at on_start: MGO does not exist then.
				# AI ONLY, like the Phase 2/3 versions — a human player's
				# treaties are not dissolved without asking, and a human
				# claimant is not blocked by them in the first place.
				if = {
					limit = {
						is_ai = yes
					}
					trigger_event_non_silently = mr_dominance.28
				}
				# The war-preparation lock, PD-style: the AI claimant
				# must not freelance-declare wars — the railroad event
				# declares for it. Never applied to a human player.
				if = {
					limit = {
						is_ai = yes
					}
					add_country_modifier = {
						modifier = MR_mongol_preparing_for_conquest
						years = -1
						mode = replace
					}
				}
			}
			# The claim is real now — point the panel header at MGO and
			# seed the railroad's target variable with a country that is
			# guaranteed to exist (PD seeds c:BRA in on_start).
			set_variable = {
				name = mr_leading_country
				value = c:MGO
			}
			set_variable = {
				name = mr_conquest_target_country
				value = c:MGO
			}
			set_global_variable = mr_beat_104_fired
		}
		# --- Beat: Victory on the Steppe, when Karakorum falls to MGO ---
		if = {
			limit = {
				country_exists = c:MGO
				NOT = {
					has_global_variable = mr_beat_105_fired
				}
				c:MGO = {
					owns = location:karakorum
				}
			}
			c:MGO = {
				trigger_event_non_silently = mr_dominance.105
			}
			set_global_variable = mr_beat_105_fired
		}
		# --- Beat: Scattered to the Winds, a genuine reversal ---
		# Fires if the claimant loses Karakorum again after taking it.
		if = {
			limit = {
				has_global_variable = mr_beat_105_fired
				NOT = {
					has_global_variable = mr_beat_107_fired
				}
				OR = {
					NOT = {
						country_exists = c:MGO
					}
					AND = {
						country_exists = c:MGO
						c:MGO = {
							NOT = {
								owns = location:karakorum
							}
						}
					}
				}
			}
			every_country = {
				limit = {
					culture = {
						has_culture_group = culture_group:mongolian_group
					}
					government_type = government_type:steppe_horde
				}
				trigger_event_non_silently = mr_dominance.107
			}
			set_global_variable = mr_beat_107_fired
		}
		# --- ORGANIC BIRTH (design doc, flavor_tim.8 shape) ---
		# The dynamic path the design doc always intended: a free
		# Mongol steppe horde that takes Karakorum between 1368 and
		# 1375 takes up the Chinggisid claim without waiting for the
		# failsafe. form_country ignores MGO_f's own 85%-of-Mongolia
		# allow block (the same way PD forms PRU_f in
		# pd_brandenburg_dhe.4), so holding the sacred capital is
		# enough. AI converts on the spot; a HUMAN is OFFERED the
		# banner instead (mr_dominance.11) — this mod never converts
		# a player who has not said yes, the same philosophy as the
		# postponable war events and the is_ai-gated locks.
		if = {
			limit = {
				NOT = {
					country_exists = c:MGO
				}
				NOT = {
					has_global_variable = mr_birth_failsafe_fired
				}
			}
			random_country = {
				limit = {
					culture = {
						has_culture_group = culture_group:mongolian_group
					}
					government_type = government_type:steppe_horde
					owns = location:karakorum
					is_subject = no
					is_ai = yes
				}
				save_scope_as = mr_emergence_candidate
				form_country = formable_country:MGO_f
				trigger_event_silently = mr_dominance.998
			}
			random_country = {
				limit = {
					culture = {
						has_culture_group = culture_group:mongolian_group
					}
					government_type = government_type:steppe_horde
					owns = location:karakorum
					is_subject = no
					is_ai = no
					NOT = {
						has_global_variable = mr_organic_offer_made
					}
				}
				trigger_event_non_silently = mr_dominance.11
				set_global_variable = mr_organic_offer_made
			}
		}
		# --- BIRTH FAILSAFE (design doc, layer a) ---
		# If nobody has organically taken up the claim by ~1375,
		# convert the best-placed steppe power by force. Not gated by
		# the auto-conquest rule: without MGO the railroad has no actor
		# and every later phase is unreachable.
		#
		# Candidate quality tiers: the STRONGEST free AI horde at
		# peace, else the strongest free AI horde even at war, else
		# anyone at all. Tiers 1-2 are AI-only — a human who declined
		# the organic offer must not be conscripted while any AI
		# candidate exists; tier 3 remains the absolute last resort
		# (no AI mongol horde left on the steppe), where a vassal khan
		# is first released by its overlord (cancel_subject, the
		# overlord-run release effect: subject_types/readme.txt:50,
		# on_action/_hardcoded.txt:4808).
		if = {
			limit = {
				current_date > 1375.1.1
				NOT = {
					country_exists = c:MGO
				}
				NOT = {
					has_global_variable = mr_birth_failsafe_fired
				}
			}
			if = {
				limit = {
					any_country = {
						culture = {
							has_culture_group = culture_group:mongolian_group
						}
						government_type = government_type:steppe_horde
						has_presence_in = scripted_geography:MR_geo_heartland

						has_presence_in = scripted_geography:MR_geo_heartland
						is_subject = no

						is_ai = yes
					}
					order_by = military_strength
					max = 1
					check_range_bounds = no
					save_scope_as = mr_emergence_candidate
					form_country = formable_country:MGO_f

				}
			}
			else_if = {
				limit = {
					any_country = {
						culture = {
							has_culture_group = culture_group:mongolian_group
						}
						government_type = government_type:steppe_horde
						has_presence_in = scripted_geography:MR_geo_heartland
						is_subject = no
						is_ai = yes
					}
				}
				ordered_country = {
					limit = {
						culture = {
							has_culture_group = culture_group:mongolian_group
						}
						government_type = government_type:steppe_horde
						has_presence_in = scripted_geography:MR_geo_heartland
						is_subject = no
						is_ai = yes
					}
					order_by = military_strength
					max = 1
					check_range_bounds = no
					save_scope_as = mr_emergence_candidate
					form_country = formable_country:MGO_f
					trigger_event_silently = mr_dominance.998

			else_if = {

						}
						government_type = government_type:steppe_horde

		}
		# --- COMPLETION FAILSAFE (design doc, layer b) ---
		# Five years before the phase closes. This does NOT just buff the
		# AI and hope — it hands over the objective outright, which is
		# what the design document specifies and what PD actually does
		# (brandenburg_rise.txt: every_ownable_location_in_area +
		# change_location_owner).
		#
		# Gated on the conquest-automation game rule and on is_ai = yes:
		# a human player who fails the phase is allowed to fail it.
		# Locations are only taken from AI owners, never from the player.
		if = {
			limit = {
				has_game_rule = MR_mongol_resurgence_auto_conquest_yes
				current_date > 1415.1.1
				NOT = {
					mr_resurgence_end_trigger = yes
				}
				NOT = {
					has_global_variable = mr_failsafe_p1_fired
				}
				country_exists = c:MGO
				c:MGO = {
					is_ai = yes
					# at_war = no // Disabled because the AI might stuck in a endless war loop, which would prevent the failsafe from firing. The failsafe should fire regardless of war status.
				}
			}
			set_global_variable = mr_failsafe_p1_fired
			# 1. Hand MGO the Mongolian heartland outright.
			scripted_geography:MR_geo_heartland = {
				every_location_in_scripted_geography = {
					limit = {
						# owner ?= not owner =: an ownerless location made
						# the bare link error ("Event target link 'owner'
						# returned an invalid object").
						owner ?= {
							is_ai = yes
							NOT = {
								top_overlord_or_this ?= c:MGO
							}
						}
					}
					change_location_owner = c:MGO
					add_core = c:MGO
				}
			}
			# 2. Reduce the surviving rival hordes to tributaries.
			every_country = {
				limit = {
					is_ai = yes
					culture = {
						has_culture_group = culture_group:mongolian_group
					}
					government_type = government_type:steppe_horde
					NOT = {
						top_overlord_or_this ?= c:MGO
					}
					has_presence_in = scripted_geography:MR_geo_heartland
				}
				make_subject_of = {
					target = c:MGO
					type = subject_type:tributary
				}
			}
			c:MGO = {
				trigger_event_silently = mr_dominance.999
			}
		}
		# NOTE ON PLACEMENT: the two panel blocks sit at the END of
		# on_monthly. Everything that can change what they should show —
		# Beat 104 pointing mr_leading_country at the newborn MGO, the
		# birth failsafe creating it, the completion failsafe handing over
		# the heartland — runs below them in this same tick. Computed at
		# the top, the panel lagged a month behind its own situation; that
		# showed up in Phase 2 testing as a progress bar reading 0 while
		# every end requirement was already green.
		# ========== PANEL HEADER: RIVAL MUST NOT BE THE CLAIMANT ==========
		# mr_rival_country used to be picked once, in on_start, and never
		# looked at again. The failsafe then hands the banner to the
		# STRONGEST candidate — which is exactly the country on_start had
		# already saved as the rival — and Beat 104 points
		# mr_leading_country at the same country. Both header variables
		# then held one country and the panel drew MGO twice, once as
		# claimant and once as its own rival.
		#
		# So the rival is re-picked every month, with the current leader
		# and both claimant tags excluded. Same selection idiom as
		# on_start (war_of_religions.txt:54-59), same sub-continent cap.
		# The tag exclusions are belt and braces. They were REQUIRED when
		# this block ran at the top of on_monthly: Beat 104 pointed
		# mr_leading_country at the newborn MGO further down the same
		# tick, so for one month the leader variable still named the
		# pre-birth claimant and only the tag test kept MGO out of its
		# own rival slot. The block now runs last, after Beat 104 and
		# after the birth failsafe, so the leader exclusion alone would
		# do — the tag tests stay because they cost nothing and they
		# survive any future reordering.
		if = {
			limit = {
				country_exists = var:mr_leading_country
			}
			var:mr_leading_country = {
				save_scope_as = mr_monthly_leader
			}
			ordered_country = {
				limit = {
					culture = {
						has_culture_group = culture_group:mongolian_group
					}
					government_type = government_type:steppe_horde
					NOT = {
						this = scope:mr_monthly_leader
					}
					NOR = {
						tag = MGO
						tag = MGE
					}
					capital ?= {
						OR = {
							sub_continent = sub_continent:east_asia
							sub_continent = sub_continent:north_asia
						}
					}
				}
				order_by = military_strength
				max = 1
				check_range_bounds = no
				save_scope_as = mr_monthly_rival
			}
			if = {
				limit = {
					country_exists = scope:mr_monthly_rival
				}
				set_variable = {
					name = mr_rival_country
					value = scope:mr_monthly_rival
				}
			}
			# No eligible rival left on the board. The slot is cleared
			# ONLY if what it still holds is the claimant itself — an
			# empty header slot is better than a duplicated flag, but a
			# stale-yet-distinct rival is better than an empty one.
			else_if = {
				limit = {
					has_variable = mr_rival_country
					country_exists = var:mr_rival_country
					var:mr_rival_country = {
						this = scope:mr_monthly_leader
					}
				}
				remove_variable = mr_rival_country
				# The panel would otherwise keep drawing the departed
				# rival's last score next to an empty slot.
				set_variable = {
					name = MR_rival_score
					value = 0
				}
			}
		}
		# ========== PANEL SCORES (PD PD_bra_score / PD_teu_score) ==========
		# Strength scores for the two header countries: military strength
		# plus income plus a fifth of prestige. The .gui reads
		# MR_mgo_score / MR_rival_score off the situation scope.
		if = {
			limit = {
				country_exists = var:mr_leading_country
			}
			var:mr_leading_country = {
				save_scope_as = mr_leading_scope
			}
			if = {
				limit = {
					country_exists = scope:mr_leading_scope
				}
				set_variable = {
					name = MR_mgo_score
					value = scope:mr_leading_scope.military_strength
				}
				change_variable = {
					name = MR_mgo_score
					add = scope:mr_leading_scope.monthly_income_trade_and_tax
				}
				scope:mr_leading_scope = {
					set_variable = {
						name = mr_temp_prestige
						value = prestige
					}
					change_variable = {
						name = mr_temp_prestige
						multiply = 0.2
					}
				}
				change_variable = {
					name = MR_mgo_score
					add = scope:mr_leading_scope.var:mr_temp_prestige
				}
			}
		}
		if = {
			limit = {
				# has_variable first: the rival slot can now be CLEARED
				# above when the only candidate left was the claimant.
				has_variable = mr_rival_country
				country_exists = var:mr_rival_country
			}
			var:mr_rival_country = {
				save_scope_as = mr_rival_scope
			}
			if = {
				limit = {
					country_exists = scope:mr_rival_scope
				}
				set_variable = {
					name = MR_rival_score
					value = scope:mr_rival_scope.military_strength
				}
				change_variable = {
					name = MR_rival_score
					add = scope:mr_rival_scope.monthly_income_trade_and_tax
				}
				scope:mr_rival_scope = {
					set_variable = {
						name = mr_temp_prestige
						value = prestige
					}
					change_variable = {
						name = mr_temp_prestige
						multiply = 0.2
					}
				}
				change_variable = {
					name = MR_rival_score
					add = scope:mr_rival_scope.var:mr_temp_prestige
				}
			}
		}
	}
	on_ending = {

		# 2. Resolution: unified, or the steppe stays divided. Branch on

			# Direct set as well as in mr_dominance.2's immediate, so the

			set_global_variable = mr_railroad_failed
		}
	}
