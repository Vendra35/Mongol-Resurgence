# Mongol Railroad Mod - Situation: Road to Empire (Phase 2)

mongol_imperial_situation = {
	title = "situation_mongol_imperial_title"
	description = "situation_mongol_imperial_desc"

	trigger = {
		has_global_variable = mongol_imperial_active
		NOT = { has_global_variable = mongol_imperial_complete }
		NOT = { has_global_variable = mongol_railroad_failed }
	}

	visible = {
		mr_imperial_visible = yes
	}

	icon = gfx_interface_situation_mongol_imperial

	targets = {
		MGO = { country_score = 1 }
		MGE = { country_score = 2 }
	}

	sort_order = 11

	progress = { target = 100 }

	completion = {
		trigger = { mr_imperial_end_trigger = yes }
		effect = { set_global_variable = { name = mongol_imperial_complete } }
	}

	abort = {
		trigger = { has_global_variable = mongol_railroad_failed }
	}

	actions = {
		mongol_imperial_form_action = {
			trigger = {
				country_exists = c:MGO
				c:MGO = {
					controls = location:karakorum
					government_type = government_type:steppe_horde
					country_rank_level >= 3
				}
			}
			effect = { c:MGO = { form_country = formable_country:MGE_f } }
			ai_weight = 1000
		}
		mongol_imperial_cb_action = {
			trigger = {
				country_exists = c:MGO
				c:MGO = { government_type = government_type:steppe_horde }
			}
			effect = {
				c:MGO = {
					add_casus_belli = { type = mongol_imperial target = this }
				}
			}
			ai_weight = 500
		}
	}
}
