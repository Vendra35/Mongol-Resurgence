# =================================================================
# PHASE 2 - THE PAX MONGOLICA (1420-1550)
# =================================================================
# With the steppe unified, the horde turns outward: mastery of the
# Silk Road corridor and a contest for hegemony with the Ming. The
# goal is the red-lined regions themselves — Khorasan, Xinjiang and
# North China — cleared of every power outside the claimant's realm.
# The campaign beats come from the mr_imperial namespace; the
# phase-level beats from mr_dominance.120-127.
#
# AI railroad: same three-part machine as Phase 1, on the PD
# Ascension pattern (the_prussian_ascension.txt) — the AI claimant
# carries MR_mongol_preparing_for_conquest (blocked_from_declaring_
# war) and the situation picks its targets and declares its wars
# for it via mr_dominance.993 (PD-203 shape, so a human claimant
# can postpone).
# =================================================================
mongol_imperial = {
	monthly_spawn_chance = monthly_spawn_chance_unique
	can_start = {
		mr_can_start_imperial = yes
	}
	visible = {
		mr_imperial_visible = yes
	}
	can_end = {
		OR = {
			mr_imperial_end_trigger = yes
			# Time expiry. Without this the situation can only ever end in
			# success, which leaves on_ending's failure branch unreachable
			# and lets a failed phase run to the end of the campaign.
			custom_tooltip = {
				text = mr_imperial_timeout_tt
				current_date > 1550.1.1
			}
		}
	}
	hint_tag = hint_mongol_imperial
	on_start = {
		set_global_variable = mr_imperial_started
		set_variable = {
			name = mr_imperial_momentum
			value = 0
		}
		# --- AI RAILROAD BOOKKEEPING (PD ascension shape) ---
		# The cooldown is seeded nearly ripe (PD seeds 115 of 120) so
		# the first war comes about a year in. MGO is guaranteed to
		# exist here — can_start requires it — so seeding the target
		# variable with c:MGO is safe, unlike in Phase 1.
		set_variable = {
			name = mr_imp_conquest_cooldown
			value = 48
		}
		set_variable = {
			name = mr_imp_conquest_target
			value = 0
		}
		set_variable = {
			name = mr_imp_conquest_target_country
			value = c:MGO
		}
		# Values the .gui panel reads off the situation scope.
		if = {
			limit = {
				country_exists = c:MGO
			}
			set_variable = {
				name = mr_leading_country
				value = c:MGO
			}
		}
		# No mr_rival_country here, unlike Phase 1. This panel uses
		# one_country_header_template — mongol_imperial.gui overrides
		# CountryContext only, so a rival variable would be written to the
		# savefile every campaign and read by nothing. Phase 1 is the one
		# that needs it: two_countries_header_template, plus a monthly
		# recompute and a re-pick when the rival dies. A once-in-on_start
		# pick would be worse than nothing anyway — by mid-phase it points
		# at whatever country happened to hold north China in 1420.
		if = {
			limit = {
				country_exists = c:MGO
			}
			c:MGO = {
				# The Phase 1 REWARDS (MR_unified_mongol_banner,
				# MR_mongol_historical_modifier_2) are deliberately NOT
				# stripped here — they are permanent, and their tooltips
				# say so. Only the phase BUFFS are temporary, and those
				# are removed by Phase 1's own on_ending. Rewards do not
				# stack out of control because each phase grants exactly
				# one, and each phase's buff still expires with it.
				# The war-preparation lock, PD Ascension style: the AI
				# claimant must not freelance-declare — the railroad
				# event declares for it. Never applied to a human.
				if = {
					limit = {
						is_ai = yes
					}
					add_country_modifier = {
						modifier = MR_mongol_preparing_for_conquest
						years = -1
						mode = replace
					}
				}
				trigger_event_non_silently = mr_dominance.120
				# Phase buff per the buff rule, PD-style: Terminator gets
				# the full conquest engine, Historical the balanced
				# variant, Vanilla nothing.
				if = {
					limit = {
						has_game_rule = MR_buff_enabled
					}
					add_country_modifier = {
						modifier = MR_imperial_conquest
						years = -1
						mode = replace
					}
				}
				else_if = {
					limit = {
						has_game_rule = MR_buff_historical
					}
					add_country_modifier = {
						modifier = MR_imperial_historical_modifier
						years = -1
						mode = replace
					}
				}
				trigger_event_non_silently = mr_imperial.100
				every_country = {
					limit = {
						OR = {
							has_presence_in = scripted_geography:MR_geo_north_china
							has_presence_in = scripted_geography:MR_geo_xinjiang
							has_presence_in = scripted_geography:MR_geo_khorasan
							has_presence_in = scripted_geography:MR_geo_manchuria
							has_presence_in = scripted_geography:MR_geo_tibet
							has_presence_in = scripted_geography:MR_geo_siberian_marches
							has_presence_in = scripted_geography:MR_geo_persia
						}
						NOT = {
							tag = MGO
						}
					}
					save_scope_as = mr_silk_road_target
					c:MGO = {
						add_casus_belli = {
							target = scope:mr_silk_road_target
							type = casus_belli:cb_MR_silk_road
							# Covers the whole 1420-1550 phase window — the
							# old 50-year grant expired mid-phase.
							years = 130
						}
					}
				}
			}
		}
		# --- THE WORLD REACTS (PD pd_brandenburg.201/202/200 shape) ---
		# SPECTATORS: a stake in the theatre, no land the phase demands.
		every_country = {
			limit = {
				OR = {
					has_presence_in = sub_continent:east_asia
					has_presence_in = sub_continent:north_asia
					has_presence_in = sub_continent:central_asia
					has_presence_in = sub_continent:middle_east
					has_presence_in = sub_continent:south_asia
				}
				NOT = {
					OR = {
						has_presence_in = scripted_geography:MR_geo_khorasan
						has_presence_in = scripted_geography:MR_geo_xinjiang
						has_presence_in = scripted_geography:MR_geo_north_china
						has_presence_in = scripted_geography:MR_geo_manchuria
						has_presence_in = scripted_geography:MR_geo_tibet
						has_presence_in = scripted_geography:MR_geo_siberian_marches
					}
				}
				NOR = {
					tag = MGO
					tag = MGE
				}
			}
			trigger_event_non_silently = mr_dominance.22
		}
		# VICTIMS: they hold ground inside the phase's goal regions.
		every_country = {
			limit = {
				OR = {
					has_presence_in = scripted_geography:MR_geo_khorasan
					has_presence_in = scripted_geography:MR_geo_xinjiang
					has_presence_in = scripted_geography:MR_geo_north_china
					has_presence_in = scripted_geography:MR_geo_manchuria
					has_presence_in = scripted_geography:MR_geo_tibet
					has_presence_in = scripted_geography:MR_geo_siberian_marches
				}
				NOR = {
					tag = MGO
					tag = MGE
				}
			}
			trigger_event_non_silently = mr_dominance.23
		}
		# The claimant sheds alliances with those same powers, so the
		# railroad's declare event is not blocked by its own treaties.
		every_country = {
			limit = {
				tag = MGO
				# AI ONLY. PD fires its equivalent at c:PRU unconditionally
				# (the_prussian_ascension.txt:138), but PD has no standing
				# rule about it and this mod does: never force-convert, lock
				# or rob a human player. Dissolving a player's treaties with
				# no way to refuse is exactly that. It is also unnecessary
				# for them — the reason this exists is that the AI's declare
				# event stalls on its own alliances, and a human gets a
				# visible, postponable war event and can break alliances
				# themselves.
				is_ai = yes
			}
			trigger_event_non_silently = mr_dominance.24
		}
	}
	on_monthly = {
		change_variable = {
			name = mr_imperial_momentum
			add = 1
		}
		# --- READ-ONLY VANILLA HOOK: the Tumu Crisis (~1449) ---
		# Vanilla's flavor_chi_mon sets lost_emperor while the emperor
		# is captive. The reaction event used to be watched only by the
		# Chahar situation (1604-1634) and was therefore UNREACHABLE —
		# the crisis belongs to THIS phase's window (1420-1550), so the
		# watch lives here now; the Chahar copy stays as a fallback for
		# exotic runs. mr_steppe.2 is fire_only_once, so the two
		# watchers can never double-deliver.
		if = {
			limit = {
				mr_vanilla_tumu_crisis_active = yes
				NOT = {
					has_global_variable = mr_tumu_reaction_seen
				}
			}
			every_country = {
				limit = {
					culture = {
						has_culture_group = culture_group:mongolian_group
					}
					government_type = government_type:steppe_horde
				}
				trigger_event_non_silently = mr_steppe.2
			}
			set_global_variable = mr_tumu_reaction_seen
		}
		# ========== THE KHAN'S OWN PASTURE ==========
		# The ongoing half of the MR_select_core_region action. Vanilla
		# does this in on_location_changed_owner (_hardcoded.txt:4489),
		# but that on_action is defined with an effect block and
		# redefining the name in a mod would replace the vanilla block
		# outright — so the situation does it monthly instead. Anything
		# the claimant has taken inside its chosen region since last
		# month becomes core, and the Khan's death ends the privilege
		# exactly as it ends Timur's (_hardcoded.txt:3517).
		every_country = {
			limit = {
				OR = {
					tag = MGO
					tag = MGE
				}
				has_variable = mr_core_region
			}
			# Saved rather than reached with prev.prev: that construct has
			# ZERO uses anywhere in vanilla, and this file does not invent
			# syntax. save_scope_as + scope: is the attested way down.
			save_scope_as = mr_pasture_holder
			if = {
				limit = {
					has_ruler = yes
					ruler ?= {
						has_character_modifier = MR_great_khan
					}
				}
				var:mr_core_region = {
					# TWO passes, deliberately not one. Coring is a
					# one-way door, so it is limited to what is not core
					# yet; the modifier is NOT, so it is applied on its
					# own terms to every owned location that lacks it.
					# Folding them together (one limit, both effects) was
					# a real bug: the else branch below strips the
					# modifier from every location, and a single-pass
					# version could only re-add it to locations that were
					# still un-cored — which, after the first pass, is
					# none of them. The pasture modifier was lost for
					# good after any interregnum tick.
					every_location_in_region = {
						limit = {
							owner ?= scope:mr_pasture_holder
							NOT = {
								integration_level = core
							}
						}
						change_integration_level = core
					}
					every_location_in_region = {
						limit = {
							owner ?= scope:mr_pasture_holder
							NOT = {
								has_location_modifier = MR_khans_own_pasture
							}
						}
						add_location_modifier = {
							modifier = MR_khans_own_pasture
							years = -1
							mode = add_and_extend
						}
					}
				}
			}
			# No Great Khan on the throne: the pasture is just land again.
			else = {
				var:mr_core_region = {
					every_location_in_region = {
						limit = {
							has_location_modifier = MR_khans_own_pasture
						}
						remove_location_modifier = MR_khans_own_pasture
					}
				}
			}
		}
		# ========== AI RAILROAD: FIND A TARGET ==========
		# PD Ascension shape: pick the weakest neighbour holding land in
		# the goal corridor (khorasan / xinjiang / north_china) whose
		# alliance web is not overwhelming — a RATIO gate, see the long
		# note on the same block in MR_mongol_resurgence.txt. No is_ai gate —
		# human claimants get the war event with the postpone option.
		if = {
			limit = {
				var:mr_imp_conquest_target = 0
				country_exists = c:MGO
				c:MGO = {
					at_war = no
				}
			}
			c:MGO = {
				ordered_neighbor_country = {
					limit = {
						is_rebel_country = no
						NOT = {
							has_truce_with = prev
						}
						NOT = {
							top_overlord_or_this ?= prev
						}
						relative_defensive_alliance_strength = {
							target = c:MGO
							value < 1.5
						}
						OR = {
							has_presence_in = scripted_geography:MR_geo_khorasan
							has_presence_in = scripted_geography:MR_geo_xinjiang
							has_presence_in = scripted_geography:MR_geo_north_china
							has_presence_in = scripted_geography:MR_geo_manchuria
							has_presence_in = scripted_geography:MR_geo_tibet
							has_presence_in = scripted_geography:MR_geo_siberian_marches
						}
					}
					max = 1
					order_by = {
						value = num_locations
						multiply = -1
					}
					situation:mongol_imperial = {
						set_variable = {
							name = mr_imp_conquest_target_country
							value = prev
						}
						set_variable = {
							name = mr_imp_conquest_target
							value = 1
						}
					}
				}
			}
		}
		change_variable = {
			name = mr_imp_conquest_cooldown
			add = 1
		}
		# --- FALLBACK: the chosen target stopped being valid ---
		if = {
			limit = {
				var:mr_imp_conquest_target = 1
				country_exists = c:MGO
				OR = {
					NOT = {
						country_exists = var:mr_imp_conquest_target_country
					}
					var:mr_imp_conquest_target_country = {
						OR = {
							AND = {
								country_exists = c:MGO
								top_overlord_or_this ?= c:MGO
							}
							NOT = {
								OR = {
									has_presence_in = scripted_geography:MR_geo_khorasan
									has_presence_in = scripted_geography:MR_geo_xinjiang
									has_presence_in = scripted_geography:MR_geo_north_china
									has_presence_in = scripted_geography:MR_geo_manchuria
									has_presence_in = scripted_geography:MR_geo_tibet
									has_presence_in = scripted_geography:MR_geo_siberian_marches
								}
							}
						}
					}
				}
			}
			set_variable = {
				name = mr_imp_conquest_target
				value = 0
			}
		}
		# ========== AI RAILROAD: DECLARE ==========
		# PD Ascension principle: the later phases press harder than
		# the unification. Terminator roughly every 4-5 months,
		# otherwise every 8-9 (cooldown > 4 / > 8) — the CODE's numbers
		# are the spec (audit D2 ruling 2026-07-30: the user kept the
		# fast pace over the older written ~3-year figure).
		if = {
			limit = {
				var:mr_imp_conquest_target = 1
				OR = {
					AND = {
						has_game_rule = MR_buff_enabled
						var:mr_imp_conquest_cooldown > 4
					}
					AND = {
						NOT = {
							has_game_rule = MR_buff_enabled
						}
						var:mr_imp_conquest_cooldown > 8
					}
				}
				country_exists = c:MGO
			}
			c:MGO = {
				if = {
					limit = {
						OR = {
							AND = {
								has_game_rule = MR_buff_enabled
								months_since_war >= 4
							}
							AND = {
								NOT = {
									has_game_rule = MR_buff_enabled
								}
								months_since_war >= 8
							}
						}
						# NOT prev — two hops (c:MGO -> situation -> target),
						# so prev is the SITUATION here. See the long note on
						# the same block in MR_mongol_resurgence.txt.
						situation:mongol_imperial = {
							var:mr_imp_conquest_target_country = {
								country_exists = c:MGO
								NOT = {
									top_overlord_or_this ?= c:MGO
								}
								NOT = {
									has_truce_with = c:MGO
								}
							}
						}
					}
					trigger_event_silently = mr_dominance.993
				}
			}
		}
		# --- The Mongol Awakening (~4 years) ---
		if = {
			limit = {
				var:mr_imperial_momentum >= 48
				NOT = {
					has_global_variable = mr_beat_121_fired
				}
				country_exists = c:MGO
			}
			c:MGO = {
				trigger_event_non_silently = mr_dominance.121
			}
			set_global_variable = mr_beat_121_fired
		}
		# --- The First Wave: the campaign opens (~8 years) ---
		if = {
			limit = {
				var:mr_imperial_momentum >= 96
				NOT = {
					has_global_variable = mr_beat_101i_fired
				}
				country_exists = c:MGO
			}
			c:MGO = {
				trigger_event_non_silently = mr_imperial.101
			}
			set_global_variable = mr_beat_101i_fired
		}
		# --- The Horde at the Gates (~14 years) ---
		if = {
			limit = {
				var:mr_imperial_momentum >= 168
				NOT = {
					has_global_variable = mr_beat_122_fired
				}
			}
			every_country = {
				limit = {
					has_presence_in = scripted_geography:MR_geo_north_china
				}
				trigger_event_non_silently = mr_dominance.122
			}
			set_global_variable = mr_beat_122_fired
		}
		# --- The City Falls, once a great seat is actually taken ---
		if = {
			limit = {
				country_exists = c:MGO
				NOT = {
					has_global_variable = mr_beat_102i_fired
				}
				c:MGO = {
					OR = {
						owns = location:dadu
						owns = location:samarkand
					}
				}
			}
			c:MGO = {
				trigger_event_non_silently = mr_imperial.102
			}
			set_global_variable = mr_beat_102i_fired
		}
		# --- The Great Invasion: the branch (~22 years) ---
		if = {
			limit = {
				var:mr_imperial_momentum >= 264
				NOT = {
					has_global_variable = mr_beat_123_fired
				}
				country_exists = c:MGO
			}
			c:MGO = {
				trigger_event_non_silently = mr_dominance.123
			}
			set_global_variable = mr_beat_123_fired
		}
		# --- The Counter-Attack: the settled powers push back ---
		if = {
			limit = {
				var:mr_imperial_momentum >= 336
				NOT = {
					has_global_variable = mr_beat_103i_fired
				}
				country_exists = c:MGO
				c:MGO = {
					at_war = yes
				}
			}
			c:MGO = {
				trigger_event_non_silently = mr_imperial.103
			}
			set_global_variable = mr_beat_103i_fired
		}
		# --- The Empire Rises ---
		if = {
			limit = {
				var:mr_imperial_momentum >= 420
				NOT = {
					has_global_variable = mr_beat_124_fired
				}
				country_exists = c:MGO
			}
			c:MGO = {
				trigger_event_non_silently = mr_dominance.124
			}
			set_global_variable = mr_beat_124_fired
		}
		# --- COMPLETION FAILSAFE: five years before 1550 ---
		# Hands over the goal regions outright rather than buffing and
		# hoping. Gated on the imperial conquest-automation rule and on
		# is_ai; never takes a location from the player. The handover
		# covers exactly what the end trigger demands: the Khorasan,
		# Xinjiang and North China regions.
		if = {
			limit = {
				has_game_rule = MR_imperial_auto_conquest_yes
				current_date > 1545.1.1
				NOT = {
					mr_imperial_end_trigger = yes
				}
				NOT = {
					has_global_variable = mr_failsafe_p2_fired
				}
				country_exists = c:MGO
				c:MGO = {
					is_ai = yes
					at_war = no
				}
			}
			set_global_variable = mr_failsafe_p2_fired
			scripted_geography:MR_geo_heartland = {
				every_location_in_scripted_geography = {
					limit = {
						# owner ?= not owner =: an ownerless location made
						# the bare link error at :559 in testing.
						owner ?= {
							is_ai = yes
							NOT = {
								top_overlord_or_this ?= c:MGO
							}
						}
					}
					change_location_owner = c:MGO
					add_core = c:MGO
				}
			}
			scripted_geography:MR_geo_khorasan = {
				every_location_in_scripted_geography = {
					limit = {
						# owner ?= not owner =: an ownerless location made
						# the bare link error at :559 in testing.
						owner ?= {
							is_ai = yes
							NOT = {
								top_overlord_or_this ?= c:MGO
							}
						}
					}
					change_location_owner = c:MGO
					# add_core = c:MGO // The reason we don't grant cores immediately is that instantly coring those lands breaks historical accuracy, and it also makes the AI way too strong.
				}
			}
			scripted_geography:MR_geo_xinjiang = {
				every_location_in_scripted_geography = {
					limit = {
						# owner ?= not owner =: an ownerless location made
						# the bare link error at :559 in testing.
						owner ?= {
							is_ai = yes
							NOT = {
								top_overlord_or_this ?= c:MGO
							}
						}
					}
					change_location_owner = c:MGO
					add_core = c:MGO
				}
			}
			scripted_geography:MR_geo_north_china = {
				every_location_in_scripted_geography = {
					limit = {
						# owner ?= not owner =: an ownerless location made
						# the bare link error at :559 in testing.
						owner ?= {
							is_ai = yes
							NOT = {
								top_overlord_or_this ?= c:MGO
							}
						}
					}
					change_location_owner = c:MGO
					add_core = c:MGO
				}
			}
			scripted_geography:MR_geo_manchuria = {
				every_location_in_scripted_geography = {
					limit = {
						# owner ?= not owner =: the bare link errors on
						# ownerless locations.
						owner ?= {
							is_ai = yes
							NOT = {
								top_overlord_or_this ?= c:MGO
							}
						}
					}
					change_location_owner = c:MGO
					# add_core = c:MGO // The reason we don't grant cores immediately is that instantly coring those lands breaks historical accuracy, and it also makes the AI way too strong.
				}
			}

		# NOTE ON PLACEMENT: this block sits at the END of on_monthly, not
		# the start. The completion failsafe below-of-old ran AFTER the
		# score was written, so in the month the failsafe handed over the

	on_ending = {
		# 1. Strip the phase modifiers, PD-style. every_country over
		# the tag pair, not c:MGO — a human claimant can legally have
		# formed MGE mid-phase under the Vanilla buff rule.
		every_country = {
			limit = {
				OR = {
					tag = MGO
					tag = MGE
				}
			}
			remove_country_modifier = MR_imperial_conquest
			remove_country_modifier = MR_imperial_historical_modifier
			remove_country_modifier = MR_mongol_preparing_for_conquest
		}
		# 2. Branch on the goal (the trigger is dual-tag too).
		if = {
			limit = {
				mr_imperial_end_trigger = yes
			}
			# THE PROCLAMATION: the Yeke Mongol Ulus rises the moment
			# the Silk Road is mastered, not a century later.
			# form_country ignores MGE_f's own allow block (PD forms
			# PRU_f the same way in pd_brandenburg_dhe.4), and MGE_f's
			# form_effect raises the country to empire rank and grants
			# vanilla's restoration_of_the_mongol_empire modifier
			# (00_formable_countries.txt:3575).
			if = {
				limit = {
					NOT = {
						country_exists = c:MGE
					}
					country_exists = c:MGO
				}
				c:MGO = {
					form_country = formable_country:MGE_f
					c:MGE = {
						# MGE should be empire rank on birth, not a kingdom or duchy.
						if = {
							limit = {
								country_rank_level < 4
							}
							set_country_rank_effect = {
								rank = country_rank:rank_empire
							}
						}
						# The AI is set to aggressive for immersion. We use railroding, so personality does not matter, but the AI will still be more aggressive in its diplomacy and war behavior.
						set_personality = ai_personality:ai_aggressive
					}
				}
			}
			# THE HAND-OFF LIVES HERE, not in on_ended: the end trigger
			# requires country_exists = c:MGO, and the proclamation just
			# retired that tag — re-evaluating the trigger in on_ended
			# would silently return false and Phase 3 would never start.
			# This branch already proved the trigger true while MGO was
			# alive, so the global is set directly (the "on_ending sets
			# terminals directly" doctrine; mr_dominance.125 sets it
			# redundantly, like mr_dominance.1 does for Phase 1).
			set_global_variable = mr_phase_two_complete
			# The tag just changed, so the resolution events go to
			# whichever of the two the claimant now is — tag = is a
			# string comparison and never errors.
			every_country = {
				limit = {
					OR = {
						tag = MGE
						tag = MGO
					}
				}
				trigger_event_non_silently = mr_dominance.125
				trigger_event_non_silently = mr_imperial.200
			}
		}
		else_if = {
			limit = {
				any_country = {
					OR = {
						tag = MGO
						tag = MGE
					}
				}
			}
			# Dual-tag: a self-formed MGE that failed the corridor goal
			# still gets its failure resolution.
			every_country = {
				limit = {
					OR = {
						tag = MGO
						tag = MGE
					}
				}
				trigger_event_non_silently = mr_dominance.127
				trigger_event_non_silently = mr_imperial.210
			}
			# Phase 2 failure is terminal for the railroad — without this
			# the state machine hung with no resolution recorded.
			set_global_variable = mr_railroad_failed
		}
		else = {
			# Neither claimant tag exists at all — still record the
			# failure so the railroad terminates instead of hanging.
			set_global_variable = mr_railroad_failed
		}
	}
