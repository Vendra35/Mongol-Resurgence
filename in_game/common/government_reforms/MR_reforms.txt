# The Yassa Code — the unique reform mr_legacy_of_the_yassa unlocks.
# Gated exactly like vanilla's basic_timariot_system
# (country_specific.txt:1105-1119: unique + has_or_had_tag + has_advance),
# and it carries government_reform_slots = 1 in its own modifier so
# taking it never consumes a slot (vanilla does the same at :1099; the
# 1066 sister project proved the pattern in game). NO `government =`
# field ON PURPOSE — the timariot precedent — so the reform SURVIVES the
# horde's settle-to-monarchy transition, which is historical (the Yuan
# did exactly this in 1272).
#
# RULING REVISED 2026-07-30, after the first Great Partition game test.
# The transition was left unblocked that morning; the same evening it
# was measured doing real damage — an AI Khaghan that settles loses
# vanilla's whole steppe advance tree, horse archers included, and
# cannot fight Ming China in Phase 3. The AI is now held as a horde
# until mr_railroad_complete by
# generic_action_ai_lists/zz_MR_government_transition_addon.txt. A
# HUMAN claimant is still free to settle whenever they like, which is
# exactly why these two reforms must keep surviving the transition.
MR_yassa_code = {
	unique = yes
	potential = {
		OR = {
			has_or_had_tag = MGO
			has_or_had_tag = MGE
		}
		## has_advance = mr_legacy_of_the_yassa // Unlocked from advance not like this. See Mongol Resurgence\in_game\common\advances\MR_advances.txt
	}
	country_modifier = {
		army_light_cavalry_power = 0.1
		army_light_cavalry_maintenance_cost_modifier = -0.1
		land_morale_modifier = 0.05
		government_reform_slots = 1
	}
	years = 2
}
