# Mongol Railroad Mod - DHE Events (Dynamic Historical Events)
# flavor_mr.1 - flavor_mr.199
# =============================================
# DHE_1: The Last Stand of the Yuan (1335-1368)
# =============================================
country_event = {
	id = flavor_mr.1
	title = "dhe_mr.1.t"
	desc = "dhe_mr.1.d"
	picture = gfx_event_mongol_court
	type = country_event
	scope = country
	pre_trigger = {
		is_ai = no
	}
	trigger = {
		current_year >= 1335
		current_year <= 1368
		has_game_rule = {
			rule = rule_mongol_railroad
			setting = historical
		}
		tag = YUAN
		any_enemy_country = {
			controls = location:zhongdu
		}
	}
	mean_time_to_happen = {
		months = 36
	}
	immediate = {
		hidden_effect = {
			set_variable = {
				name = mr_dhe_1
				value = 1
			}
		}
	}
	option = {
		name = "dhe_mr.1.a"
		ai_chance = {
			factor = 100
		}
		custom_tooltip = "dhe_mr.1.a.tt"
	}
}
