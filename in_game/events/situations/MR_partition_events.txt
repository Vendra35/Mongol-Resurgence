# built yet; when they are, they attach here.

# away in on_ending, each at its own threshold.
# =================================================================

# =================================================================
# THE KURULTAI
# =================================================================
# Fired to the Khaghan every five years once cohesion has dropped
# below 85, from mr_great_partition's on_monthly.
#
# No option is free. That is the design: the historical Khaghans had
# exactly these three moves and every one of them cost something.
# Granting a pasture buys real cohesion but hands an ulus away for
# good; binding the ulus to the centre keeps the map whole and buys
# nothing, only unrest; paying the khan delays the question at the
# price of the treasury. Doing nothing is not on the list, because
# it was never available to them either.
#
# Every option carries ai_chance, so an AI Khaghan plays this too —
# this mod is usually watched in observer and a decision only a human
# could make would freeze the situation.
mr_partition.100 = {
	type = country_event
	category = situation_event
	title = mr_partition.100.title
	desc = mr_partition.100.desc
	outcome = neutral
	major = yes
	illustration_tags = {
		10 = interior
		10 = regular
	}
	# --- Grant the ulus its own pastures ---
	# Releases the FURTHEST ulus still held, in the same order the
	# collapse uses: Crimea first, Dzungaria last. Its weight leaves the
	# cohesion sum, but the concession is worth more than any single
	# ulus, so the net is a realm that is smaller and steadier.
	option = {
		name = mr_partition.100.a
		ai_chance = {
			factor = 40
		}
		# WHAT THE ASSEMBLY PAYS FOR, rewritten 2026-07-30. It used to pay in
		# cohesion, which is now impossible on purpose: a gauge that can be
		# bought back while the map stays partitioned is a gauge that lies. It
		# also made this option strictly bad once the buy-back was capped -
		# permanent ground out, a few points of a number in.
		#
		# So the payment is the thing letting an ulus go PEACEFULLY actually
		# buys: the assembly is satisfied and the realm that remains is
		# steadier. That is also what happened - the uluses that separated by
		# agreement cost the centre far less than the ones it tried to hold.
		# Guarded on an ulus actually being there to give away, and the guard
		# runs FIRST because the release makes its own condition false.
		if = {
			limit = {
				OR = {
					mr_ulus_crimea_held = yes
					mr_ulus_nogai_held = yes
					mr_ulus_kazakh_held = yes
					mr_ulus_bashkiria_held = yes
					mr_ulus_tarim_held = yes
					mr_ulus_dzungaria_held = yes
				}
			}
			add_stability = stability_mild_bonus
			add_legitimacy = legitimacy_mild_bonus
		}
		# AUDIT 2026-07-30: these six blocks used to carry their OWN copy of
		# the handover - change_location_owner straight to the tag, and
		# nothing else. They were written before the partition had a
		# schedule, and once it did the two paths disagreed in four ways:
		# the kurultai never set the mr_returned_* flags, it limited on
		# mr_in_claimant_realm instead of mr_partition_takeable so it could
		# take a HUMAN vassal ground, it gave no cores, no grace modifier
		# and no rank, and it knew only six of the fourteen theatres.
		#
		# Now it calls the same effects the schedule calls. One handover,
		# one behaviour, one set of flags - and the kurultai keeps its own
		# job, which is choosing WHICH ulus goes and paying for it.
		# The chain is ordered by distance from Karakorum, the same order
		# the collapse uses: the furthest ulus still held is the one the
		# assembly is given.
		if = {
			limit = {
				mr_ulus_crimea_held = yes
			}
			mr_return_crimea = yes
		}
		else_if = {
			limit = {
				mr_ulus_nogai_held = yes
			}
			mr_return_nogai = yes
		}
		else_if = {
			limit = {
				mr_ulus_kazakh_held = yes
			}
			mr_return_kazakh = yes
		}
		else_if = {
			limit = {
				mr_ulus_bashkiria_held = yes
			}
			mr_return_bashkiria = yes
		}
		else_if = {
			limit = {
				mr_ulus_tarim_held = yes
			}
			mr_return_tarim = yes
		}
		else_if = {
			limit = {
				mr_ulus_dzungaria_held = yes
			}
			mr_return_dzungaria = yes
		}
	}
	# --- Bind them to the centre ---
	# The map stays whole and nothing is bought. AUDIT 2026-07-30: this
	# comment used to say the realm "pays for the refusal in unrest",
	# and MR_kurultai_defied was written to match - which broke the
	# mod's own hardest rule, on an option the AI takes 40% of the time.
	# The refusal is paid in the cabinet, in control, in legitimacy and
	# horde unity, and above all in the situation's own gauge: cohesion
	# now falls FASTER than it was falling, so more uluses clear their
	# thresholds when the empire finally comes apart. Same pressure,
	# expressed where the situation can actually resolve it.
	option = {
		name = mr_partition.100.b
		ai_chance = {
			factor = 40
		}
		add_country_modifier = {
			modifier = MR_kurultai_defied
			years = 25
			mode = add_and_extend
		}
		add_horde_unity = horde_unity_mild_penalty
		# The cost is the modifier above and nothing else now. It used to also
		# push cohesion down through the concessions term; that term is gone,
		# and the schedule was always the thing actually taking the map apart.
		# Refusing the assembly buys exactly what it says: the map stays whole
		# this year, and the centre pays for it in order and legitimacy.
	}
	# --- Buy the khan's loyalty ---
	# Delays the question rather than answering it: a smaller concession
	# than autonomy, and the treasury pays. add_gold takes a raw number;
	# there are no named script values for money (vanilla passes raw too,
	# qa_debug.txt:516).
	option = {
		name = mr_partition.100.c
		ai_chance = {
			factor = 20
		}
		add_gold = -500
		# And the khan is quiet for twice as long. The cooldown counts up to 60
		# in the situation on_monthly and is reset to 0 when the kurultai
		# fires; seeding it at -60 instead buys ten years before the assembly
		# asks again rather than five. This is what paying a khan actually got
		# you, and unlike the old +10 to a score it is something the Khaghan
		# can feel.
		situation:mr_great_partition = {
			set_variable = {
				name = mr_partition_kurultai_cooldown
				value = -60
			}
		}
	}
}
