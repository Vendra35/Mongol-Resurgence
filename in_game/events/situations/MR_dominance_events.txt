# MONGOL RESURGENCE - SITUATION LIFECYCLE
# =================================================================

# by a situation (on_start / on_monthly / on_ended) or by another
# event, via trigger_event_non_silently — never by a date window.
# They therefore carry no dynamic_historical_event block, and no
# trigger of their own beyond option-level gating.
#

# the real EU5 field set before these are actually called. Until
# then these events are defined but unreferenced.

#   120-125  Phase 2 beats, 127 the Phase 2 setback

#            133 the Phase 3 collapse

				}
			}
			scope:mr_break_claimant = {
				remove_relation = {
					type = relation_type:alliance
					first = prev

			}
		}

# =========================== PHASE 1 =============================
# 1368-1420: the unification of Mongolia.
# --- The Banner Offered ------------------------------------------
# The HUMAN half of the organic birth: an AI horde that takes
# Karakorum converts on the spot, but a player is asked first —
# the same never-force-a-player philosophy as the postponable war
# events. Declining passes the mantle: the 1375 failsafe then
# prefers AI candidates. (A player can still form MGO manually via
# the formable later.)
mr_dominance.11 = {
	type = country_event
	category = situation_event
	title = mr_dominance.11.title
	desc = mr_dominance.11.desc
	outcome = positive
	fire_only_once = yes
	major = yes
	illustration_tags = {
		10 = interior
		10 = happy
	}
	option = {
		name = mr_dominance.11.a
		ai_chance = {
			factor = 100
		}
		form_country = formable_country:MGO_f
		# MGO should be kingdom rank on birth, not a duchy or county.
		c:MGO = {
			if = {
				limit = {
					country_rank_level < 3
				}
				set_country_rank_effect = {
					rank = country_rank:rank_kingdom
				}
			}
			# The AI is set to aggressive for immersion. We use railroding, so personality does not matter, but the AI will still be more aggressive in its diplomacy and war behavior.
			set_personality = ai_personality:ai_aggressive
		}
		# Records that the birth is handled, so neither the organic
		# block nor the 1375 failsafe fires again.
		trigger_event_silently = mr_dominance.998
	}
	option = {
		name = mr_dominance.11.b
		ai_chance = {
			factor = 0
		}
		# The mantle passes. Nothing else — the failsafe will find
		# another khan in 1375.
	}
}

# --- A New Khan Rises -------------------------------------------
# Fires for MGO the month it first exists. Besides the beat, this
# is where the Great Khan himself is created: a new Borjigin ruler
# with the Timur treatment (rise_of_timur.txt gives Timur the
# timmy_wants_to_play character modifier and the conqueror trait;
# character creation fields follow pirate_events.txt:35 /
# banking.txt:58 / election.txt:165).
mr_dominance.104 = {
	type = country_event
	category = situation_event
	title = mr_dominance.104.title
	desc = mr_dominance.104.desc
	outcome = positive
	fire_only_once = yes
	illustration_tags = {
		10 = interior
		10 = happy
	}
	immediate = {
		hidden_effect = {
			create_character = {
				estate = estate_type:nobles_estate
				first_name = Batu
				age = 20
				culture = root.culture
				religion = root.religion
				dynasty = dynasty:borjigin_dynasty
				adm = {
					80
					100
				}
				dip = {
					60
					100
				}
				mil = {
					90
					100
				}
				save_scope_as = mr_great_khan
				create_in_limbo = yes
			}
			set_new_ruler = scope:mr_great_khan
			scope:mr_great_khan = {
				add_character_modifier = {
					modifier = MR_great_khan
					years = -1
					mode = add_and_extend
				}
				add_character_modifier = {
					modifier = MR_historically_needed
					years = 70
				}
				# FIRST KHAN ONLY — the second and third generations
				# (mr_dominance.120/.130) deliberately do NOT get this.
				# See the note on the modifier itself.
				add_character_modifier = {
					modifier = MR_chinggisid_blitz
					years = -1
					mode = add_and_extend
				}
				add_trait = trait:conqueror
				add_trait = trait:tactical_genius
				add_trait = trait:strategist
				add_trait = trait:cruel
			}
		}
	}
	option = {
		name = mr_dominance.104.a
		ai_chance = {
			factor = 100
		}
		add_prestige = prestige_severe_bonus
	}
}

		}

		## which is set to 10 for Steppe Horse Archers and 5 for Auxiliary units and Cavalry spawn units. The create_sub_unit_with_owner command creates a sub-unit of the specified type and assigns it to the AI's capital, with the AI as the owner.
		## For Steppe Horse Archers units

				has_advance = horse_lords
			}

				while = {
					count = 10

						type = a_steppe_horse_archers
						owner = root
					}
				}
			}
		}

		if = {
			limit = {
				has_advance = horse_lords
			}
			capital = {
				while = {
					count = 5
					create_sub_unit_with_owner = {
						type = a_a_urughs
						owner = root
					}
				}
			}
		}
		## For Cavalry spawn units Age of Revolutions(Age 6) to Age of Reformation(Age 4). 
		## We don't include Age 2 and Age 3 because Steppe Horse Archers units are so much better than them even tho they (Steppe Horse Archers) are belong to Age 1 (Age of Traditions) units.

			limit = {
				has_advance = unlock_light_dragoons_advance

					count = 5

					}
				}
			}

			limit = {
				has_advance = unlock_pistoleers_advance
			}

				while = {
					count = 5
					create_sub_unit_with_owner = {
