namespace = mr_history
# =================================================================
# MONGOL RESURGENCE - HISTORICAL DHE LAYER
# =================================================================
# The real 14th-15th century backdrop the divergence grows out of:
# the Yuan expulsion, the fragmentation of the successor khanates,
# and Timur's interlude. These fire on the historical record; the
# mr_dominance / mr_imperial layers carry the divergence itself.
#
# Firing is handled by dynamic_historical_event (tag + date window +
# monthly_chance), which is the vanilla idiom for a historically
# anchored event. Reference: pirate_events.txt, flavor_tim.8.
#
# Tag availability verified against the 1337 start, NOT merely
# against setup/countries — a tag can be defined there and still hold
# no land on the map:
#
#   ON THE MAP AT 1337:  CHI (Great Yuan)  CHG (Chagatai Horde)
#                        GLH (Golden Horde)  DLH (Delhi)
#                        CHB MZF INJ GRG JLY (Ilkhanid successors)
#   EMERGE LATER:        TIM  via flavor_tim.8 (create_country_from_
#                             cores_in_our_locations, ~1360-1400)
#                        OIR  via flavor_chi.txt, out of the Yuan collapse
#   NOT USABLE:          HLG — defined but has ZERO script uses in all
#                        of vanilla. The Ilkhanate broke up in 1335,
#                        two years before the game begins, so there is
#                        no Ilkhanate to fragment. Events keyed to HLG
#                        can never fire.
#
# Events targeting TIM/OIR are conditional by design: they fire only
# if vanilla spawned those countries, which is the historical path.
# =================================================================
# --- The Last Stand of the Yuan (1351-1368) ----------------------
# Dated to the outbreak of the Red Turban Rebellion (1351), not the
# 1337 start — the event's own text is about Red Turban banners, so
# firing it before the rebellion existed read as nonsense in game.
# monthly_chance = 100: fires the month the window opens (percent
# semantics, 80 vanilla uses at 100).
mr_history.1 = {
	type = country_event
	title = mr_history.1.title
	desc = mr_history.1.desc
	historical_info = mr_history.1.historical_info
	outcome = negative
	fire_only_once = yes
	illustration_tags = {
		10 = armed
		10 = exterior
	}
	dynamic_historical_event = {
		tag = CHI
		from = 1351.6.1
		to = 1368.1.1
		monthly_chance = 100
	}
	trigger = {
		NOT = {
			has_game_rule = mr_railroad_off
		}
		owns = location:dadu
	}
	option = {
		name = mr_history.1.a
		historical_option = yes
		ai_chance = {
			factor = 70
		}
		# Accept that the Mandate is slipping: the dynasty turns its
		# eyes back to the steppe it came from.
		add_prestige = prestige_mild_penalty
		add_stability = stability_weak_penalty
	}
	option = {
		name = mr_history.1.b
		ai_chance = {
			factor = 30
		}
		# Fight for every province — costly, but the army remembers.
		add_army_tradition = army_tradition_mild_bonus
		add_stability = stability_weak_penalty
	}
}

# --- The Chagatai Schism (1347-1365) -----------------------------
# The east/west split that eventually produces Moghulistan and,
# through it, Timur's launching ground. Dated to the historical
# break: Tughlugh Timur in the east and Qazaghan's emirs in the
# west, c. 1347.
mr_history.3 = {
	type = country_event
	title = mr_history.3.title
	desc = mr_history.3.desc
	historical_info = mr_history.3.historical_info
	outcome = neutral
	fire_only_once = yes
	illustration_tags = {
		10 = interior
		10 = regular
	}
	dynamic_historical_event = {
		tag = CHG
		from = 1347.1.1
		to = 1365.1.1
		monthly_chance = 100
	}
	trigger = {
		NOT = {
			has_game_rule = mr_railroad_off
		}
	}
	# The historical schism was as much religious as territorial:
	# the western half Islamised, the eastern half held to the steppe cults.
	option = {
		name = mr_history.3.a
		historical_option = yes
		ai_chance = {
			factor = 50
		}
		change_religion = religion:sunni
		add_legitimacy = legitimacy_mild_bonus
	}
	option = {
		name = mr_history.3.b
		ai_chance = {
			factor = 50
		}
		change_religion = religion:tengri
		add_horde_unity = horde_unity_mild_bonus
	}
}

# --- The Looting of Delhi (1398) ---------------------------------
# Dated to the sack itself (December 1398). Both sides feel it: the
# Timurid gets the plunder, and Delhi gets mr_history.9 — the ruin.
mr_history.6 = {
	type = country_event
	title = mr_history.6.title
	desc = mr_history.6.desc
	historical_info = mr_history.6.historical_info
	outcome = positive
	fire_only_once = yes
	illustration_tags = {
		10 = armed
		10 = exterior
	}
	dynamic_historical_event = {
		tag = TIM
		from = 1398.10.1
		to = 1404.1.1
		monthly_chance = 100
	}
	trigger = {
		NOT = {
			has_game_rule = mr_railroad_off
		}
		owns = location:samarkand
		country_exists = c:DLH
	}
	option = {
		name = mr_history.6.a
		historical_option = yes
		ai_chance = {
			factor = 100
		}
		add_prestige = prestige_severe_bonus
		# The wealth of the richest city in the world rides north.
		add_gold = 5000
		if = {
			limit = {
				country_exists = c:DLH
			}
			c:DLH = {
				trigger_event_non_silently = mr_history.9
			}
		}
	}
}
