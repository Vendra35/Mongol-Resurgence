namespace = mr_history

# =================================================================
# MONGOL RESURGENCE - HISTORICAL DHE LAYER
# =================================================================
# The real 14th-15th century backdrop the divergence grows out of:
# the Yuan expulsion, the fragmentation of the successor khanates,
# and Timur's interlude. These fire on the historical record; the
# mr_dominance / mr_imperial layers carry the divergence itself.
#
# Firing is handled by dynamic_historical_event (tag + date window +
# monthly_chance), which is the vanilla idiom for a historically
# anchored event. Reference: pirate_events.txt, flavor_tim.8.
#
# Tags verified against vanilla setup/countries:
#   CHI Yuan/China   HLG Hulaguids (Ilkhanate)   CHG Chagatai
#   GLH Golden Horde   OIR Oirat   TIM Timurids   DLH Delhi
# =================================================================


# --- The Last Stand of the Yuan (1335-1368) ----------------------
mr_history.1 = {
	type = country_event
	title = mr_history.1.title
	desc = mr_history.1.desc
	historical_info = mr_history.1.historical_info
	outcome = negative
	fire_only_once = yes
	illustration_tags = {
		10 = armed
		10 = exterior
	}

	dynamic_historical_event = {
		tag = CHI
		from = 1335.1.1
		to = 1368.1.1
		monthly_chance = 1
	}

	trigger = {
		has_game_rule = mr_railroad_historical
		owns = location:dadu
	}

	immediate = {
		hidden_effect = {
			set_variable = {
				name = mr_history_yuan_decline
				value = 1
			}
		}
	}

	option = {
		name = mr_history.1.a
		historical_option = yes
		ai_chance = { factor = 100 }
	}
}

# --- The Chagatai Schism (1334-1368) -----------------------------
# The east/west split that eventually produces Moghulistan and,
# through it, Timur's launching ground.
mr_history.3 = {
	type = country_event
	title = mr_history.3.title
	desc = mr_history.3.desc
	historical_info = mr_history.3.historical_info
	outcome = neutral
	fire_only_once = yes
	illustration_tags = {
		10 = interior
		10 = regular
	}

	dynamic_historical_event = {
		tag = CHG
		from = 1334.1.1
		to = 1368.1.1
		monthly_chance = 1
	}

	trigger = {
		has_game_rule = mr_railroad_historical
	}

	immediate = {
		hidden_effect = {
			set_variable = {
				name = mr_history_chagatai_schism
				value = 1
			}
		}
	}

	# The historical schism was as much religious as territorial:
	# the western half Islamised, the eastern half held to the steppe cults.
	option = {
		name = mr_history.3.a
		historical_option = yes
		ai_chance = { factor = 50 }
		change_religion = religion:sunni
	}

	option = {
		name = mr_history.3.b
		ai_chance = { factor = 50 }
		change_religion = religion:tengri
	}
}
