# =================================================================
# MONGOL RESURGENCE - HORDE MECHANICS DHE LAYER
# =================================================================
# The institutions a reunified horde builds as it turns from a
# tribal confederation into an empire: drilled armies, siege craft,
# the Kurultai, the Silk Road, the Yam relay — plus the two heavy
# AI catch-up events (9-10) mongolized from PD's tested
# pd_brandenburg_dhe.1/2 templates.
#
# Localisation lives in MR_l_english.yml (mr_dominance_dhe.1-12,
# including the .entry keys used by the DHE timeline).
#
# The heavy AI catch-up events are AI-only per the .entry text; the
# Kurultai (4), Observatory (11) and Census (12) reach players too.
# Each event grants its OWN reward — the phase buffs
# (MR_mongol_warrior_spirit etc.) are granted by the situations
# under the buff rule alone, never re-granted here.
# Phase assignment follows the three-situation timeline (P2-window
# events also carry tag = MGE, in case of an early proclamation):
#   Phase 1 (1368-1420, MGO): 1 Forge, 4 Kurultai, 5 Absorption, 9 Tumen, 12 Census
#   Phase 2 (1420-1550, MGO/MGE): 3 Wall Breakers, 6 Silk Road, 2 Encampment, 10 Karakorum, 11 Observatory
#   Phase 3 (1550-1650, MGE): 7 Melting Pot, 8 Yam
# =================================================================
# --- The Forge of Warriors --------------------------------------

# --- The Reforged Tumen (AI catch-up, Phase 1) -------------------
# Mongolized from PD's pd_brandenburg_dhe.1 (the tested template all
# constructs here come from): the AI claimant that holds Karakorum
# gets the decimal tumen system rebuilt for it — the steppe advance,
# free military buildings at the sacred capital, horse archers in
# the field, and the camps turned Mongol.
mr_dominance_dhe.9 = {
	type = country_event
	title = mr_dominance_dhe.9.title
	desc = mr_dominance_dhe.9.desc
	outcome = positive
	fire_only_once = yes
	illustration_tags = {
		10 = armed
		10 = exterior
	}
	dynamic_historical_event = {
		# tag = MGE as well: the proclamation can land before 1480.
		tag = MGO
		tag = MGE
		from = 1368.1.1
		to = 1480.1.1
		monthly_chance = 100
	}
	trigger = {
		NOT = {
			has_game_rule = mr_railroad_off
		}
		is_ai = yes
	}
	option = {
		name = mr_dominance_dhe.9.a
		ai_chance = {
			factor = 100
		}
		# If the country is not already Mongolian culture, this will make it so.
		if = {
			limit = {
				NOT = {
					culture = culture:mongolian_culture
				}
			}
			change_culture = culture:mongolian_culture
		}
		# If the country does not already have Oirat culture accepted, this will make it so.
		if = {
			limit = {
				NOT = {
					has_primary_or_accepted_culture = culture:oirat_culture
				}
			}
			add_accepted_culture = culture:oirat_culture
		}
		# If the country does not already have Daur culture accepted, this will make it so.
		if = {
			limit = {
				NOT = {
					has_primary_or_accepted_culture = culture:daur_culture
				}
			}
			add_accepted_culture = culture:daur_culture
		}
		# If the country does not already have Tumed culture accepted, this will make it so.
		if = {
			limit = {
				NOT = {
					has_primary_or_accepted_culture = culture:tumed_culture
				}
			}
			add_accepted_culture = culture:tumed_culture
		}
		# If the country does not already have Buryat culture accepted, this will make it so.
		if = {
			limit = {
				NOT = {
					has_primary_or_accepted_culture = culture:buryat_culture
				}
			}
			add_accepted_culture = culture:buryat_culture
		}
		if = {
			limit = {
				NOT = {
					has_advance = horse_lords
				}
			}
			custom_tooltip = {
				text = mr_dominance_dhe.9.a.tt1
				research_advance = advance_type:horse_lords
			}
		}
		custom_tooltip = mr_dominance_dhe.9.a.tt2
		custom_tooltip = {
			text = mr_dominance_dhe.9.a.tt3
			# Instead of checking location:karakorum, we check MGO's capital, because the MGO country may not own Karakorum on the map when it is born, but it still needs to get these troops for wars to succeed.
			capital ?= {
				while = {
					count = 5
					construct_building = {
						building_type = building_type:kurultai
						cost_multiplier = 0
						cost_multiplier_reason = "MR_dhe_free_buildings"
						instant = yes
					}
					construct_building = {
						building_type = building_type:armory
						cost_multiplier = 0
						cost_multiplier_reason = "MR_dhe_free_buildings"
						instant = yes
					}
				}
				while = {
					count = 20
					create_sub_unit_with_owner = {
						# owner = root, not c:MGO: this event can fire
						# under the MGE tag after an early proclamation,
						# and a dead c:MGO link would break the spawn
						# (root is the firing country either way — the
						# same shape as mr_dominance.996).
						type = a_steppe_horse_archers
						owner = root
					}
				}
				every_pop = {
					limit = {
						culture != culture:mongolian_culture
					}
					split_pop = {
						fraction = 0.9
						culture = culture:mongolian_culture
					}
				}
			}
		}
		custom_tooltip = mr_dominance_dhe.9.a.tt4
	}
}

# --- Karakorum Restored (AI catch-up, Phase 2) -------------------
# Mongolized from PD's pd_brandenburg_dhe.2: Ogedei's capital rises
# again — city rank, walls, barracks, a market on the caravan road,
# and the horde's kin peoples accepted at court.
mr_dominance_dhe.10 = {
	type = country_event
	title = mr_dominance_dhe.10.title
	desc = mr_dominance_dhe.10.desc
	outcome = positive
	fire_only_once = yes
	illustration_tags = {
		10 = interior
		10 = regular
	}
	dynamic_historical_event = {
		# tag = MGE as well: the proclamation can land before this
		# window closes, and an unfired event must survive the switch.
		tag = MGO
		tag = MGE
		from = 1420.1.1
		to = 1550.1.1
		monthly_chance = 100
	}
	trigger = {
		NOT = {
			has_game_rule = mr_railroad_off
		}
		is_ai = yes
		owns = location:karakorum
	}
	option = {
		name = mr_dominance_dhe.10.a
		ai_chance = {
			factor = 100
		}
		# If the capital is not Karakorum, we can set it to Karakorum (if possible).
		if = {
			limit = {
				capital != location:karakorum
			}
			set_capital = location:karakorum
		}
		if = {
			limit = {
				NOT = {
					has_advance = barracks_advance
				}
			}
			custom_tooltip = {
				text = mr_dominance_dhe.10.a.tt1
				research_advance = advance_type:barracks_advance
			}
		}
		custom_tooltip = mr_dominance_dhe.10.a.tt2
		custom_tooltip = {
			text = mr_dominance_dhe.10.a.tt3
			location:karakorum = {
				change_location_rank = location_rank:city
				while = {
					count = 5
					construct_building = {
						building_type = building_type:barracks
						cost_multiplier = 0
						cost_multiplier_reason = "MR_dhe_free_buildings"
						instant = yes
					}
					construct_building = {
						building_type = building_type:armory
						cost_multiplier = 0
						cost_multiplier_reason = "MR_dhe_free_buildings"
						instant = yes
					}
				}
				construct_building = {
					building_type = building_type:castle
					cost_multiplier = 0
					cost_multiplier_reason = "MR_dhe_free_buildings"
					instant = yes
				}
				while = {
					count = 40
					create_sub_unit_with_owner = {
						# owner = root, not c:MGO — see mr_dominance_dhe.9.
						type = a_steppe_horse_archers
						owner = root
					}
				}
				every_pop = {
					limit = {
						culture != culture:mongolian_culture
					}
					split_pop = {
						fraction = 0.9
						culture = culture:mongolian_culture
					}
				}
			}
		}
		# A market on the caravan road. No owner check needed — the
		# event's own trigger just proved root owns Karakorum this
		# tick (and a c:MGO comparison would break under the MGE tag).
		if = {
			limit = {
				location:karakorum = {
					is_market_center = no
				}
			}
			create_market = {
				builder = root
				location = location:karakorum
				instant = yes
				price = price:create_market
			}
		}
		custom_tooltip = mr_dominance_dhe.10.a.tt4
	}
}
