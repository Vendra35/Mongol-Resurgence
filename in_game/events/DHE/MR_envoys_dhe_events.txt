######################################
# 1 — The Ming Embassy
######################################

mr_envoys.1 = {
	type = country_event
	title = mr_envoys.1.title
	desc = mr_envoys.1.desc
	historical_info = mr_envoys.1.historical_info
	outcome = neutral
	fire_only_once = yes

	illustration_tags = {
		10 = regular
		10 = interior
	}

	dynamic_historical_event = {
		tag = MGO
		from = 1372.1.1
		to = 1400.1.1
		monthly_chance = 8
	}

	# Fires while the horde court still stands to receive an embassy —
	# the settle-exposure map's cheap thematic gate. No railroad-rule
	# gate: story flavor plays under every rule setting.
	trigger = {
		# The master switch gates this event like every other piece of MR
		# content. It used to be omitted here on the reasoning that story
		# flavour should play under any rule setting - but MGO can exist
		# with the switch OFF, because vanilla ships its own MGO_f formable
		# that a player can form by hand, so these events fired for someone
		# who had explicitly turned this mod's content off. CLAUDE.md also
		# states that all content checks this rule, and that statement is
		# only true once these are gated.
		NOT = {
			has_game_rule = mr_railroad_off
		}
		government_type = government_type:steppe_horde
	}

	option = {
		# Defiance — the historical Northern Yuan line with early Ming.
		# The answer rides south in every saddle for a decade.
		name = mr_envoys.1.a
		historical_option = yes
		add_prestige = prestige_mild_bonus
		add_country_modifier = {
			modifier = MR_defiance_of_the_south
			years = 10
			mode = add
		}
	}

	option = {
		# Feigned submission: take the gifts, promise nothing — and
		# hear the clans mutter about a khan who takes southern silver.
		name = mr_envoys.1.b
		add_gold = 500
		add_legitimacy = legitimacy_mild_penalty
	}

	option = {
		# Melt their seals into stirrups. The south laughs; the tribes
		# cheer.
		name = mr_envoys.1.c
		add_horde_unity = horde_unity_mild_bonus
		add_prestige = prestige_weak_penalty
	}
}

mr_envoys.2 = {
	type = country_event

	illustration_tags = {

		10 = interior
	}

	dynamic_historical_event = {
		tag = MGO
		from = 1380.1.1
		to = 1405.2.1
		monthly_chance = 6
	}

	# Swallowed harmlessly in campaigns where Timur never rises — being
	# skipped when conditions fail is the flavor law's point. The bare
	# c:TIM guard is the mr_history.6 precedent (country_exists in the
	# same trigger; c:DLH there).
	trigger = {
		# The master switch gates this event like every other piece of MR
		# content. It used to be omitted here on the reasoning that story
		# flavour should play under any rule setting - but MGO can exist
		# with the switch OFF, because vanilla ships its own MGO_f formable
		# that a player can form by hand, so these events fired for someone
		# who had explicitly turned this mod's content off. CLAUDE.md also
		# states that all content checks this rule, and that statement is
		# only true once these are gated.
		NOT = {
			has_game_rule = mr_railroad_off
		}
		country_exists = c:TIM
	}

	option = {
		# Wary courtesy between conquerors — the historical steppe
		# diplomacy of the 1380s-90s; Tokhtamysh learned the price of
		# the alternative.
		name = mr_envoys.2.a
		historical_option = yes
		add_prestige = prestige_mild_bonus
	}

	option = {
		# Send his envoys back shaved. The army relishes what follows,
		# and the horde's gaze turns west for a decade.
		name = mr_envoys.2.b
		add_army_tradition = army_tradition_mild_bonus
		add_country_modifier = {
			modifier = MR_eyes_westward
			years = 10
			mode = add
		}
	}

	option = {
		# A Chinggisid bride for the Gurkani — Timur styled himself
		# son-in-law of the house of Chinggis his whole life; let him
		# pay a bride price worthy of the honor.
		name = mr_envoys.2.c
		add_gold = 750
		add_prestige = prestige_mild_penalty
	}
}
