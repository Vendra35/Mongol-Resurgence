# =================================================================
# MONGOL RESURGENCE - MESSAGE TYPES
# =================================================================
# A generic action sends a message when it is performed, and the
# message TYPE has to be registered or the engine logs
#   [message_handler.cpp:421] Failed to find message type:
#   PERFORM_MR_select_core_region_ACTION
# 149 of vanilla's 155 situation-type actions carry one, so this is
# effectively mandatory for the type.
#
# Vanilla keeps them all in main_menu/gui/messagetypes.txt (1348
# entries). This mod must NOT ship a file with that name — same path,
# same name would replace the vanilla file wholesale and take all
# 1348 with it. This is a separate filename in the same folder, on
# the assumption that the engine reads the folder rather than the one
# path. IF THE ERROR SURVIVES, that assumption is wrong: the file is
# simply ignored (it cannot do harm either way), and the cost of
# leaving it unregistered is one log line plus no popup when the
# action is used — the action itself works regardless.
#
# Field set and message_category copied from
# PERFORM_rot_select_core_region_ACTION, the vanilla action this one
# is modelled on.
# =================================================================
PERFORM_MR_select_core_region_ACTION = {
	log = yes
	onmap = no
	popup = yes
	idle = no
	option = yes
	pausepopup = no
	message_category = government
}
