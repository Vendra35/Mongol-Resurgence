# =================================================================
# MONGOL RESURGENCE (MR) - GAME RULES
# =================================================================
# Option names are global identifiers and are what `has_game_rule`
# takes, so they carry the MR_/mr_ prefix rather than bare words.
# Structure mirrors PD's PD_config.txt: per-phase conquest
# automation and a three-tier military buff rule, plus a master
# switch so the whole mod can be disabled. The buff rule also sets
# the AI railroad's war pace (Terminator / Historical / Vanilla);
# the old separate timeline-pacing rule was removed — unlike
# Prussia, the Mongol window IS the historical timeline.
# Reference: vanilla main_menu/common/game_rules (player_normal_difficulty).

# --- Master switch: all MR content on or off ---
# Every can_start / on_action / DHE in the mod checks
# NOT = { has_game_rule = mr_railroad_off }.
mr_railroad = {
	default = mr_railroad_on
	mr_railroad_on = {
		flag = general_rule
	}
	mr_railroad_off = {
		flag = general_rule
	}
}

# --- Situation 1 Conquest Automation ---
# Gates the Phase 1 completion failsafe (the 1415 territorial
# handover). The birth failsafe (MGO forming at all) is NOT gated:
# without MGO the railroad has no actor and every later phase is
# unreachable.
MR_mongol_resurgence_auto_conquest = {
	default = MR_mongol_resurgence_auto_conquest_yes
	MR_mongol_resurgence_auto_conquest_yes = {
		flag = general_rule
	}
	MR_mongol_resurgence_auto_conquest_no = {
		flag = general_rule
	}
}

# --- Imperial Expansion Automation (Situations 2 and 3) ---
# Gates the Phase 2 (1545) and Phase 3 (1645) completion failsafes.
MR_imperial_auto_conquest = {
	default = MR_imperial_auto_conquest_yes
	MR_imperial_auto_conquest_yes = {
		flag = general_rule
	}
	MR_imperial_auto_conquest_no = {
		flag = general_rule
	}
}

# --- Military Buffs and Conquest Speed ---
# Which phase buff the claimant receives, and (like PD's
# BRA_buff_enabled) how aggressively the AI railroad paces its
# wars: Terminator ~6 months, Historical ~1 year, Vanilla ~2 years
# between Phase 1 unification wars.
MR_mongol_buff_rule = {
	default = MR_buff_historical
	MR_buff_disabled = {
		flag = general_rule
	}
	MR_buff_historical = {
		flag = general_rule
	}
	MR_buff_enabled = {
		flag = general_rule
	}
}

# --- The Great Partition: does history resume on schedule? ---
# Gates the partition's TIMED secessions — the fourteen theatres
# that hand the empire back to the powers that actually held that
# ground around 1650-1700. It is the same kind of switch as the
# per-phase conquest automation above: the situation still opens,
# still scores cohesion and still ends, but with the schedule off
# nothing secedes until the empire genuinely comes apart.
#
# Default ON, because with it off the endgame is a 120-year no-op in
# most games: cohesion only falls when the Khaghan actually LOSES
# ground, and nothing in the mod makes a restored empire lose any.
# OFF is for a player running a world conquest who does not want the
# map taken back on a timer.
MR_partition_schedule = {
	default = MR_partition_schedule_yes
	MR_partition_schedule_yes = {
		flag = general_rule
	}
	MR_partition_schedule_no = {
		flag = general_rule
	}
}
